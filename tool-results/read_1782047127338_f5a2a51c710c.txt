   134→   134→const emptyTrend = (rowCount: number) => {
   135→   135→  // Demo trend derived from row count for visual interest.
   136→   136→  // Real trend would require an API endpoint.
   137→   137→  if (rowCount === 0) return { delta: 0, direction: 'none' as const }
   138→   138→  const delta = Math.round((rowCount % 13) - 5)
   139→   139→  return { delta: Math.abs(delta), direction: delta >= 0 ? ('up' as const) : ('down' as const) }
   140→   140→}

   156→   156→  const [newTableName, setNewTableName] = useState('')
   157→   157→  const [newTableDesc, setNewTableDesc] = useState('')

   250→   250→        title: 'Table deleted',
   251→   251→        description: `"${table.name}" has been removed.`,

   254→   254→    } catch (err) {
   255→   255→      toast({
   256→   256→        title: 'Failed to delete table',
   257→   257→        description: err instanceof Error ? err.message : 'Unknown error',
   258→   258→        variant: 'destructive',
   259→   259→      })

   283→   283→        prev.map((t) => (t.id === selectedTable.id ? { ...t, columns: [...t.columns, updatedCol] } : t)),

   291→   291→      toast({
   292→   292→        title: 'Failed to add column',
   293→   293→        description: err instanceof Error ? err.message : 'Unknown error',
   294→   294→        variant: 'destructive',
   295→   295→      })

   299→   299→  }
