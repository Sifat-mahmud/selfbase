    99→const TYPE_BADGE_COLORS: Record<string, string> = {
   100→  INTEGER: 'bg-amber-500/10 text-amber-700 dark:text-amber-400 border-amber-200',
   101→  TEXT: 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-400 border-emerald-200',
   102→  DECIMAL: 'bg-teal-500/10 text-teal-700 dark:text-teal-400 border-teal-200',
   103→  BOOLEAN: 'bg-cyan-500/10 text-cyan-700 dark:text-cyan-400 border-cyan-200',
   104→  TIMESTAMP: 'bg-purple-500/10 text-purple-700 dark:text-purple-400 border-purple-200',
   105→  JSON: 'bg-rose-500/10 text-rose-700 dark:text-rose-400 border-rose-200',
   106→}
   107→
   108→const TRANSFORMS: Record<string, Record<string, string>> = {
   109→  TEXT: {
   110→    INTEGER: 'parseInt',
   111→    DECIMAL: 'parseFloat',
   112→    BOOLEAN: 'truthy check',
   113→    TIMESTAMP: 'Date.parse',
   114→    JSON: 'JSON.stringify',
   115→  },
   116→  INTEGER: { TEXT: 'String()' },
   117→  DECIMAL: { TEXT: 'String()' },
   118→  BOOLEAN: { TEXT: 'String()' },
   119→}
   120→
   121→function getTransform(srcType: string, targetType: string): string | undefined {
   122→  if (srcType === targetType) return undefined
   123→  return TRANSFORMS[srcType]?.[targetType]
   124→}

   177→// ─── Main Component ──────────────────────────────────────────────────────────
   178→
   179→export function PipelineWizard({ open, onOpenChange, onCreated, tables }: PipelineWizardProps) {
   180→  const { toast } = useToast()
   181→
   182→  // Wizard state
   183→  const [step, setStep] = useState(0)
   184→  const [direction, setDirection] = useState(1)
   185→
   186→  // Step 1: URL fetch
   187→  const [url, setUrl] = useState('')
   188→  const [fetchMethod] = useState('GET')
   189→  const [fetchLoading, setFetchLoading] = useState(false)
   190→  const [fetchResult, setFetchResult] = useState<SmartPreviewResult | null>(null)
   191→
   192→  // Step 2: Data path selection
   193→  const [selectedPath, setSelectedPath] = useState<DataPath | null>(null)
   194→
   195→  // Step 3: Destination table
   196→  const [destMode, setDestMode] = useState<'existing' | 'new'>('existing')
   197→  const [selectedTableId, setSelectedTableId] = useState('')
   198→  const [newTableName, setNewTableName] = useState('')
   199→  const [creatingTable, setCreatingTable] = useState(false)
   200→  const [createdTable, setCreatedTable] = useState<{
   201→    id: string
   202→    name: string
   203→    columns: Array<{ name: string; type: string }>
   204→  } | null>(null)
   205→
   206→  // Step 4: Column mapping
   207→  const [mappings, setMappings] = useState<ColumnMapping[]>([])
   208→  const [activeSource, setActiveSource] = useState<string | null>(null)
   209→
   210→  // Step 5: Schedule
   211→  const [fetchInterval, setFetchInterval] = useState(300)
   212→  const [pipelineName, setPipelineName] = useState('')
   213→  const [showConfig, setShowConfig] = useState(false)
   214→  const [creating, setCreating] = useState(false)
   215→
   216→  // ─── Reset wizard ────────────────────────────────────────────────────────
   217→  const resetWizard = useCallback(() => {
   218→    setStep(0)
   219→    setDirection(1)
   220→    setUrl('')
   221→    setFetchResult(null)
   222→    setSelectedPath(null)
   223→    setDestMode('existing')
   224→    setSelectedTableId('')
   225→    setNewTableName('')
   226→    setCreatingTable(false)
   227→    setCreatedTable(null)
   228→    setMappings([])
   229→    setActiveSource(null)
   230→    setFetchInterval(300)
   231→    setPipelineName('')
   232→    setShowConfig(false)
   233→    setCreating(false)
   234→  }, [])
   235→
   236→  const handleClose = useCallback(
   237→    (val: boolean) => {
   238→      onOpenChange(val)
   239→      if (!val) resetWizard()
   240→    },
   241→    [onOpenChange, resetWizard]
   242→  )
   243→
   244→  // ─── Step 1: Fetch data ──────────────────────────────────────────────────
   245→  const handleFetch = async () => {
   246→    if (!url.trim()) return
   247→    setFetchLoading(true)
   248→    setFetchResult(null)
   249→    try {
   250→      const result = await apiPost<SmartPreviewResult>('/api/pipelines/smart-preview', {
   251→        url: url.trim(),
   252→        method: fetchMethod,
   253→      })
   254→      setFetchResult(result)
   255→    } catch (err) {
   256→      setFetchResult({
   257→        success: false,
   258→        error: err instanceof Error ? err.message : 'Failed to fetch URL',
   259→      })
   260→    } finally {
   261→      setFetchLoading(false)
   262→    }
   263→  }
   264→
   265→  // ─── Step 2 → Step 3: When selecting a data path ────────────────────────
   266→  const handleSelectPath = (path: DataPath) => {
   267→    setSelectedPath(path)
   268→    // Auto-generate pipeline name
   269→    try {
   270→      const u = new URL(url)
   271→      const domainPart = u.hostname.replace('www.', '')
   272→      const pathPart = u.pathname.split('/').filter(Boolean).slice(0, 2).join('_')
   273→      setPipelineName(`${domainPart}${pathPart ? '_' + pathPart : ''}_pipeline`)
   274→    } catch {
   275→      setPipelineName('new_pipeline')
   276→    }
   277→  }
   278→
   279→  // ─── Step 3: Get target columns ─────────────────────────────────────────
   280→  const targetColumns = useMemo(() => {
   281→    if (destMode === 'new' && createdTable) {
   282→      return createdTable.columns
   283→    }
   284→    if (destMode === 'new' && selectedPath) {
   285→      // Preview: source columns will become target columns
   286→      return selectedPath.columns.map((c) => ({ name: c.name, type: c.type }))
   287→    }
   288→    const tbl = tables.find((t) => t.id === selectedTableId)
   289→    return tbl?.columns || []
   290→  }, [destMode, createdTable, selectedPath, tables, selectedTableId])
   291→
   292→  const targetTableId = useMemo(() => {
   293→    if (destMode === 'new' && createdTable) return createdTable.id
   294→    return selectedTableId
   295→  }, [destMode, createdTable, selectedTableId])
   296→
   297→  // ─── Step 3: Create new table ───────────────────────────────────────────
   298→  const handleCreateTable = async () => {
   299→    if (!newTableName.trim() || !selectedPath) return
   300→    setCreatingTable(true)
   301→    try {
   302→      const result = await apiPost<{
   303→        id: string
   304→        name: string
   305→        columns: Array<{ name: string; type: string }>
   306→      }>('/api/pipelines/auto-create-table', {
   307→        name: newTableName.trim(),
   308→        columns: selectedPath.columns.map((c) => ({ name: c.name, type: c.type })),
   309→      })
   310→      setCreatedTable(result)
   311→      toast({ title: 'Table created', description: `"${result.name}" has been created.` })
   312→    } catch (err) {
   313→      toast({
   314→        title: 'Failed to create table',
   315→        description: err instanceof Error ? err.message : 'Unknown error',
   316→        variant: 'destructive',
   317→      })
   318→    } finally {
   319→      setCreatingTable(false)
   320→    }
   321→  }
   322→
   323→  // ─── Step 4: Column mapping ─────────────────────────────────────────────
   324→  const autoMap = useCallback(() => {
   325→    if (!selectedPath || targetColumns.length === 0) return
   326→    const newMappings: ColumnMapping[] = []
   327→    const srcLower = selectedPath.columns.map((c) => ({
   328→      ...c,
   329→      lower: c.name.toLowerCase().replace(/[^a-z0-9]/g, ''),
   330→    }))
   331→    const tgtUsed = new Set<string>()
   332→
   333→    for (const src of srcLower) {
   334→      const match = targetColumns.find(
   335→        (tgt) =>
   336→          !tgtUsed.has(tgt.name) &&
   337→          tgt.name.toLowerCase().replace(/[^a-z0-9]/g, '') === src.lower
   338→      )
   339→      if (match) {
   340→        tgtUsed.add(match.name)
   341→        const transform = getTransform(src.type, match.type)
   342→        newMappings.push({
   343→          src: src.name,
   344→          target: match.name,
   345→          type: match.type,
   346→          transform,
   347→        })
   348→      }
   349→    }
   350→    setMappings(newMappings)
   351→  }, [selectedPath, targetColumns])
   352→
   353→  const unmapAll = useCallback(() => {
   354→    setMappings([])
   355→    setActiveSource(null)
   356→  }, [])
   357→
   358→  const handleSourceClick = useCallback(
   359→    (srcName: string) => {
   360→      if (activeSource === srcName) {
   361→        setActiveSource(null)
   362→        return
   363→      }
   364→      // If already mapped, unmap it
   365→      const existing = mappings.find((m) => m.src === srcName)
   366→      if (existing) {
   367→        setMappings((prev) => prev.filter((m) => m.src !== srcName))
   368→        setActiveSource(null)
   369→        return
   370→      }
   371→      setActiveSource(srcName)
   372→    },
   373→    [activeSource, mappings]
   374→  )
   375→
   376→  const handleTargetClick = useCallback(
   377→    (tgtName: string) => {
   378→      if (!activeSource) return
   379→      // Remove any existing mapping to this target
   380→      const existing = mappings.find((m) => m.target === tgtName)
   381→      let newMappings = mappings.filter((m) => m.target !== tgtName && m.src !== activeSource)
   382→
   383→      const srcCol = selectedPath?.columns.find((c) => c.name === activeSource)
   384→      const tgtCol = targetColumns.find((c) => c.name === tgtName)
   385→      if (srcCol && tgtCol) {
   386→        const transform = getTransform(srcCol.type, tgtCol.type)
   387→        newMappings = [
   388→          ...newMappings,
   389→          { src: srcCol.name, target: tgtCol.name, type: tgtCol.type, transform },
   390→        ]
   391→      }
   392→      setMappings(newMappings)
   393→      setActiveSource(null)
   394→    },
   395→    [activeSource, mappings, selectedPath, targetColumns]
   396→  )
   397→
   398→  // ─── Step 5: Create pipeline ────────────────────────────────────────────
   399→  const handleCreate = async () => {
   400→    if (!pipelineName.trim() || !url.trim()) return
   401→    setCreating(true)
   402→    try {
   403→      const payload: Record<string, unknown> = {
   404→        name: pipelineName.trim(),
   405→        sourceType: fetchResult?.sourceType === 'html' ? 'scraper' : 'rest',
   406→        url: url.trim(),
   407→        method: fetchMethod,
   408→        fetchInterval,
   409→        onConflict: 'update',
   410→        isActive: fetchInterval > 0,
   411→        columnMappings: JSON.stringify(mappings),
   412→      }
   413→
   414→      if (selectedPath && selectedPath.path !== '__html_table__' && selectedPath.path !== '(root)') {
   415→        payload.jsonPath = selectedPath.path
   416→      }
   417→
   418→      if (targetTableId) {
   419→        payload.targetTableId = targetTableId
   420→      }
   421→
   422→      // Auto-detect primary key columns (first mapped column or 'id' if exists)
   423→      const pkCol = mappings.find(
   424→        (m) => m.src.toLowerCase() === 'id' || m.target.toLowerCase() === 'id'
   425→      )
   426→      if (pkCol) {
   427→        payload.primaryKeyCols = JSON.stringify([pkCol.target])
   428→      }
   429→
   430→      await apiPost('/api/pipelines', payload)
   431→      toast({
   432→        title: 'Pipeline created!',
   433→        description: `"${pipelineName}" has been created and is now ${fetchInterval > 0 ? 'active' : 'in manual mode'}.`,
   434→      })
   435→      handleClose(false)
   436→      onCreated()
   437→    } catch (err) {
   438→      toast({
   439→        title: 'Failed to create pipeline',
   440→        description: err instanceof Error ? err.message : 'Unknown error',
   441→        variant: 'destructive',
   442→      })
   443→    } finally {
   444→      setCreating(false)
   445→    }
   446→  }
   447→
   448→  // ─── Navigation ─────────────────────────────────────────────────────────
   449→  const canNext = useMemo(() => {
   450→    switch (step) {
   451→      case 0:
   452→        return fetchResult?.success && (fetchResult.dataPaths?.length ?? 0) > 0
   453→      case 1:
   454→        return selectedPath !== null
   455→      case 2:
   456→        return (
   457→          (destMode === 'existing' && selectedTableId !== '') ||
   458→          (destMode === 'new' && createdTable !== null)
   459→        )
   460→      case 3:
   461→        return true // mappings are optional
   462→      case 4:
   463→        return pipelineName.trim() !== '' && url.trim() !== ''
   464→      default:
   465→        return false
   466→    }
   467→  }, [step, fetchResult, selectedPath, destMode, selectedTableId, createdTable, pipelineName, url])
   468→
   469→  const goNext = () => {
   470→    if (step === 2 && destMode === 'new' && !createdTable) {
   471→      // Auto-create table before proceeding
   472→      handleCreateTable()
   473→      return
   474→    }
   475→    if (step === 3 && mappings.length === 0) {
   476→      autoMap()
   477→    }
   478→    setDirection(1)
   479→    setStep((s) => Math.min(s + 1, 4))
   480→  }
   481→
   482→  const goBack = () => {
   483→    setDirection(-1)
   484→    setStep((s) => Math.max(s - 1, 0))
   485→  }
   486→
   487→  // ─── Slide animation variants ───────────────────────────────────────────
   488→  const slideVariants = {
   489→    enter: (d: number) => ({ x: d > 0 ? 200 : -200, opacity: 0 }),
   490→    center: { x: 0, opacity: 1 },
   491→    exit: (d: number) => ({ x: d > 0 ? -200 : 200, opacity: 0 }),
   492→  }
   493→
   494→  // ─── Step content rendering ─────────────────────────────────────────────
   495→
   496→  const renderStep1 = () => (
   497→    <div className="space-y-4">
   498→      <div className="text-center space-y-1">
   499→        <h2 className="text-xl font-bold">Paste Your Data Source URL</h2>
   500→        <p className="text-sm text-muted-foreground">
   501→          Enter any API URL, webpage, or data feed — we&apos;ll analyze what&apos;s available
   502→        </p>
   503→      </div>
   504→
   505→      <div className="flex gap-2">
   506→        <div className="relative flex-1">
   507→          <Globe className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   508→          <Input
   509→            value={url}
   510→            onChange={(e) => setUrl(e.target.value)}
   511→            placeholder="Paste any API URL, webpage, or data feed..."
   512→            className="pl-10 h-11"
   513→            onKeyDown={(e) => {
   514→              if (e.key === 'Enter') handleFetch()
   515→            }}
   516→          />
   517→        </div>
   518→        <Button
   519→          onClick={handleFetch}
   520→          disabled={fetchLoading || !url.trim()}
   521→          className="h-11 bg-gradient-to-r from-emerald-500 to-teal-500 hover:from-emerald-600 hover:to-teal-600 text-white px-6"
   522→        >
   523→          {fetchLoading ? (
   524→            <>
   525→              <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Analyzing...
   526→            </>
   527→          ) : (
   528→            <>
   529→              <Search className="mr-2 h-4 w-4" /> Fetch Data
   530→            </>
   531→          )}
   532→        </Button>
   533→      </div>
   534→
   535→      {fetchResult && fetchResult.success && (
   536→        <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
   537→          <Card className="border-emerald-200 dark:border-emerald-800 bg-emerald-50/50 dark:bg-emerald-950/20">
   538→            <CardContent className="p-4 flex items-center gap-3">
   539→              <CheckCircle2 className="h-5 w-5 text-emerald-500 shrink-0" />
   540→              <div className="flex-1">
   541→                <p className="font-medium text-emerald-700 dark:text-emerald-400">
   542→                  Found {fetchResult.dataPaths?.length ?? 0} data table
   543→                  {(fetchResult.dataPaths?.length ?? 0) !== 1 ? 's' : ''}!
   544→                </p>
   545→                <p className="text-xs text-muted-foreground">
   546→                  Fetched in {fetchResult.durationMs ?? 0}ms
   547→                </p>
   548→              </div>
   549→              <Badge
   550→                variant="outline"
   551→                className={
   552→                  fetchResult.sourceType === 'json'
   553→                    ? 'bg-amber-500/10 text-amber-700 border-amber-200'
   554→                    : 'bg-teal-500/10 text-teal-700 border-teal-200'
   555→                }
   556→              >
   557→                {fetchResult.sourceType?.toUpperCase()}
   558→              </Badge>
   559→            </CardContent>
   560→          </Card>
   561→        </motion.div>
   562→      )}
   563→
   564→      {fetchResult && !fetchResult.success && (
   565→        <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}>
   566→          <Card className="border-red-200 dark:border-red-800 bg-red-50/50 dark:bg-red-950/20">
   567→            <CardContent className="p-4 flex items-center gap-3">
   568→              <XCircle className="h-5 w-5 text-red-500 shrink-0" />
   569→              <div className="flex-1">
   570→                <p className="font-medium text-red-700 dark:text-red-400">
   571→                  {fetchResult.error || 'Failed to fetch data'}
   572→                </p>
   573→              </div>
   574→              <Button size="sm" variant="outline" onClick={handleFetch}>
   575→                Retry
   576→              </Button>
   577→            </CardContent>
   578→          </Card>
   579→        </motion.div>
   580→      )}
   581→    </div>
