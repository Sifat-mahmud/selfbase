   200→async function fetchConfigMap(): Promise<Record<string, unknown>> {
   201→  try {
   202→    const res = await fetch('/api/config?limit=200')
   203→    if (!res.ok) return {}
   204→    const body = await res.json()
   205→    const list: Array<{ key: string; value: unknown }> = body?.data ?? body ?? []
   206→    const map: Record<string, unknown> = {}
   207→    for (const item of list) {
   208→      map[item.key] = item.value
   209→    }
   210→    return map
   211→  } catch {
   212→    return {}
   213→  }
   214→}
   215→
   216→function deriveInitial<T extends Record<string, unknown>>(
   217→  keys: string[],
   218→  configMap: Record<string, unknown>,
   219→): T {
   220→  const out = {} as Record<string, unknown>
   221→  for (const k of keys) {
   222→    const fallback = CONFIG_DEFAULTS[k]
   223→    if (k in configMap && configMap[k] !== undefined && configMap[k] !== null && configMap[k] !== '') {
   224→      out[k] = configMap[k]
   225→    } else {
   226→      out[k] = fallback
   227→    }
   228→  }
   229→  return out as T
   230→}

   232→async function saveConfig(values: Record<string, unknown>): Promise<{ ok: boolean; saved: number }> {
   233→  let saved = 0
   234→  let failed = 0
   235→  // Send sequentially to avoid race conditions on the upsert endpoint.
   236→  for (const [key, value] of Object.entries(values)) {
   237→    try {
   238→      const res = await fetch('/api/config', {
   239→        method: 'POST',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify({ key, value }),
   242→      })
   243→      if (res.ok) saved++
   244→      else failed++
   245→    } catch {
   246→      failed++
   247→    }
   248→  }
   249→  return { ok: failed === 0, saved }
   250→}

   376→function ValidationIndicator({ value, keyName }: { value: unknown; keyName: string }) {
   377→  const defaults: Record<string, unknown> = CONFIG_DEFAULTS
   378→  const isInsecureDefault =
   379→    (keyName === 'security.corsOrigins' && value === '*') ||
   380→    (keyName === 'security.ipWhitelist' && (value === '' || value == null)) ||
   381→    (keyName === 'general.adminEmail' && value === 'admin@selfbase.local') ||
   382→    (keyName === 'storage.cdnBaseUrl' && value === 'https://cdn.selfbase.local')
   383→  const isMissing =
   384→    value === '' || value === undefined || value === null
   385→  const isDefault =
   386→    keyName in defaults && value === defaults[keyName]
   387→
   388→  if (isMissing) {
   389→    return (
   390→      <Tooltip>
   391→        <TooltipTrigger asChild>
   392→          <XCircle className="h-3.5 w-3.5 text-red-500 shrink-0" />
   393→        </TooltipTrigger>
   394→        <TooltipContent>Required field is missing</TooltipContent>
   395→      </Tooltip>
   396→    )
   397→  }
   398→  if (isInsecureDefault) {
   399→    return (
   400→      <Tooltip>
   401→        <TooltipTrigger asChild>
   402→          <AlertTriangle className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   403→        </TooltipTrigger>
   404→        <TooltipContent>Using default or insecure value</TooltipContent>
   405→      </Tooltip>
   406→    )
   407→  }
   408→  if (isDefault) {
   409→    return (
   410→      <Tooltip>
   411→        <TooltipTrigger asChild>
   412→          <AlertTriangle className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   413→        </TooltipTrigger>
   414→        <TooltipContent>Using default value</TooltipContent>
   415→      </Tooltip>
   416→    )
   417→  }
   418→  return (
   419→    <Tooltip>
   420→      <TooltipTrigger asChild>
   421→        <CheckCircle2 className="h-3.5 w-3.5 text-emerald-500 shrink-0" />
   422→      </TooltipTrigger>
   423→      <TooltipContent>Properly configured</TooltipContent>
   424→    </Tooltip>
   425→  )
   426→}
