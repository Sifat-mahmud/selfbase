   274→   274→function KvEditor({ rows, onChange, keyPlaceholder = 'key', valuePlaceholder = 'value' }: {
   275→   275→  rows: KeyValueRow[]; onChange: (rows: KeyValueRow[]) => void; keyPlaceholder?: string; valuePlaceholder?: string
   276→   276→}) {
   277→   277→  const updateRow = (id: string, field: 'key' | 'value', val: string) => onChange(rows.map(r => r.id === id ? { ...r, [field]: val } : r))
   278→   278→  const removeRow = (id: string) => onChange(rows.filter(r => r.id !== id))
   279→   279→  const addRow = () => onChange([...rows, { id: newRowId(), key: '', value: '' }])
   280→   280→
   281→   281→  return (
   282→   282→    <div className="space-y-2">
   283→   283→      {rows.map(row => (
   284→   284→        <div key={row.id} className="flex items-center gap-2">
   285→   285→          <Input value={row.key} onChange={e => updateRow(row.id, 'key', e.target.value)} placeholder={keyPlaceholder} className="h-8 font-mono text-xs flex-1" />
   286→   286→          <span className="text-muted-foreground text-xs">:</span>
   287→   287→          <Input value={row.value} onChange={e => updateRow(row.id, 'value', e.target.value)} placeholder={valuePlaceholder} className="h-8 font-mono text-xs flex-1" />
   288→   288→          <Button variant="ghost" size="icon" className="h-8 w-8 shrink-0 text-muted-foreground hover:text-destructive" onClick={() => removeRow(row.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   289→   289→        </div>
   290→   290→      ))}
   291→   291→      <Button variant="outline" size="sm" className="h-7 text-xs gap-1" onClick={addRow}><Plus className="h-3 w-3" />Add row</Button>
   292→   292→    </div>
   293→   293→  )
   294→   294→}

   300→   300→function AuthFlowSection({ onTokenObtained }: { onTokenObtained: (token: string) => void }) {
   301→   301→  const { toast } = useToast()
   302→   302→  const [apiKeys, setApiKeys] = useState<ApiKeyInfo[]>([])
   303→   303→  const [creating, setCreating] = useState(false)
   304→   304→  const [newKeyName, setNewKeyName] = useState('')
   305→   305→  const [newKeyPerms, setNewKeyPerms] = useState('read,write')
   306→   306→  const [createdKey, setCreatedKey] = useState<string | null>(null)
   307→   307→  const [showCreatedKey, setShowCreatedKey] = useState(false)
   308→   308→  const [activeToken, setActiveToken] = useState<string | null>(null)
   309→   309→  const [tokenInfo, setTokenInfo] = useState<{ expiresAt: string; permissions: string[]; app: { name: string } } | null>(null)
   310→   310→  const [loginLoading, setLoginLoading] = useState(false)
   311→   311→  const [selectedApiKey, setSelectedApiKey] = useState<string>('')
   312→   312→
   313→   313→  useEffect(() => { loadApiKeys() }, [])
   314→   314→
   315→   315→  async function loadApiKeys() {
   316→   316→    try {
   317→   317→      const data = await apiGet<{ data: ApiKeyInfo[] }>('/api/api-keys')
   318→   318→      const list = Array.isArray(data) ? data : (data as any)?.data ?? []
   319→   319→      setApiKeys(list)
   320→   320→    } catch { /* ignore */ }
   321→   321→  }
   322→   322→
   323→   323→  async function createApiKey() {
   324→   324→    if (!newKeyName.trim()) return
   325→   325→    setCreating(true)
   326→   326→    try {
   327→   327→      const result = await apiPost<{ success: boolean; apiKey: { key: string } }>('/api/api-keys', {
   328→   328→        name: newKeyName.trim(),
   329→   329→        permissions: newKeyPerms,
   330→   330→      })
   331→   331→      if ((result as any).apiKey?.key) {
   332→   332→        setCreatedKey((result as any).apiKey.key)
   333→   333→        setShowCreatedKey(true)
   334→   334→        setNewKeyName('')
   335→   335→        await loadApiKeys()
   336→   336→        toast({ title: 'API Key created', description: 'Copy the key now — it won\'t be shown again!' })
   337→   337→      }
   338→   338→    } catch (err) {
   339→   339→      toast({ title: 'Failed to create key', variant: 'destructive' })
   340→   340→    } finally {
   341→   341→      setCreating(false)
   342→   342→    }
   343→   343→  }
   344→   344→
   345→   345→  async function revokeKey(id: string) {
   346→   346→    try {
   347→   347→      await apiDelete(`/api/api-keys/${id}`)
   348→   348→      await loadApiKeys()
   349→   349→      toast({ title: 'API Key revoked' })
   350→   350→    } catch {
   351→   351→      toast({ title: 'Failed to revoke key', variant: 'destructive' })
   352→   352→    }
   353→   353→  }

   355→   355→  async function loginWithApiKey() {
   356→   356→    if (!selectedApiKey) {
   357→   357→      toast({ title: 'Select an API key first', variant: 'destructive' })
   358→   358→      return
   359→   359→    }
   360→   360→    setLoginLoading(true)
   361→   361→    try {
   362→   362→      const res = await fetch('/api/v1/auth/login', {
   363→   363→        method: 'POST',
   364→   364→        headers: { Authorization: `Bearer ${selectedApiKey}` },
   365→   365→      })
   366→   366→      const data = await res.json()
   367→   367→      if (res.ok && data.token) {
   368→   368→        setActiveToken(data.token)
   369→   369→        setTokenInfo({
   370→   370→          expiresAt: data.expiresAt,
   371→   371→          permissions: data.permissions,
   372→   372→          app: data.app,
   373→   373→        })
   374→   374→        onTokenObtained(data.token)
   375→   375→        toast({ title: 'Token obtained!', description: `Valid for ${Math.round(data.expiresIn / 60)} minutes` })
   376→   376→      } else {
   377→   377→        toast({ title: 'Login failed', description: data.error || 'Invalid API key', variant: 'destructive' })
   378→   378→      }
   379→   379→    } catch {
   380→   380→      toast({ title: 'Login failed', variant: 'destructive' })
   381→   381→    } finally {
   382→   382→      setLoginLoading(false)
   383→   383→    }
   384→   384→  }
   385→   385→
   386→   386→  async function validateToken() {
   387→   387→    if (!activeToken) return
   388→   388→    try {
   389→   389→      const res = await fetch('/api/v1/auth/validate', {
   390→   390→        method: 'POST',
   391→   391→        headers: { Authorization: `Bearer ${activeToken}` },
   392→   392→      })
   393→   393→      const data = await res.json()
   394→   394→      if (data.valid) {
   395→   395→        toast({ title: 'Token is valid', description: `Expires: ${new Date(data.expiresAt).toLocaleTimeString()}` })
   396→   396→      } else {
   397→   397→        toast({ title: 'Token invalid', description: data.error, variant: 'destructive' })
   398→   398→        setActiveToken(null)
   399→   399→        setTokenInfo(null)
   400→   400→      }
   401→   401→    } catch {
   402→   402→      toast({ title: 'Validation failed', variant: 'destructive' })
   403→   403→    }
   404→   404→  }
   405→   405→

   413→   413→      setActiveToken(null)
   414→   414→      setTokenInfo(null)
   415→   415→      toast({ title: 'Token revoked' })
   416→   416→    } catch { /* ignore */ }
   417→   417→  }

   423→   423→        <CardHeader className="pb-3">
   424→   424→          <CardTitle className="text-base flex items-center gap-2">
   425→   425→            <Shield className="h-5 w-5 text-emerald-600" />
   426→   426→            API Authentication Flow
   427→   427→          </CardTitle>
   428→   428→          <CardDescription>
   429→   429→            How external apps (iOS, Android, web) authenticate with SelfBase
   430→   430→          </CardDescription>
