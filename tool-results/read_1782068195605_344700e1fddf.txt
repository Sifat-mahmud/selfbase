   155→   155→/**
   156→   156→ * Render a type-aware editor cell for inline editing.
   157→   157→ * - BOOLEAN  -> Switch
   158→   158→ * - INTEGER/DECIMAL -> number Input
   159→   159→ * - JSON/TIMESTAMP -> wider Input
   160→   160→ * - TEXT/default  -> Input
   161→   161→ */
   162→   162→function renderCellInput(
   163→   163→  col: ColumnItem,
   164→   164→  value: unknown,
   165→   165→  onChange: (v: unknown) => void,
   166→   166→  disabled?: boolean,
   167→   167→) {
   168→   168→  const typeUpper = col.type.toUpperCase()
   169→   169→  if (typeUpper === 'BOOLEAN') {
   170→   170→    return (
   171→   171→      <Switch
   172→   172→        checked={Boolean(value)}
   173→   173→        onCheckedChange={(v) => onChange(v)}
   174→   174→        disabled={disabled}
   175→   175→        aria-label={col.name}
   176→   176→      />
   177→   177→    )
   178→   178→  }
   179→   179→  if (typeUpper === 'INTEGER' || typeUpper === 'DECIMAL') {
   180→   180→    return (
   181→   181→      <Input
   182→   182→        type="number"
   183→   183→        value={value === null || value === undefined ? '' : String(value)}
   184→   184→        onChange={(e) => {
   185→   185→          const raw = e.target.value
   186→   186→          if (raw === '') {
   187→   187→            onChange(null)
   188→   188→          } else if (typeUpper === 'INTEGER') {
   189→   189→            const n = parseInt(raw, 10)
   190→   190→            onChange(Number.isNaN(n) ? null : n)
   191→   191→          } else {
   192→   192→            const n = parseFloat(raw)
   193→   193→            onChange(Number.isNaN(n) ? null : n)
   194→   194→          }
   195→   195→        }}
   196→   196→        disabled={disabled}
   197→   197→        className="h-8 font-mono text-xs min-w-[80px]"
   198→   198→        aria-label={col.name}
   199→   199→      />
   200→   200→    )
   201→   201→  }
   202→   202→  if (typeUpper === 'JSON' || typeUpper === 'TIMESTAMP') {
   203→   203→    return (
   204→   204→      <Input
   205→   205→        value={value === null || value === undefined ? '' : String(value)}
   206→   206→        onChange={(e) => onChange(e.target.value)}
   207→   207→        disabled={disabled}
   208→   208→        className="h-8 font-mono text-xs min-w-[140px]"
   209→   209→        placeholder={typeUpper === 'JSON' ? '{...}' : '2024-01-01T00:00:00Z'}
   210→   210→        aria-label={col.name}
   211→   211→      />
   212→   212→    )
   213→   213→  }
   214→   214→  return (
   215→   215→    <Input
   216→   216→      value={value === null || value === undefined ? '' : String(value)}
   217→   217→      onChange={(e) => onChange(e.target.value)}
   218→   218→      disabled={disabled}
   219→   219→      className="h-8 font-mono text-xs min-w-[100px]"
   220→   220→      aria-label={col.name}
   221→   221→    />
   222→   222→  )
   223→   223→}

   612→   612→  // Client-side filter: any cell value contains the query (case-insensitive)
   613→   613→  const filteredRows = useMemo(() => {

   615→   615→    const q = searchQuery.toLowerCase()
   616→   616→    return dataRows.filter((row) => {

   632→   632→      let av: unknown = undefined
   633→   633→      let bv: unknown = undefined
   634→   634→      try {
   635→   635→        av = JSON.parse(a.data)[sortColumn]
   636→   636→      } catch {
   637→   637→        av = undefined
   638→   638→      }
   639→   639→      try {
   640→   640→        bv = JSON.parse(b.data)[sortColumn]
   641→   641→      } catch {
   642→   642→        bv = undefined
   643→   643→      }
   644→   644→      if (av === null || av === undefined) return 1
   645→   645→      if (bv === null || bv === undefined) return -1
   646→   646→      if (typeof av === 'number' && typeof bv === 'number') {
   647→   647→        return sortDirection === 'asc' ? av - bv : bv - av
   648→   648→      }
   649→   649→      const cmp = String(av).localeCompare(String(bv))
   650→   650→      return sortDirection === 'asc' ? cmp : -cmp
   651→   651→    })
   652→   652→  }, [filteredRows, sortColumn, sortDirection])
