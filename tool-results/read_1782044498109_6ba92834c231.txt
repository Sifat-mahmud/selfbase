   138→  const { input, env } = ctx;
   139→  // Your code here

   141→  return { ok: true, received: Object.keys(input || {}).length };

   144→
   145→export function FunctionsView() {
   146→  const { toast } = useToast()
   147→  const [functions, setFunctions] = useState<SbFunctionItem[]>([])
   148→  const [loading, setLoading] = useState(true)
   149→  const [search, setSearch] = useState('')
   150→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   151→  const [selectedFunction, setSelectedFunction] = useState<SbFunctionItem | null>(null)
   152→  const [runningId, setRunningId] = useState<string | null>(null)
   153→
   154→  // Create form
   155→  const [newName, setNewName] = useState('')
   156→  const [newDesc, setNewDesc] = useState('')
   157→  const [newRuntime, setNewRuntime] = useState('javascript')
   158→  const [newTrigger, setNewTrigger] = useState<'http' | 'schedule' | 'event'>('http')
   159→  const [newTimeout, setNewTimeout] = useState('30000')
   160→  const [newMemory, setNewMemory] = useState('128')
   161→  const [newCron, setNewCron] = useState('')
   162→  const [newEventPattern, setNewEventPattern] = useState('')
   163→  const [newEnvVars, setNewEnvVars] = useState('{}')
   164→  const [codeText, setCodeText] = useState(defaultCode)
   165→
   166→  const loadAll = useCallback(async () => {
   167→    setLoading(true)
   168→    try {
   169→      const data = await apiGet<SbFunctionItem[]>('/api/functions')
   170→      setFunctions(Array.isArray(data) ? data : [])
   171→    } catch (err) {
   172→      toast({
   173→        title: 'Failed to load functions',
   174→        description: err instanceof Error ? err.message : 'Unknown error',
   175→        variant: 'destructive',
   176→      })
   177→    } finally {
   178→      setLoading(false)
   179→    }
   180→  }, [toast])
   181→
   182→  useEffect(() => {
   183→    void loadAll()
   184→  }, [loadAll])
   185→
   186→  const handleCreate = async () => {
   187→    if (!newName.trim()) {
   188→      toast({ title: 'Name required', variant: 'destructive' })
   189→      return
   190→    }
   191→    const triggerConfig: Record<string, unknown> = {}
   192→    if (newTrigger === 'schedule' && newCron) triggerConfig.cron = newCron
   193→    if (newTrigger === 'event' && newEventPattern) triggerConfig.pattern = newEventPattern
   194→    try {
   195→      const created = await apiPost<SbFunctionItem>('/api/functions', {
   196→        name: newName.trim(),
   197→        description: newDesc || null,
   198→        code: codeText,
   199→        runtime: newRuntime,
   200→        triggerType: newTrigger,
   201→        triggerConfig: Object.keys(triggerConfig).length > 0 ? JSON.stringify(triggerConfig) : null,
   202→        envVars: newEnvVars || null,
   203→        timeoutMs: Number(newTimeout) || 30000,
   204→        memoryMb: Number(newMemory) || 128,
   205→        isActive: true,
   206→      })
   207→      setFunctions((prev) => [{ ...created, functionRuns: [] }, ...prev])
   208→      setShowCreateDialog(false)
   209→      setNewName('')
   210→      setNewDesc('')
   211→      setNewRuntime('javascript')
   212→      setNewTrigger('http')
   213→      setNewTimeout('30000')
   214→      setNewMemory('128')
   215→      setNewCron('')
   216→      setNewEventPattern('')
   217→      setNewEnvVars('{}')
   218→      setCodeText(defaultCode)
   219→      toast({ title: 'Function created', description: `"${created.name}" has been deployed.` })
   220→    } catch (err) {
   221→      toast({
   222→        title: 'Failed to create function',
   223→        description: err instanceof Error ? err.message : 'Unknown error',
   224→        variant: 'destructive',
   225→      })
   226→    }
   227→  }
   228→
   229→  const handleToggleActive = async (fn: SbFunctionItem) => {
   230→    try {
   231→      await apiPut(`/api/functions/${fn.id}`, { isActive: !fn.isActive })
   232→      setFunctions((prev) =>
   233→        prev.map((f) => (f.id === fn.id ? { ...f, isActive: !f.isActive } : f)),
   234→      )
   235→      if (selectedFunction?.id === fn.id) {
   236→        setSelectedFunction((prev) => (prev ? { ...prev, isActive: !prev.isActive } : prev))
   237→      }
   238→      toast({ title: `Function ${fn.isActive ? 'deactivated' : 'activated'}` })
   239→    } catch (err) {
   240→      toast({
   241→        title: 'Failed to toggle function',
   242→        description: err instanceof Error ? err.message : 'Unknown error',
   243→        variant: 'destructive',
   244→      })
   245→    }
   246→  }
   247→
   248→  const handleRun = async (fn: SbFunctionItem) => {
   249→    setRunningId(fn.id)
   250→    try {
   251→      const result = await apiPost<{ runId: string; status: string; output?: unknown; durationMs?: number; error?: string }>(
   252→        `/api/functions/${fn.id}/run`,
   253→        {},
   254→      )
   255→      toast({
   256→        title: result.status === 'success' ? 'Function executed' : 'Function run failed',
   257→        description:
   258→          result.status === 'success'
   259→            ? `Completed in ${result.durationMs ?? 0}ms.`
   260→            : result.error ?? `Status: ${result.status}`,
   261→        variant: result.status === 'success' ? 'default' : 'destructive',
   262→      })
   263→      await loadAll()
   264→      if (selectedFunction?.id === fn.id) {
   265→        const fresh = (await apiGet<SbFunctionItem[]>('/api/functions')).find((f) => f.id === fn.id)
   266→        if (fresh) setSelectedFunction(fresh)
   267→      }
   268→    } catch (err) {
   269→      toast({
   270→        title: 'Function run failed',
   271→        description: err instanceof Error ? err.message : 'Unknown error',
   272→        variant: 'destructive',
   273→      })
   274→    } finally {
   275→      setRunningId(null)
   276→    }
   277→  }
   278→
   279→  const handleDelete = async (fn: SbFunctionItem) => {
   280→    try {
   281→      await apiDelete(`/api/functions/${fn.id}`)
   282→      setFunctions((prev) => prev.filter((f) => f.id !== fn.id))
   283→      if (selectedFunction?.id === fn.id) setSelectedFunction(null)
   284→      toast({ title: 'Function deleted', variant: 'destructive' })
   285→    } catch (err) {
   286→      toast({
   287→        title: 'Failed to delete function',
   288→        description: err instanceof Error ? err.message : 'Unknown error',
   289→        variant: 'destructive',
   290→      })
   291→    }
   292→  }
   293→
   294→  const filteredFunctions = functions.filter(
   295→    (f) =>
   296→      f.name.toLowerCase().includes(search.toLowerCase()) ||
   297→      (f.description ?? '').toLowerCase().includes(search.toLowerCase()),
   298→  )
   299→
   300→  if (loading) {
   301→    return (
   302→      <div className="space-y-6">
   303→        <Skeleton className="h-8 w-48" />
   304→        <div className="grid gap-4 md:grid-cols-2">
   305→          {Array.from({ length: 4 }).map((_, i) => (
   306→            <Card key={i}>
   307→              <CardContent className="p-4">
   308→                <Skeleton className="h-40 w-full" />
   309→              </CardContent>
   310→            </Card>
   311→          ))}
   312→        </div>
   313→      </div>
   314→    )
   315→  }
   316→
   317→  if (selectedFunction) {
   318→    const TriggerIcon = triggerIcons[selectedFunction.triggerType]
   319→    const runs = selectedFunction.functionRuns ?? []
   320→    return (
   321→      <motion.div
   322→        initial={{ opacity: 0, y: 8 }}
   323→        animate={{ opacity: 1, y: 0 }}
   324→        transition={{ duration: 0.2 }}
   325→        className="space-y-6"
   326→      >
   327→        <div className="flex items-center gap-2 flex-wrap">
   328→          <Button variant="ghost" size="sm" onClick={() => setSelectedFunction(null)}>
   329→            <ArrowRight className="h-4 w-4 rotate-180" /> Back
   330→          </Button>
   331→          <div className="h-6 w-px bg-border" />
   332→          <div className="rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 p-2 text-white">
   333→            <Code2 className="h-5 w-5" />
   334→          </div>
   335→          <h1 className="text-2xl font-bold tracking-tight">{selectedFunction.name}</h1>
   336→          <Badge variant="outline" className={`capitalize ${triggerBadgeColors[selectedFunction.triggerType]}`}>
   337→            <TriggerIcon className="h-3 w-3 mr-1" />
   338→            {selectedFunction.triggerType}
   339→          </Badge>
   340→          <Badge variant={selectedFunction.isActive ? 'default' : 'secondary'}>
   341→            {selectedFunction.isActive ? 'Active' : 'Inactive'}
   342→          </Badge>
   343→          <div className="ml-auto flex gap-2">
   344→            <Button
   345→              size="sm"
   346→              onClick={() => void handleRun(selectedFunction)}
   347→              disabled={runningId === selectedFunction.id || !selectedFunction.isActive}
   348→            >
   349→              {runningId === selectedFunction.id ? (
   350→                <Loader2 className="mr-1 h-3.5 w-3.5 animate-spin" />
   351→              ) : (
   352→                <Play className="mr-1 h-3.5 w-3.5" />
   353→              )}
   354→              Run
   355→            </Button>
   356→          </div>
   357→        </div>
   358→
   359→        <div className="grid gap-4 md:grid-cols-4">
   360→          <Card>
   361→            <CardContent className="pt-4">
   362→              <div className="text-sm text-muted-foreground">Trigger</div>
   363→              <div className="flex items-center gap-1 text-lg font-bold capitalize">
   364→                <TriggerIcon className="h-4 w-4" />
   365→                {selectedFunction.triggerType}
   366→              </div>
   367→            </CardContent>
   368→          </Card>
   369→          <Card>
   370→            <CardContent className="pt-4">
   371→              <div className="text-sm text-muted-foreground">Timeout</div>
   372→              <div className="text-lg font-bold">{selectedFunction.timeoutMs / 1000}s</div>
   373→            </CardContent>
   374→          </Card>
   375→          <Card>
   376→            <CardContent className="pt-4">
   377→              <div className="text-sm text-muted-foreground">Memory</div>
   378→              <div className="text-lg font-bold">{selectedFunction.memoryMb} MB</div>
   379→            </CardContent>
   380→          </Card>
   381→          <Card>
   382→            <CardContent className="pt-4">
   383→              <div className="text-sm text-muted-foreground">Total Runs</div>
   384→              <div className="text-lg font-bold">{runs.length}</div>
   385→            </CardContent>
   386→          </Card>
   387→        </div>
   388→
   389→        <Card>
   390→          <CardHeader className="flex flex-row items-center justify-between">
   391→            <div>
   392→              <CardTitle className="text-base">Source Code</CardTitle>
   393→              <CardDescription>
   394→                {selectedFunction.runtime} · {selectedFunction.code.split('\n').length} lines
   395→              </CardDescription>
   396→            </div>
   397→            <div className="flex gap-2">
   398→              <Tooltip>
   399→                <TooltipTrigger asChild>
   400→                  <Button
   401→                    variant="outline"
   402→                    size="sm"
   403→                    onClick={() => {
   404→                      navigator.clipboard.writeText(selectedFunction.code)
   405→                      toast({ title: 'Code copied' })
   406→                    }}
   407→                  >
   408→                    <Copy className="h-3.5 w-3.5 mr-1" /> Copy
   409→                  </Button>
   410→                </TooltipTrigger>
   411→                <TooltipContent>Copy code to clipboard</TooltipContent>
   412→              </Tooltip>
   413→              <Button size="sm" onClick={() => void handleRun(selectedFunction)} disabled={runningId === selectedFunction.id || !selectedFunction.isActive}>
   414→                <Play className="h-3.5 w-3.5 mr-1" /> Run
   415→              </Button>
   416→            </div>
   417→          </CardHeader>
   418→          <CardContent>
   419→            <div className="rounded-md bg-slate-950 text-emerald-400 p-4 overflow-auto max-h-[400px]">
   420→              <pre className="text-sm font-mono whitespace-pre-wrap">{selectedFunction.code}</pre>
   421→            </div>
   422→          </CardContent>
   423→        </Card>
   424→
   425→        {selectedFunction.triggerConfig && (
   426→          <Card>
   427→            <CardHeader>
   428→              <CardTitle className="text-base">Trigger Configuration</CardTitle>
   429→              <CardDescription>How this function is invoked</CardDescription>
   430→            </CardHeader>
   431→            <CardContent>
   432→              <div className="rounded-md bg-muted/50 p-3">
   433→                <pre className="text-sm font-mono">
   434→                  {JSON.stringify(parseJsonField(selectedFunction.triggerConfig, {}), null, 2)}
   435→                </pre>
   436→              </div>
   437→            </CardContent>
   438→          </Card>
   439→        )}
   440→
   441→        <Card>
   442→          <CardHeader>
   443→            <CardTitle className="text-base">Run History</CardTitle>
   444→            <CardDescription>Recent function executions</CardDescription>
   445→          </CardHeader>
   446→          <CardContent>
   447→            {runs.length === 0 ? (
   448→              <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   449→                <Terminal className="h-10 w-10 mb-2 opacity-40" />
   450→                <p className="text-sm">No runs yet</p>
   451→                <p className="text-xs mt-1">Click "Run" to execute this function.</p>
   452→              </div>
   453→            ) : (
   454→              <div className="rounded-md border max-h-96 overflow-y-auto">
   455→                <Table>
   456→                  <TableHeader className="sticky top-0 bg-background">
   457→                    <TableRow>
   458→                      <TableHead>Status</TableHead>
   459→                      <TableHead>Trigger</TableHead>
   460→                      <TableHead>Duration</TableHead>
   461→                      <TableHead>Memory</TableHead>
   462→                      <TableHead>Started</TableHead>
   463→                    </TableRow>
   464→                  </TableHeader>
   465→                  <TableBody>
   466→                    {runs.map((run) => {
   467→                      const RunIcon = statusIcons[run.status]
   468→                      return (
   469→                        <TableRow key={run.id} className="hover:bg-muted/40">
   470→                          <TableCell>
   471→                            <div className="flex items-center gap-1.5">
   472→                              <RunIcon className={`h-3.5 w-3.5 ${statusColors[run.status]}`} />
   473→                              <span className={`text-sm capitalize font-medium ${statusColors[run.status]}`}>
   474→                                {run.status}
   475→                              </span>
   476→                            </div>
   477→                          </TableCell>
   478→                          <TableCell>
   479→                            <Badge variant="outline" className="text-xs capitalize">
   480→                              {run.triggeredBy ?? '—'}
   481→                            </Badge>
   482→                          </TableCell>
   483→                          <TableCell className="font-mono text-sm">
   484→                            {run.durationMs ? `${run.durationMs}ms` : '—'}
   485→                          </TableCell>
   486→                          <TableCell className="text-sm">
   487→                            {run.memoryUsedMb ? `${Math.round(run.memoryUsedMb)} MB` : '—'}
   488→                          </TableCell>
   489→                          <TableCell className="text-sm text-muted-foreground">
   490→                            {new Date(run.startedAt).toLocaleString()}
   491→                          </TableCell>
   492→                        </TableRow>
   493→                      )
   494→                    })}
   495→                  </TableBody>
   496→                </Table>
   497→              </div>
   498→            )}
   499→          </CardContent>
   500→        </Card>
   501→      </motion.div>
   502→    )
   503→  }
   504→
   505→  return (
   506→    <motion.div
   507→      initial={{ opacity: 0, y: 8 }}
   508→      animate={{ opacity: 1, y: 0 }}
   509→      transition={{ duration: 0.2 }}
   510→      className="space-y-6"
   511→    >
   512→      <div className="flex items-center justify-between flex-wrap gap-2">
   513→        <div>
   514→          <h1 className="text-2xl font-bold tracking-tight bg-gradient-to-r from-emerald-600 to-teal-600 bg-clip-text text-transparent">
   515→            Functions
   516→          </h1>
   517→          <p className="text-muted-foreground">Manage serverless functions and triggers</p>
   518→        </div>
   519→        <Dialog open={showCreateDialog} onOpenChange={setShowCreateDialog}>
   520→          <DialogTrigger asChild>
   521→            <Button>
   522→              <Plus className="mr-1 h-4 w-4" /> New Function
   523→            </Button>
   524→          </DialogTrigger>
   525→          <DialogContent className="max-w-3xl">
   526→            <DialogHeader>
   527→              <DialogTitle>Create Function</DialogTitle>
   528→              <DialogDescription>Deploy a new serverless function</DialogDescription>
   529→            </DialogHeader>
   530→            <div className="space-y-4 py-2 max-h-[65vh] overflow-y-auto">
   531→              <div className="grid gap-3 md:grid-cols-2">
   532→                <div className="space-y-2">
   533→                  <Label>Function Name</Label>
   534→                  <Input value={newName} onChange={(e) => setNewName(e.target.value)} placeholder="myFunction" />
   535→                </div>
   536→                <div className="space-y-2">
   537→                  <Label>Runtime</Label>
   538→                  <Select value={newRuntime} onValueChange={setNewRuntime}>
   539→                    <SelectTrigger>
   540→                      <SelectValue />
   541→                    </SelectTrigger>
   542→                    <SelectContent>
   543→                      <SelectItem value="javascript">JavaScript</SelectItem>
   544→                      <SelectItem value="typescript">TypeScript</SelectItem>
   545→                    </SelectContent>
   546→                  </Select>
   547→                </div>
   548→              </div>
   549→              <div className="space-y-2">
   550→                <Label>Description</Label>
   551→                <Input value={newDesc} onChange={(e) => setNewDesc(e.target.value)} placeholder="What does this function do?" />
   552→              </div>
   553→              <div className="grid gap-3 md:grid-cols-3">
   554→                <div className="space-y-2">
   555→                  <Label>Trigger Type</Label>
   556→                  <Select value={newTrigger} onValueChange={(v) => setNewTrigger(v as typeof newTrigger)}>
   557→                    <SelectTrigger>
   558→                      <SelectValue />
   559→                    </SelectTrigger>
   560→                    <SelectContent>
   561→                      <SelectItem value="http">HTTP Endpoint</SelectItem>
   562→                      <SelectItem value="schedule">Scheduled (Cron)</SelectItem>
   563→                      <SelectItem value="event">Event-driven</SelectItem>
   564→                    </SelectContent>
   565→                  </Select>
   566→                </div>
   567→                <div className="space-y-2">
   568→                  <Label>Timeout (ms)</Label>
   569→                  <Input type="number" value={newTimeout} onChange={(e) => setNewTimeout(e.target.value)} />
   570→                </div>
   571→                <div className="space-y-2">
   572→                  <Label>Memory (MB)</Label>
   573→                  <Input type="number" value={newMemory} onChange={(e) => setNewMemory(e.target.value)} />
