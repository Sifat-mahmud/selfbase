   169→export function MonitoringView() {
   170→  const { toast } = useToast()

   172→  const [recording, setRecording] = useState(false)
   173→  const [heartbeatData, setHeartbeatData] = useState<HeartbeatRecord[]>([])

   182→  // Alert form

   188→  const [newAlertEmail, setNewAlertEmail] = useState('')

   201→      setAlertConfigs(Array.isArray(alerts) ? alerts : [])
   202→      setTableMetrics(Array.isArray(metrics) ? metrics : [])

   204→      setUptimeInfo(uptime ?? null)

   209→          id: `evt-${a.id}`,

   212→          metricValue: a.threshold * 1.15,
   213→          threshold: a.threshold,

   215→          isResolved: true,
   216→          resolvedAt: a.lastTriggeredAt,
   217→          createdAt: a.lastTriggeredAt as string,
   218→        }))
   219→      setAlertEvents(events)
   220→    } catch (err) {
   221→      toast({
   222→        title: 'Failed to load monitoring data',
   223→        description: err instanceof Error ? err.message : 'Unknown error',
   224→        variant: 'destructive',
   225→      })
   226→    } finally {
   227→      setLoading(false)
   228→    }
   229→  }, [toast])
   230→

   232→    void loadAll()

   234→

   237→      toast({ title: 'Threshold required', variant: 'destructive' })

   239→    }
   240→    try {
   241→      const created = await apiPost<AlertConfigItem>('/api/monitoring/alerts', {
   242→        metricType: newAlertMetric,
   243→        threshold: Number(newAlertThreshold),
   244→        operator: newAlertOperator,
   245→        duration: Number(newAlertDuration) || 300,
   246→        webhookUrl: newAlertWebhook || null,
   247→        emailTo: newAlertEmail || null,
   248→        isEnabled: true,

   250→      setAlertConfigs((prev) => [created, ...prev])

   252→      setNewAlertThreshold('')

   257→      toast({
   258→        title: 'Failed to create alert',
   259→        description: err instanceof Error ? err.message : 'Unknown error',
   260→        variant: 'destructive',
   261→      })
   262→    }
   263→  }
   264→
   265→  const handleDeleteAlert = async (id: string) => {
   266→    try {
   267→      await apiDelete(`/api/monitoring/alerts/${id}`)
   268→      setAlertConfigs((prev) => prev.filter((a) => a.id !== id))

   271→      toast({
   272→        title: 'Failed to delete alert',
   273→        description: err instanceof Error ? err.message : 'Unknown error',
   274→        variant: 'destructive',
   275→      })
   276→    }
   277→  }
   278→

   281→      await apiPost<AlertConfigItem>(`/api/monitoring/alerts/${alert.id}`, {
   282→        isEnabled: !alert.isEnabled,
   283→      }).catch(async () => {
   284→        // fallback to PUT
   285→        const { apiPut } = await import('@/lib/api-client')

   287→          isEnabled: !alert.isEnabled,
   288→        })
   289→      })
   290→      setAlertConfigs((prev) =>
   291→        prev.map((a) => (a.id === alert.id ? { ...a, isEnabled: !a.isEnabled } : a)),
   292→      )
   293→      toast({ title: `Alert ${alert.isEnabled ? 'disabled' : 'enabled'}` })
   294→    } catch (err) {
   295→      toast({
   296→        title: 'Failed to update alert',
   297→        description: err instanceof Error ? err.message : 'Unknown error',
   298→        variant: 'destructive',
   299→      })
   300→    }
   301→  }
   302→

   310→        cpuTotal,
   311→        cpuScraper: Math.floor(cpuTotal * 0.25),
   312→        cpuApi: Math.floor(cpuTotal * 0.5),
   313→        cpuFunctions: Math.floor(cpuTotal * 0.25),
   314→        ramUsedMb: ramUsed,
   315→        diskUsedMb: Math.floor(1800 + Math.random() * 400),
   316→        activeConnections: Math.floor(8 + Math.random() * 30),
   317→        reqPerSec: Math.floor(50 + Math.random() * 200),
   318→        intervalSec: 60,
   319→        loadScore: Math.floor(cpuTotal * 0.8 + Math.random() * 15),
   320→      })
   321→      toast({ title: 'Heartbeat recorded', description: 'Refreshed live metrics.' })
   322→      await loadAll()
   323→    } catch (err) {
   324→      toast({
   325→        title: 'Failed to record heartbeat',
   326→        description: err instanceof Error ? err.message : 'Unknown error',
   327→        variant: 'destructive',
   328→      })
   329→    } finally {
   330→      setRecording(false)
   331→    }
   332→  }
   333→
   334→  if (loading) {
   335→    return (
   336→      <div className="space-y-6">
   337→        <Skeleton className="h-8 w-48" />
   338→        <div className="grid gap-4 md:grid-cols-4">
   339→          {Array.from({ length: 4 }).map((_, i) => (
   340→            <Card key={i}>
   341→              <CardContent className="p-4">
   342→                <Skeleton className="h-16 w-full" />
   343→              </CardContent>
   344→            </Card>
   345→          ))}
   346→        </div>
   347→        <Skeleton className="h-[300px] w-full" />
   348→      </div>
   349→    )
   350→  }
   351→
   352→  const latest = heartbeatData[0]
   353→  const uptimePercent = uptimeInfo?.uptimePercent ?? 100
   354→  const cpuVal = loadInfo?.cpu.total ?? latest?.cpuTotal ?? 0
   355→  const ramPercent = loadInfo?.memory.percent ?? (latest ? Math.round((latest.ramUsedMb / 4096) * 100) : 0)
   356→  const loadScore = loadInfo?.loadScore ?? latest?.loadScore ?? 0
   357→
   358→  const chartData = [...heartbeatData]
   359→    .reverse()
   360→    .map((h) => ({
   361→      time: new Date(h.recordedAt).toLocaleTimeString('en-US', {
   362→        hour: '2-digit',
   363→        minute: '2-digit',
   364→      }),
   365→      cpu: h.cpuTotal,
   366→      ram: Math.round((h.ramUsedMb / 4096) * 100),
   367→      disk: Math.round((h.diskUsedMb / 5120) * 100),
   368→      load: h.loadScore,
   369→      rps: h.reqPerSec,
   370→      connections: h.activeConnections,
   371→      recordedAt: h.recordedAt,
   372→    }))
   373→
   374→  // Aggregate table call metrics by tableName
   375→  const aggregatedMetrics: TableCallItem[] = (() => {
   376→    const map = new Map<string, TableCallItem>()
   377→    for (const m of tableMetrics) {
   378→      const existing = map.get(m.tableName)
   379→      if (!existing) {
   380→        map.set(m.tableName, { ...m })
   381→      } else {
   382→        existing.callCount += m.callCount
   383→        existing.errorCount += m.errorCount
   384→        existing.avgLatencyMs = Math.round((existing.avgLatencyMs + m.avgLatencyMs) / 2)
   385→        existing.maxLatencyMs = Math.max(existing.maxLatencyMs, m.maxLatencyMs)
   386→      }
   387→    }
   388→    return Array.from(map.values()).sort((a, b) => b.callCount - a.callCount)
   389→  })()
   390→
   391→  // Segmented uptime bar
   392→  const uptimeSegments = (() => {
   393→    if (!uptimeInfo || uptimeInfo.heartbeatCount === 0) {
   394→      return Array.from({ length: 30 }, () => 'down' as const)
   395→    }
   396→    // Map downtime periods to segments (30 segments across the range)
   397→    const segments: ('up' | 'down')[] = []
   398→    const totalMs = uptimeInfo.totalMs
   399→    const segCount = 30
   400→    const segMs = totalMs / segCount
   401→    const from = new Date(uptimeInfo.range.from).getTime()
   402→    for (let i = 0; i < segCount; i++) {
   403→      const segStart = from + i * segMs
   404→      const segEnd = segStart + segMs
   405→      const isDown = uptimeInfo.downtimePeriods.some(
   406→        (p) =>
   407→          new Date(p.end).getTime() > segStart && new Date(p.start).getTime() < segEnd,
   408→      )
   409→      segments.push(isDown ? 'down' : 'up')
   410→    }
   411→    return segments
   412→  })()

   449→                <Heart className="h-5 w-5 text-emerald-600" />
   450→              </div>
   451→            </div>
   452→            <Progress value={uptimePercent} className="mt-2 h-1.5" />
   453→            <p className="mt-1 text-xs text-muted-foreground">
   454→              {uptimeInfo?.heartbeatCount ?? 0} heartbeats · {uptimeInfo?.downtimePeriods.length ?? 0} downtime periods
   455→            </p>
   456→          </CardContent>
   457→        </Card>
   458→        <Card>
   459→          <CardContent className="pt-4">
   460→            <div className="flex items-center justify-between">
   461→              <div>
   462→                <div className="text-sm text-muted-foreground">CPU Usage</div>
   463→                <div className="text-2xl font-bold">{cpuVal}%</div>
   464→              </div>
   465→              <div className="rounded-md bg-amber-500/10 p-2">
   466→                <Cpu className="h-5 w-5 text-amber-600" />
   467→              </div>
   468→            </div>
   469→            <Progress value={cpuVal} className="mt-2 h-1.5" />
   470→            <p className="mt-1 text-xs text-muted-foreground">
   471→              API {loadInfo?.cpu.api ?? latest?.cpuApi ?? 0}% · Scraper {loadInfo?.cpu.scraper ?? latest?.cpuScraper ?? 0}% · Fn {loadInfo?.cpu.functions ?? latest?.cpuFunctions ?? 0}%
   472→            </p>
   473→          </CardContent>
   474→        </Card>
   475→        <Card>
   476→          <CardContent className="pt-4">
   477→            <div className="flex items-center justify-between">
   478→              <div>
   479→                <div className="text-sm text-muted-foreground">RAM Usage</div>
   480→                <div className="text-2xl font-bold">{ramPercent}%</div>
   481→              </div>
   482→              <div className="rounded-md bg-teal-500/10 p-2">
   483→                <HardDrive className="h-5 w-5 text-teal-600" />
   484→              </div>
   485→            </div>
   486→            <Progress value={ramPercent} className="mt-2 h-1.5" />
   487→            <p className="mt-1 text-xs text-muted-foreground">
   488→              {loadInfo?.memory.usedMb ?? latest?.ramUsedMb ?? 0} MB / {loadInfo?.memory.totalMb ?? 4096} MB
   489→            </p>
   490→          </CardContent>
   491→        </Card>
   492→        <Card>
   493→          <CardContent className="pt-4">
   494→            <div className="flex items-center justify-between">
   495→              <div>
   496→                <div className="text-sm text-muted-foreground">Load Score</div>
   497→                <div className="text-2xl font-bold">{Math.round(loadScore)}/100</div>
   498→              </div>
   499→              <div className="rounded-md bg-emerald-500/10 p-2">
   500→                <Gauge className="h-5 w-5 text-emerald-600" />
   501→              </div>
   502→            </div>
   503→            <Progress value={loadScore} className="mt-2 h-1.5" />
   504→            <p className="mt-1 text-xs text-muted-foreground capitalize">
   505→              {loadInfo?.loadLevel ?? 'low'} load · {loadInfo?.requestsPerSecond ?? latest?.reqPerSec ?? 0} req/s
   506→            </p>
   507→          </CardContent>
   508→        </Card>
   509→      </div>
   510→
   511→      {/* Segmented Uptime Bar */}
   512→      <Card>
   513→        <CardHeader>
   514→          <CardTitle className="text-base">Uptime — Last 24 Hours</CardTitle>
   515→          <CardDescription>
   516→            Segmented view: green = up, red = down. Each segment represents ~48 minutes.
   517→          </CardDescription>
   518→        </CardHeader>
   519→        <CardContent>
   520→          <div className="flex gap-0.5 h-6 rounded-md overflow-hidden">
   521→            {uptimeSegments.map((seg, i) => (
   522→              <Tooltip key={i}>
   523→                <TooltipTrigger asChild>
   524→                  <div
   525→                    className={`flex-1 ${seg === 'up' ? 'bg-emerald-500' : 'bg-red-500'} hover:opacity-80 cursor-pointer transition-opacity`}
   526→                  />
   527→                </TooltipTrigger>
   528→                <TooltipContent>
   529→                  Segment #{i + 1}: {seg === 'up' ? 'Operational' : 'Down'}
   530→                </TooltipContent>
   531→              </Tooltip>
   532→            ))}
   533→          </div>
   534→          <div className="mt-2 flex items-center justify-between text-xs text-muted-foreground">
   535→            <span>24h ago</span>
   536→            <div className="flex items-center gap-3">
   537→              <span className="flex items-center gap-1">
   538→                <span className="h-2 w-2 rounded-full bg-emerald-500" /> Up

   541→                <span className="h-2 w-2 rounded-full bg-red-500" /> Down

   549→      <Tabs defaultValue="heartbeat" className="space-y-4">
