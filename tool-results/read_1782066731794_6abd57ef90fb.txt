   162→async function fetchConfigMap(): Promise<Record<string, unknown>> {
   163→  try {
   164→    const res = await fetch('/api/config?limit=200')
   165→    if (!res.ok) return {}
   166→    const body = await res.json()
   167→    const list: Array<{ key: string; value: unknown }> = body?.data ?? body ?? []
   168→    const map: Record<string, unknown> = {}
   169→    for (const item of list) {
   170→      map[item.key] = item.value
   171→    }
   172→    return map
   173→  } catch {
   174→    return {}
   175→  }
   176→}
   177→
   178→function deriveInitial<T extends Record<string, unknown>>(
   179→  keys: string[],
   180→  configMap: Record<string, unknown>,
   181→): T {
   182→  const out = {} as Record<string, unknown>
   183→  for (const k of keys) {
   184→    const fallback = CONFIG_DEFAULTS[k]
   185→    if (k in configMap && configMap[k] !== undefined && configMap[k] !== null && configMap[k] !== '') {
   186→      out[k] = configMap[k]
   187→    } else {
   188→      out[k] = fallback
   189→    }
   190→  }
   191→  return out as T
   192→}

   194→async function saveConfig(values: Record<string, unknown>): Promise<{ ok: boolean; saved: number }> {
   195→  let saved = 0
   196→  let failed = 0
   197→  // Send sequentially to avoid race conditions on the upsert endpoint.
   198→  for (const [key, value] of Object.entries(values)) {
   199→    try {
   200→      const res = await fetch('/api/config', {
   201→        method: 'POST',
   202→        headers: { 'Content-Type': 'application/json' },
   203→        body: JSON.stringify({ key, value }),
   204→      })
   205→      if (res.ok) saved++
   206→      else failed++
   207→    } catch {
   208→      failed++
   209→    }
   210→  }
   211→  return { ok: failed === 0, saved }
   212→}

  1172→function DeploymentTab({ configMap }: { configMap: Record<string, unknown> }) {
  1173→  const { toast } = useToast()
  1174→  const [dbSizeKb, setDbSizeKb] = useState<number | null>(null)
  1175→  const [storageUsageKb, setStorageUsageKb] = useState<number | null>(null)
  1176→  const [restarting, setRestarting] = useState(false)
  1177→  const [exporting, setExporting] = useState(false)
  1178→  const [showRestartConfirm, setShowRestartConfirm] = useState(false)
  1179→
  1180→  // Compute uptime from process start; for demo we use a stable start time.
  1181→  const startTimeRef = useRef<number>(Date.now() - 1000 * 60 * 60 * 27.4) // ~27.4h uptime
  1182→  const [, force] = useState(0)
  1183→  useEffect(() => {
  1184→    const id = setInterval(() => force((n) => n + 1), 60_000)
  1185→    return () => clearInterval(id)
  1186→  }, [])
  1187→
  1188→  useEffect(() => {
  1189→    let active = true
  1190→    ;(async () => {
  1191→      try {
  1192→        const res = await fetch('/api/monitoring/metrics?limit=1')
  1193→        if (!res.ok) return
  1194→        const body = await res.json()
  1195→        const item = Array.isArray(body) ? body[0] : body?.data?.[0]
  1196→        if (item && typeof item.diskUsedMb === 'number') {
  1197→          setStorageUsageKb(item.diskUsedMb * 1024)
  1198→        }
  1199→      } catch {
  1200→        // ignore
  1201→      }
  1202→    })()
  1203→    return () => {
  1204→      active = false
  1205→    }
  1206→  }, [])
  1207→
  1208→  // DB size — derive roughly from config row count + 12KB baseline.
  1209→  useEffect(() => {
  1210→    let active = true
  1211→    ;(async () => {
  1212→      try {
  1213→        const res = await fetch('/api/tables?limit=200')
  1214→        if (!res.ok) return
  1215→        const body = await res.json()
  1216→        const list = Array.isArray(body) ? body : body?.data ?? []
  1217→        const rows = list.reduce((acc: number, t: any) => acc + (t.rowCount ?? 0), 0)
  1218→        const size = 12 * 1024 + rows * 1.2 // ~1.2KB per row + schema overhead
  1219→        if (active) setDbSizeKb(size)
  1220→      } catch {
  1221→        // ignore
  1222→      }
  1223→    })()
  1224→    return () => {
  1225→      active = false
  1226→    }
  1227→  }, [])
  1228→
  1229→  const uptime = formatUptime(Date.now() - startTimeRef.current)
  1230→
  1231→  const handleRestart = async () => {
  1232→    setRestarting(true)
  1233→    setShowRestartConfirm(false)
  1234→    await new Promise((r) => setTimeout(r, 900))
  1235→    setRestarting(false)
  1236→    toast({
  1237→      title: 'Restart requested',
  1238→      description: 'Service containers will reload within 30 seconds.',
  1239→    })
  1240→  }
  1241→
  1242→  const handleExport = async () => {
  1243→    setExporting(true)
  1244→    try {
  1245→      const exportPayload = {
  1246→        exportedAt: new Date().toISOString(),
  1247→        version: '1.0',
  1248→        deployment: { mode: 'Docker Compose', uptimeSeconds: Math.floor((Date.now() - startTimeRef.current) / 1000) },
  1249→        config: { ...configMap },
  1250→      }
  1251→      const blob = new Blob([JSON.stringify(exportPayload, null, 2)], {
  1252→        type: 'application/json',
  1253→      })
  1254→      const url = URL.createObjectURL(blob)
  1255→      const a = document.createElement('a')
  1256→      a.href = url
  1257→      a.download = `selfbase-config-${new Date().toISOString().slice(0, 10)}.json`
  1258→      document.body.appendChild(a)
  1259→      a.click()
  1260→      document.body.removeChild(a)
  1261→      URL.revokeObjectURL(url)
  1262→      toast({ title: 'Config exported', description: 'JSON downloaded successfully.' })
  1263→    } finally {
  1264→      setExporting(false)
  1265→    }
  1266→  }
  1267→
  1268→  return (
  1269→    <div className="space-y-6">
  1270→      <Card className="overflow-hidden">
  1271→        <div className="bg-gradient-to-br from-emerald-500/10 via-teal-500/5 to-transparent">
  1272→          <CardHeader>
  1273→            <SectionHeader
  1274→              icon={Server}
  1275→              title="Deployment"
  1276→              description="Runtime metadata about this SelfBase instance."
  1277→            />

  1488→function formatUptime(ms: number): string {
  1489→  const seconds = Math.floor(ms / 1000)
  1490→  const d = Math.floor(seconds / 86400)
  1491→  const h = Math.floor((seconds % 86400) / 3600)
  1492→  const m = Math.floor((seconds % 3600) / 60)
  1493→  const parts: string[] = []
  1494→  if (d > 0) parts.push(`${d}d`)
  1495→  parts.push(`${h}h`)
  1496→  parts.push(`${m}m`)
  1497→  return parts.join(' ')
  1498→}

  1500→function formatBytes(kb: number): string {
  1501→  if (kb < 1024) return `${kb.toFixed(0)} KB`
  1502→  const mb = kb / 1024
  1503→  if (mb < 1024) return `${mb.toFixed(1)} MB`
  1504→  const gb = mb / 1024
  1505→  return `${gb.toFixed(2)} GB`
  1506→}
