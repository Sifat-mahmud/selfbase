   171→async function fetchConfigMap(): Promise<Record<string, unknown>> {
   172→  try {
   173→    const res = await fetch('/api/config?limit=200')
   174→    if (!res.ok) return {}
   175→    const body = await res.json()
   176→    const list: Array<{ key: string; value: unknown }> = body?.data ?? body ?? []
   177→    const map: Record<string, unknown> = {}
   178→    for (const item of list) {
   179→      map[item.key] = item.value
   180→    }
   181→    return map
   182→  } catch {
   183→    return {}
   184→  }
   185→}
   186→
   187→function deriveInitial<T extends Record<string, unknown>>(
   188→  keys: string[],
   189→  configMap: Record<string, unknown>,
   190→): T {
   191→  const out = {} as Record<string, unknown>
   192→  for (const k of keys) {
   193→    const fallback = CONFIG_DEFAULTS[k]
   194→    if (k in configMap && configMap[k] !== undefined && configMap[k] !== null && configMap[k] !== '') {
   195→      out[k] = configMap[k]
   196→    } else {
   197→      out[k] = fallback
   198→    }
   199→  }
   200→  return out as T
   201→}

   203→async function saveConfig(values: Record<string, unknown>): Promise<{ ok: boolean; saved: number }> {
   204→  let saved = 0
   205→  let failed = 0
   206→  // Send sequentially to avoid race conditions on the upsert endpoint.
   207→  for (const [key, value] of Object.entries(values)) {
   208→    try {
   209→      const res = await fetch('/api/config', {
   210→        method: 'POST',
   211→        headers: { 'Content-Type': 'application/json' },
   212→        body: JSON.stringify({ key, value }),
   213→      })
   214→      if (res.ok) saved++
   215→      else failed++
   216→    } catch {
   217→      failed++
   218→    }
   219→  }
   220→  return { ok: failed === 0, saved }
   221→}

   347→function ValidationIndicator({ value, keyName }: { value: unknown; keyName: string }) {
   348→  const defaults: Record<string, unknown> = CONFIG_DEFAULTS
   349→  const isInsecureDefault =
   350→    (keyName === 'security.corsOrigins' && value === '*') ||
   351→    (keyName === 'security.ipWhitelist' && (value === '' || value == null)) ||
   352→    (keyName === 'general.adminEmail' && value === 'admin@selfbase.local') ||
   353→    (keyName === 'storage.cdnBaseUrl' && value === 'https://cdn.selfbase.local')
   354→  const isMissing =
   355→    value === '' || value === undefined || value === null
   356→  const isDefault =
   357→    keyName in defaults && value === defaults[keyName]
   358→
   359→  if (isMissing) {
   360→    return (
   361→      <Tooltip>
   362→        <TooltipTrigger asChild>
   363→          <XCircle className="h-3.5 w-3.5 text-red-500 shrink-0" />
   364→        </TooltipTrigger>
   365→        <TooltipContent>Required field is missing</TooltipContent>
   366→      </Tooltip>
   367→    )
   368→  }
   369→  if (isInsecureDefault) {
   370→    return (
   371→      <Tooltip>
   372→        <TooltipTrigger asChild>
   373→          <AlertTriangle className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   374→        </TooltipTrigger>
   375→        <TooltipContent>Using default or insecure value</TooltipContent>
   376→      </Tooltip>
   377→    )
   378→  }
   379→  if (isDefault) {
   380→    return (
   381→      <Tooltip>
   382→        <TooltipTrigger asChild>
   383→          <AlertTriangle className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   384→        </TooltipTrigger>
   385→        <TooltipContent>Using default value</TooltipContent>
   386→      </Tooltip>
   387→    )
   388→  }
   389→  return (
   390→    <Tooltip>
   391→      <TooltipTrigger asChild>
   392→        <CheckCircle2 className="h-3.5 w-3.5 text-emerald-500 shrink-0" />
   393→      </TooltipTrigger>
   394→      <TooltipContent>Properly configured</TooltipContent>
   395→    </Tooltip>
   396→  )
   397→}

  1583→function formatUptime(ms: number): string {
  1584→  const seconds = Math.floor(ms / 1000)
  1585→  const d = Math.floor(seconds / 86400)
  1586→  const h = Math.floor((seconds % 86400) / 3600)
  1587→  const m = Math.floor((seconds % 3600) / 60)
  1588→  const parts: string[] = []
  1589→  if (d > 0) parts.push(`${d}d`)
  1590→  parts.push(`${h}h`)
  1591→  parts.push(`${m}m`)
  1592→  return parts.join(' ')
  1593→}
  1594→
  1595→function formatBytes(kb: number): string {
  1596→  if (kb < 1024) return `${kb.toFixed(0)} KB`
  1597→  const mb = kb / 1024
  1598→  if (mb < 1024) return `${mb.toFixed(1)} MB`
  1599→  const gb = mb / 1024
  1600→  return `${gb.toFixed(2)} GB`
  1601→}
