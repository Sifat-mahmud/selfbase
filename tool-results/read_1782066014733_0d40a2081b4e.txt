   151→/**
   152→ * Render a type-aware editor cell for inline editing.
   153→ * - BOOLEAN  -> Switch
   154→ * - INTEGER/DECIMAL -> number Input
   155→ * - JSON/TIMESTAMP -> wider Input
   156→ * - TEXT/default  -> Input
   157→ */
   158→function renderCellInput(
   159→  col: ColumnItem,
   160→  value: unknown,
   161→  onChange: (v: unknown) => void,
   162→  disabled?: boolean,
   163→) {
   164→  const typeUpper = col.type.toUpperCase()
   165→  if (typeUpper === 'BOOLEAN') {
   166→    return (
   167→      <Switch
   168→        checked={Boolean(value)}
   169→        onCheckedChange={(v) => onChange(v)}
   170→        disabled={disabled}
   171→        aria-label={col.name}
   172→      />
   173→    )
   174→  }
   175→  if (typeUpper === 'INTEGER' || typeUpper === 'DECIMAL') {
   176→    return (
   177→      <Input
   178→        type="number"
   179→        value={value === null || value === undefined ? '' : String(value)}
   180→        onChange={(e) => {
   181→          const raw = e.target.value
   182→          if (raw === '') {
   183→            onChange(null)
   184→          } else if (typeUpper === 'INTEGER') {
   185→            const n = parseInt(raw, 10)
   186→            onChange(Number.isNaN(n) ? null : n)
   187→          } else {
   188→            const n = parseFloat(raw)
   189→            onChange(Number.isNaN(n) ? null : n)
   190→          }
   191→        }}
   192→        disabled={disabled}
   193→        className="h-8 font-mono text-xs min-w-[80px]"
   194→        aria-label={col.name}
   195→      />
   196→    )
   197→  }
   198→  if (typeUpper === 'JSON' || typeUpper === 'TIMESTAMP') {
   199→    return (
   200→      <Input
   201→        value={value === null || value === undefined ? '' : String(value)}
   202→        onChange={(e) => onChange(e.target.value)}
   203→        disabled={disabled}
   204→        className="h-8 font-mono text-xs min-w-[140px]"
   205→        placeholder={typeUpper === 'JSON' ? '{...}' : '2024-01-01T00:00:00Z'}
   206→        aria-label={col.name}
   207→      />
   208→    )
   209→  }
   210→  return (
   211→    <Input
   212→      value={value === null || value === undefined ? '' : String(value)}
   213→      onChange={(e) => onChange(e.target.value)}
   214→      disabled={disabled}
   215→      className="h-8 font-mono text-xs min-w-[100px]"
   216→      aria-label={col.name}
   217→    />
   218→  )
   219→}

   553→        ids.map(async (id) => {

   557→          } catch {
   558→            failCount += 1

   563→      setSelectedRowIds(new Set())

   565→      if (failCount === 0) {
   566→        toast({
   567→          title: 'Rows deleted',
   568→          description: `${okCount} row${okCount !== 1 ? 's' : ''} removed`,
   569→        })
   570→      } else {

   572→          title: 'Bulk delete partial',
   573→          description: `${okCount} deleted, ${failCount} failed`,
   574→          variant: 'destructive',
   575→        })
   576→      }
   577→    } finally {
   578→      setBulkDeleting(false)
   579→      setBulkDeleteOpen(false)

   620→      }

   629→      let bv: unknown = undefined
   630→      try {
   631→        av = JSON.parse(a.data)[sortColumn]
   632→      } catch {
   633→        av = undefined
   634→      }
   635→      try {
   636→        bv = JSON.parse(b.data)[sortColumn]
   637→      } catch {
   638→        bv = undefined
   639→      }
   640→      if (av === null || av === undefined) return 1
   641→      if (bv === null || bv === undefined) return -1
   642→      if (typeof av === 'number' && typeof bv === 'number') {
   643→        return sortDirection === 'asc' ? av - bv : bv - av
   644→      }
   645→      const cmp = String(av).localeCompare(String(bv))
   646→      return sortDirection === 'asc' ? cmp : -cmp
   647→    })
   648→  }, [filteredRows, sortColumn, sortDirection])
