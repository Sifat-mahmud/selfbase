    73→interface ScrapeRunItem {
    74→  id: string
    75→  status: 'pending' | 'running' | 'success' | 'failed'
    76→  startedAt: string
    77→  completedAt: string | null
    78→  durationMs: number | null
    79→  pagesScraped: number
    80→  rowsExtracted: number
    81→  rowsWritten: number
    82→  errorPayload?: string | null
    83→  isPreview: boolean
    84→}
    85→
    86→interface ScraperSitemapItem {
    87→  id: string
    88→  name: string
    89→  description: string | null
    90→  startUrl: string
    91→  selectorTree: string
    92→  paginationType: 'none' | 'click' | 'scroll' | 'url_pattern'
    93→  paginationConfig?: string | null
    94→  targetTableId?: string | null
    95→  targetTable?: { name: string } | null
    96→  outputFormat: string
    97→  isActive: boolean
    98→  useStealth?: boolean
    99→  respectRobotsTxt?: boolean
   100→  fetchInterval: number
   101→  rateLimitMs: number
   102→  concurrency: number
   103→  maxPages: number
   104→  createdAt: string
   105→  scrapeRuns?: ScrapeRunItem[]
   106→}
   107→
   108→interface ScrapeRunsResponse {
   109→  data?: ScrapeRunItem[]
   110→  meta?: { page: number; limit: number; total: number }
   111→}
   112→
   113→interface PreviewResult {
   114→  id?: string
   115→  status?: string
   116→  message?: string
   117→  error?: string
   118→  rows?: Array<Record<string, unknown>>
   119→}

   151→export function ScraperView() {
   152→  const { toast } = useToast()
   153→  const [sitemaps, setSitemaps] = useState<ScraperSitemapItem[]>([])
   154→  const [allRuns, setAllRuns] = useState<ScrapeRunItem[]>([])
   155→  const [loading, setLoading] = useState(true)
   156→  const [search, setSearch] = useState('')
   157→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   158→  const [selectedSitemap, setSelectedSitemap] = useState<ScraperSitemapItem | null>(null)
   159→  const [previewTarget, setPreviewTarget] = useState<ScraperSitemapItem | null>(null)
   160→  const [previewResult, setPreviewResult] = useState<PreviewResult | null>(null)
   161→  const [previewLoading, setPreviewLoading] = useState(false)
   162→  const [runningId, setRunningId] = useState<string | null>(null)
   163→
   164→  // Create form
   165→  const [newName, setNewName] = useState('')
   166→  const [newDesc, setNewDesc] = useState('')
   167→  const [newUrl, setNewUrl] = useState('')
   168→  const [newPagination, setNewPagination] = useState<'none' | 'click' | 'scroll' | 'url_pattern'>('none')
   169→  const [newMaxPages, setNewMaxPages] = useState('50')
   170→  const [newRateLimit, setNewRateLimit] = useState('1000')
   171→  const [newConcurrency, setNewConcurrency] = useState('1')
   172→  const [newSelectorTree, setNewSelectorTree] = useState(JSON.stringify(defaultSelectorTree, null, 2))
   173→  const [newStealth, setNewStealth] = useState(false)
   174→  const [newTargetTable, setNewTargetTable] = useState('')
   175→
   176→  const loadAll = useCallback(async () => {
   177→    setLoading(true)
   178→    try {
   179→      const [s, r] = await Promise.all([
   180→        apiGet<ScraperSitemapItem[]>('/api/scrapers'),
   181→        apiGet<ScrapeRunsResponse>('/api/scrapers/runs?limit=100').catch(() => null),
   182→      ])
   183→      const list = Array.isArray(s) ? s : []
   184→      setSitemaps(
   185→        list.map((sm) => ({
   186→          ...sm,
   187→          targetTable: sm.targetTable ?? null,
   188→        })),
   189→      )
   190→      setAllRuns(Array.isArray(r?.data) ? r.data : [])
   191→    } catch (err) {
   192→      toast({
   193→        title: 'Failed to load scrapers',
   194→        description: err instanceof Error ? err.message : 'Unknown error',
   195→        variant: 'destructive',
   196→      })
   197→    } finally {
   198→      setLoading(false)
   199→    }
   200→  }, [toast])
   201→
   202→  useEffect(() => {
   203→    void loadAll()
   204→  }, [loadAll])
   205→
   206→  const handleCreate = async () => {
   207→    if (!newName.trim() || !newUrl.trim()) {
   208→      toast({ title: 'Name and URL required', variant: 'destructive' })
   209→      return
   210→    }
   211→    let parsedSelector: unknown
   212→    try {
   213→      parsedSelector = JSON.parse(newSelectorTree)
   214→    } catch {
   215→      toast({ title: 'Invalid selector tree JSON', variant: 'destructive' })
   216→      return
   217→    }
   218→    try {
   219→      const created = await apiPost<ScraperSitemapItem>('/api/scrapers', {
   220→        name: newName.trim(),
   221→        description: newDesc || null,
   222→        startUrl: newUrl.trim(),
   223→        selectorTree: JSON.stringify(parsedSelector),
   224→        paginationType: newPagination,
   225→        paginationConfig: null,
   226→        outputFormat: 'table',
   227→        isActive: true,
   228→        useStealth: newStealth,
   229→        respectRobotsTxt: true,
   230→        fetchInterval: 3600,
   231→        rateLimitMs: Number(newRateLimit) || 1000,
   232→        concurrency: Number(newConcurrency) || 1,
   233→        maxPages: Number(newMaxPages) || 50,
   234→        targetTableId: newTargetTable || null,
   235→      })
   236→      setSitemaps((prev) => [{ ...created, scrapeRuns: [] }, ...prev])
   237→      setShowCreateDialog(false)
   238→      setNewName('')
   239→      setNewDesc('')
   240→      setNewUrl('')
   241→      setNewPagination('none')
   242→      setNewMaxPages('50')
   243→      setNewRateLimit('1000')
   244→      setNewConcurrency('1')
   245→      setNewStealth(false)
   246→      setNewTargetTable('')
   247→      setNewSelectorTree(JSON.stringify(defaultSelectorTree, null, 2))
   248→      toast({ title: 'Sitemap created', description: `"${created.name}" has been created.` })
   249→    } catch (err) {
   250→      toast({
   251→        title: 'Failed to create sitemap',
   252→        description: err instanceof Error ? err.message : 'Unknown error',
   253→        variant: 'destructive',
   254→      })
   255→    }
   256→  }
   257→
   258→  const handleToggleStealth = async (sm: ScraperSitemapItem) => {
   259→    try {
   260→      await apiPut(`/api/scrapers/${sm.id}`, { useStealth: !sm.useStealth })
   261→      setSitemaps((prev) =>
   262→        prev.map((s) => (s.id === sm.id ? { ...s, useStealth: !s.useStealth } : s)),
   263→      )
   264→      if (selectedSitemap?.id === sm.id) {
   265→        setSelectedSitemap((prev) => (prev ? { ...prev, useStealth: !prev.useStealth } : prev))
   266→      }
   267→      toast({ title: `Stealth mode ${sm.useStealth ? 'disabled' : 'enabled'}` })
   268→    } catch (err) {
   269→      toast({
   270→        title: 'Failed to toggle stealth',
   271→        description: err instanceof Error ? err.message : 'Unknown error',
   272→        variant: 'destructive',
   273→      })
   274→    }
   275→  }
   276→
   277→  const handleRun = async (sm: ScraperSitemapItem) => {
   278→    setRunningId(sm.id)
   279→    try {
   280→      const result = await apiPost<{ id?: string; status?: string; message?: string; error?: string }>(
   281→        `/api/scrapers/${sm.id}/run`,
   282→        {},
   283→      )
   284→      toast({
   285→        title: 'Scrape triggered',
   286→        description: result.message ?? result.error ?? `Status: ${result.status ?? 'started'}`,
   287→      })
   288→      await loadAll()
   289→    } catch (err) {
   290→      toast({
   291→        title: 'Failed to trigger scrape',
   292→        description: err instanceof Error ? err.message : 'Unknown error',
   293→        variant: 'destructive',
   294→      })
   295→    } finally {
   296→      setRunningId(null)
   297→    }
   298→  }
   299→
   300→  const handlePreview = async (sm: ScraperSitemapItem) => {
   301→    setPreviewTarget(sm)
   302→    setPreviewResult(null)
   303→    setPreviewLoading(true)
   304→    try {
   305→      const result = await apiPost<PreviewResult>(`/api/scrapers/${sm.id}/preview`, {})
   306→      setPreviewResult(result)
   307→    } catch (err) {
   308→      setPreviewResult({
   309→        error: err instanceof Error ? err.message : 'Unknown error',
   310→      })
   311→    } finally {
   312→      setPreviewLoading(false)
   313→    }
   314→  }
   315→
   316→  const handleDelete = async (sm: ScraperSitemapItem) => {
   317→    try {
   318→      await apiDelete(`/api/scrapers/${sm.id}`)
   319→      setSitemaps((prev) => prev.filter((s) => s.id !== sm.id))
   320→      if (selectedSitemap?.id === sm.id) setSelectedSitemap(null)
   321→      toast({ title: 'Sitemap deleted', variant: 'destructive' })
   322→    } catch (err) {
   323→      toast({
   324→        title: 'Failed to delete sitemap',
   325→        description: err instanceof Error ? err.message : 'Unknown error',
   326→        variant: 'destructive',
   327→      })
   328→    }
   329→  }
   330→
   331→  const filteredSitemaps = sitemaps.filter(
   332→    (s) =>
   333→      s.name.toLowerCase().includes(search.toLowerCase()) ||
   334→      (s.description ?? '').toLowerCase().includes(search.toLowerCase()) ||
   335→      s.startUrl.toLowerCase().includes(search.toLowerCase()),
   336→  )
   337→
   338→  if (loading) {
   339→    return (
   340→      <div className="space-y-6">
   341→        <div className="flex justify-between">
   342→          <Skeleton className="h-8 w-48" />
   343→          <Skeleton className="h-9 w-36" />
   344→        </div>
   345→        {Array.from({ length: 3 }).map((_, i) => (
   346→          <Card key={i}>
   347→            <CardContent className="p-4">
   348→              <Skeleton className="h-32 w-full" />
   349→            </CardContent>
   350→          </Card>
   351→        ))}
   352→      </div>
   353→    )
   354→  }
   355→
   356→  if (selectedSitemap) {
   357→    const runs = selectedSitemap.scrapeRuns ?? allRuns.filter((r) => r.id === selectedSitemap.id)
   358→    const selectorTree = parseJsonField<unknown>(selectedSitemap.selectorTree, {})
   359→    return (
   360→      <motion.div
   361→        initial={{ opacity: 0, y: 8 }}
   362→        animate={{ opacity: 1, y: 0 }}
   363→        transition={{ duration: 0.2 }}
   364→        className="space-y-6"
   365→      >
   366→        <div className="flex items-center gap-2 flex-wrap">
   367→          <Button variant="ghost" size="sm" onClick={() => setSelectedSitemap(null)}>
   368→            <ArrowRight className="h-4 w-4 rotate-180" /> Back
   369→          </Button>
   370→          <Separator orientation="vertical" className="h-6" />
   371→          <div className="rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 p-2 text-white">
   372→            <Globe className="h-5 w-5" />
   373→          </div>
   374→          <h1 className="text-2xl font-bold tracking-tight">{selectedSitemap.name}</h1>
   375→          <Badge variant={selectedSitemap.isActive ? 'default' : 'secondary'}>
   376→            {selectedSitemap.isActive ? 'Active' : 'Inactive'}
   377→          </Badge>
   378→          <Badge variant="outline" className={`capitalize ${paginationBadgeColors[selectedSitemap.paginationType]}`}>
   379→            {selectedSitemap.paginationType}
   380→          </Badge>
   381→          <div className="ml-auto flex gap-2">
   382→            <Button variant="outline" size="sm" onClick={() => void handlePreview(selectedSitemap)}>
   383→              <Eye className="mr-1 h-3.5 w-3.5" /> Preview
   384→            </Button>
   385→            <Button
   386→              size="sm"
   387→              onClick={() => void handleRun(selectedSitemap)}
   388→              disabled={runningId === selectedSitemap.id}
   389→            >
   390→              {runningId === selectedSitemap.id ? (
   391→                <Loader2 className="mr-1 h-3.5 w-3.5 animate-spin" />
   392→              ) : (
   393→                <Play className="mr-1 h-3.5 w-3.5" />
   394→              )}
   395→              Run
   396→            </Button>
   397→          </div>
   398→        </div>
   399→
   400→        <div className="grid gap-4 md:grid-cols-4">
   401→          <Card>
   402→            <CardContent className="pt-4">
   403→              <div className="text-sm text-muted-foreground">Start URL</div>
   404→              <div className="text-sm font-mono truncate">{selectedSitemap.startUrl}</div>
   405→            </CardContent>
   406→          </Card>
   407→          <Card>
   408→            <CardContent className="pt-4">
   409→              <div className="text-sm text-muted-foreground">Pagination</div>
   410→              <div className="text-lg font-bold capitalize">{selectedSitemap.paginationType}</div>
   411→            </CardContent>
   412→          </Card>
   413→          <Card>
   414→            <CardContent className="pt-4">
   415→              <div className="text-sm text-muted-foreground">Max Pages</div>
   416→              <div className="text-lg font-bold">{selectedSitemap.maxPages}</div>
   417→            </CardContent>
   418→          </Card>
   419→          <Card>
   420→            <CardContent className="pt-4">
   421→              <div className="text-sm text-muted-foreground">Target Table</div>
   422→              <div className="text-lg font-bold font-mono">
   423→                {selectedSitemap.targetTable?.name ?? '—'}
   424→              </div>
   425→            </CardContent>
   426→          </Card>
   427→        </div>
   428→
   429→        <Card>
   430→          <CardHeader>
   431→            <CardTitle className="text-base">Selector Tree</CardTitle>
   432→            <CardDescription>CSS selectors for data extraction (JSON)</CardDescription>
   433→          </CardHeader>
   434→          <CardContent>
   435→            <div className="rounded-md bg-slate-950 text-emerald-400 p-4 overflow-auto max-h-[400px]">
   436→              <pre className="text-sm font-mono whitespace-pre-wrap">
   437→                {JSON.stringify(selectorTree, null, 2)}
   438→              </pre>
   439→            </div>
   440→            <Button variant="outline" size="sm" className="mt-3">
   441→              <Code className="mr-1 h-3.5 w-3.5" /> Edit Selector Tree
   442→            </Button>
   443→          </CardContent>
   444→        </Card>
   445→
   446→        <div className="grid gap-4 md:grid-cols-3">
   447→          <Card>
   448→            <CardHeader>
   449→              <CardTitle className="text-base">Rate Limiting</CardTitle>
   450→            </CardHeader>
   451→            <CardContent className="space-y-2">
   452→              <div className="flex justify-between text-sm">
   453→                <span className="text-muted-foreground">Delay</span>
   454→                <span className="font-mono">{selectedSitemap.rateLimitMs}ms</span>
   455→              </div>
   456→              <div className="flex justify-between text-sm">
   457→                <span className="text-muted-foreground">Concurrency</span>
   458→                <span>{selectedSitemap.concurrency}</span>
   459→              </div>
   460→              <div className="flex justify-between text-sm">
   461→                <span className="text-muted-foreground">Max Pages</span>
   462→                <span>{selectedSitemap.maxPages}</span>
   463→              </div>
   464→            </CardContent>
   465→          </Card>
   466→          <Card>
   467→            <CardHeader>
   468→              <CardTitle className="text-base">Output</CardTitle>
   469→            </CardHeader>
   470→            <CardContent className="space-y-2">
   471→              <div className="flex justify-between text-sm">
   472→                <span className="text-muted-foreground">Format</span>
   473→                <Badge variant="outline">{selectedSitemap.outputFormat}</Badge>
   474→              </div>
   475→              <div className="flex justify-between text-sm">
   476→                <span className="text-muted-foreground">Target</span>
   477→                <span className="font-mono">{selectedSitemap.targetTable?.name ?? '—'}</span>
   478→              </div>
   479→              <div className="flex justify-between text-sm items-center">
   480→                <span className="text-muted-foreground">Stealth mode</span>
   481→                <Switch
   482→                  checked={!!selectedSitemap.useStealth}
   483→                  onCheckedChange={() => void handleToggleStealth(selectedSitemap)}
   484→                />
   485→              </div>
   486→            </CardContent>
   487→          </Card>
   488→          <Card>
   489→            <CardHeader>
   490→              <CardTitle className="text-base">Actions</CardTitle>
   491→            </CardHeader>
   492→            <CardContent className="space-y-2">
   493→              <Button size="sm" className="w-full" onClick={() => void handleRun(selectedSitemap)} disabled={runningId === selectedSitemap.id}>
   494→                <Play className="mr-1 h-3.5 w-3.5" /> Run Scrape
   495→              </Button>
   496→              <Button variant="outline" size="sm" className="w-full" onClick={() => void handlePreview(selectedSitemap)}>
   497→                <Eye className="mr-1 h-3.5 w-3.5" /> Preview
   498→              </Button>
   499→            </CardContent>
   500→          </Card>
   501→        </div>
   502→
   503→        <Card>
   504→          <CardHeader>
   505→            <CardTitle className="text-base">Scrape Run History</CardTitle>
   506→            <CardDescription>Previous scrape executions ({runs.length} total)</CardDescription>
   507→          </CardHeader>
   508→          <CardContent>
   509→            {runs.length === 0 ? (
   510→              <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   511→                <FileText className="h-10 w-10 mb-2 opacity-40" />
   512→                <p className="text-sm">No runs yet</p>
   513→                <p className="text-xs mt-1">Click "Run" to trigger a scrape.</p>
   514→              </div>
   515→            ) : (
   516→              <div className="rounded-md border max-h-96 overflow-y-auto">
   517→                <Table>
   518→                  <TableHeader className="sticky top-0 bg-background">
   519→                    <TableRow>
   520→                      <TableHead>Status</TableHead>
   521→                      <TableHead>Started</TableHead>
   522→                      <TableHead>Duration</TableHead>
   523→                      <TableHead>Pages</TableHead>
   524→                      <TableHead>Extracted</TableHead>
   525→                      <TableHead>Written</TableHead>
   526→                      <TableHead>Type</TableHead>
   527→                    </TableRow>
   528→                  </TableHeader>
   529→                  <TableBody>
   530→                    {runs.slice(0, 30).map((run) => {
   531→                      const RunIcon = statusIcons[run.status]
   532→                      return (
   533→                        <TableRow key={run.id} className="hover:bg-muted/40">
   534→                          <TableCell>
   535→                            <div className="flex items-center gap-1.5">
   536→                              <RunIcon className={`h-3.5 w-3.5 ${statusColors[run.status]}`} />
   537→                              <span className={`text-sm capitalize font-medium ${statusColors[run.status]}`}>
   538→                                {run.status}
   539→                              </span>
   540→                            </div>
   541→                          </TableCell>
   542→                          <TableCell className="text-xs text-muted-foreground">
   543→                            {new Date(run.startedAt).toLocaleString()}
   544→                          </TableCell>
   545→                          <TableCell className="text-sm font-mono">
   546→                            {run.durationMs ? `${(run.durationMs / 1000).toFixed(1)}s` : '—'}
   547→                          </TableCell>
   548→                          <TableCell className="text-sm">{run.pagesScraped}</TableCell>
   549→                          <TableCell className="text-sm">{run.rowsExtracted.toLocaleString()}</TableCell>
   550→                          <TableCell className="text-sm">{run.rowsWritten.toLocaleString()}</TableCell>
   551→                          <TableCell>
   552→                            <Badge variant="outline" className="text-xs">
   553→                              {run.isPreview ? 'Preview' : 'Full'}
   554→                            </Badge>
   555→                          </TableCell>
   556→                        </TableRow>
   557→                      )
   558→                    })}
   559→                  </TableBody>
   560→                </Table>
   561→              </div>
   562→            )}
   563→          </CardContent>
