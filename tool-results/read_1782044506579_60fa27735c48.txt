   152→async function fetchConfigMap(): Promise<Record<string, unknown>> {
   153→  try {
   154→    const res = await fetch('/api/config?limit=200')
   155→    if (!res.ok) return {}
   156→    const body = await res.json()
   157→    const list: Array<{ key: string; value: unknown }> = body?.data ?? body ?? []
   158→    const map: Record<string, unknown> = {}
   159→    for (const item of list) {
   160→      map[item.key] = item.value
   161→    }
   162→    return map
   163→  } catch {
   164→    return {}
   165→  }
   166→}
   167→
   168→function deriveInitial<T extends Record<string, unknown>>(
   169→  keys: string[],
   170→  configMap: Record<string, unknown>,
   171→): T {
   172→  const out = {} as Record<string, unknown>
   173→  for (const k of keys) {
   174→    const fallback = CONFIG_DEFAULTS[k]
   175→    if (k in configMap && configMap[k] !== undefined && configMap[k] !== null && configMap[k] !== '') {
   176→      out[k] = configMap[k]
   177→    } else {
   178→      out[k] = fallback
   179→    }
   180→  }
   181→  return out as T
   182→}

   184→async function saveConfig(values: Record<string, unknown>): Promise<{ ok: boolean; saved: number }> {
   185→  let saved = 0
   186→  let failed = 0
   187→  // Send sequentially to avoid race conditions on the upsert endpoint.
   188→  for (const [key, value] of Object.entries(values)) {
   189→    try {
   190→      const res = await fetch('/api/config', {
   191→        method: 'POST',
   192→        headers: { 'Content-Type': 'application/json' },
   193→        body: JSON.stringify({ key, value }),
   194→      })
   195→      if (res.ok) saved++
   196→      else failed++
   197→    } catch {
   198→      failed++
   199→    }
   200→  }
   201→  return { ok: failed === 0, saved }
   202→}

  1163→  const { toast } = useToast()

  1165→  const [storageUsageKb, setStorageUsageKb] = useState<number | null>(null)
  1166→  const [restarting, setRestarting] = useState(false)
  1167→  const [exporting, setExporting] = useState(false)
  1168→
  1169→  // Compute uptime from process start; for demo we use a stable start time.
  1170→  const startTimeRef = useRef<number>(Date.now() - 1000 * 60 * 60 * 27.4) // ~27.4h uptime
  1171→  const [, force] = useState(0)
  1172→  useEffect(() => {
  1173→    const id = setInterval(() => force((n) => n + 1), 60_000)
  1174→    return () => clearInterval(id)
  1175→  }, [])
  1176→
  1177→  useEffect(() => {
  1178→    let active = true
  1179→    ;(async () => {
  1180→      try {
  1181→        const res = await fetch('/api/monitoring/metrics?limit=1')
  1182→        if (!res.ok) return
  1183→        const body = await res.json()
  1184→        const item = Array.isArray(body) ? body[0] : body?.data?.[0]
  1185→        if (item && typeof item.diskUsedMb === 'number') {
  1186→          setStorageUsageKb(item.diskUsedMb * 1024)
  1187→        }
  1188→      } catch {
  1189→        // ignore
  1190→      }
  1191→    })()
  1192→    return () => {
  1193→      active = false
  1194→    }
  1195→  }, [])
  1196→
  1197→  // DB size — derive roughly from config row count + 12KB baseline.
  1198→  useEffect(() => {
  1199→    let active = true
  1200→    ;(async () => {
  1201→      try {
  1202→        const res = await fetch('/api/tables?limit=200')
  1203→        if (!res.ok) return
  1204→        const body = await res.json()
  1205→        const list = Array.isArray(body) ? body : body?.data ?? []
  1206→        const rows = list.reduce((acc: number, t: any) => acc + (t.rowCount ?? 0), 0)
  1207→        const size = 12 * 1024 + rows * 1.2 // ~1.2KB per row + schema overhead
  1208→        if (active) setDbSizeKb(size)
  1209→      } catch {
  1210→        // ignore
  1211→      }
  1212→    })()
  1213→    return () => {
  1214→      active = false
  1215→    }
  1216→  }, [])
  1217→
  1218→  const uptime = formatUptime(Date.now() - startTimeRef.current)
  1219→
  1220→  const handleRestart = async () => {
  1221→    setRestarting(true)
  1222→    await new Promise((r) => setTimeout(r, 900))
  1223→    setRestarting(false)
  1224→    toast({
  1225→      title: 'Restart requested',
  1226→      description: 'Service containers will reload within 30 seconds.',
  1227→    })
  1228→  }
  1229→
  1230→  const handleExport = async () => {
  1231→    setExporting(true)
  1232→    try {
  1233→      const exportPayload = {
  1234→        exportedAt: new Date().toISOString(),
  1235→        version: '1.0',
  1236→        deployment: { mode: 'Docker Compose', uptimeSeconds: Math.floor((Date.now() - startTimeRef.current) / 1000) },
  1237→        config: { ...configMap },
  1238→      }
  1239→      const blob = new Blob([JSON.stringify(exportPayload, null, 2)], {
  1240→        type: 'application/json',
  1241→      })
  1242→      const url = URL.createObjectURL(blob)
  1243→      const a = document.createElement('a')
  1244→      a.href = url
  1245→      a.download = `selfbase-config-${new Date().toISOString().slice(0, 10)}.json`
  1246→      document.body.appendChild(a)
  1247→      a.click()
  1248→      document.body.removeChild(a)
  1249→      URL.revokeObjectURL(url)
  1250→      toast({ title: 'Config exported', description: 'JSON downloaded successfully.' })
  1251→    } finally {
  1252→      setExporting(false)
  1253→    }
  1254→  }
  1255→
  1256→  return (
  1257→    <div className="space-y-6">
  1258→      <Card className="overflow-hidden">
  1259→        <div className="bg-gradient-to-br from-emerald-500/10 via-teal-500/5 to-transparent">
  1260→          <CardHeader>
  1261→            <SectionHeader
  1262→              icon={Server}
  1263→              title="Deployment"
  1264→              description="Runtime metadata about this SelfBase instance."
  1265→            />
  1266→          </CardHeader>
  1267→          <CardContent>
  1268→            <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
  1269→              <InfoStat icon={Info} label="Version" value="v1.0" hint="SelfBase release" />
  1270→              <InfoStat
  1271→                icon={Server}
  1272→                label="Deployment Mode"
  1273→                value="Docker Compose"
  1274→                hint="Multi-container"
  1275→              />

  1456→function formatUptime(ms: number): string {
  1457→  const seconds = Math.floor(ms / 1000)
  1458→  const d = Math.floor(seconds / 86400)
  1459→  const h = Math.floor((seconds % 86400) / 3600)
  1460→  const m = Math.floor((seconds % 3600) / 60)
  1461→  const parts: string[] = []
  1462→  if (d > 0) parts.push(`${d}d`)
  1463→  parts.push(`${h}h`)
  1464→  parts.push(`${m}m`)
  1465→  return parts.join(' ')
  1466→}
  1467→
  1468→function formatBytes(kb: number): string {
  1469→  if (kb < 1024) return `${kb.toFixed(0)} KB`
  1470→  const mb = kb / 1024
  1471→  if (mb < 1024) return `${mb.toFixed(1)} MB`
  1472→  const gb = mb / 1024
  1473→  return `${gb.toFixed(2)} GB`
  1474→}
