   148→   148→/**
   149→   149→ * Render a type-aware editor cell for inline editing.
   150→   150→ * - BOOLEAN  -> Switch
   151→   151→ * - INTEGER/DECIMAL -> number Input
   152→   152→ * - JSON/TIMESTAMP -> wider Input
   153→   153→ * - TEXT/default  -> Input
   154→   154→ */
   155→   155→function renderCellInput(
   156→   156→  col: ColumnItem,
   157→   157→  value: unknown,
   158→   158→  onChange: (v: unknown) => void,
   159→   159→  disabled?: boolean,
   160→   160→) {
   161→   161→  const typeUpper = col.type.toUpperCase()
   162→   162→  if (typeUpper === 'BOOLEAN') {
   163→   163→    return (
   164→   164→      <Switch
   165→   165→        checked={Boolean(value)}
   166→   166→        onCheckedChange={(v) => onChange(v)}
   167→   167→        disabled={disabled}
   168→   168→        aria-label={col.name}
   169→   169→      />
   170→   170→    )
   171→   171→  }
   172→   172→  if (typeUpper === 'INTEGER' || typeUpper === 'DECIMAL') {
   173→   173→    return (
   174→   174→      <Input
   175→   175→        type="number"
   176→   176→        value={value === null || value === undefined ? '' : String(value)}
   177→   177→        onChange={(e) => {
   178→   178→          const raw = e.target.value
   179→   179→          if (raw === '') {
   180→   180→            onChange(null)
   181→   181→          } else if (typeUpper === 'INTEGER') {
   182→   182→            const n = parseInt(raw, 10)
   183→   183→            onChange(Number.isNaN(n) ? null : n)
   184→   184→          } else {
   185→   185→            const n = parseFloat(raw)
   186→   186→            onChange(Number.isNaN(n) ? null : n)
   187→   187→          }
   188→   188→        }}
   189→   189→        disabled={disabled}
   190→   190→        className="h-8 font-mono text-xs min-w-[80px]"
   191→   191→        aria-label={col.name}
   192→   192→      />
   193→   193→    )
   194→   194→  }
   195→   195→  if (typeUpper === 'JSON' || typeUpper === 'TIMESTAMP') {
   196→   196→    return (
   197→   197→      <Input
   198→   198→        value={value === null || value === undefined ? '' : String(value)}
   199→   199→        onChange={(e) => onChange(e.target.value)}
   200→   200→        disabled={disabled}
   201→   201→        className="h-8 font-mono text-xs min-w-[140px]"
   202→   202→        placeholder={typeUpper === 'JSON' ? '{...}' : '2024-01-01T00:00:00Z'}
   203→   203→        aria-label={col.name}
   204→   204→      />
   205→   205→    )
   206→   206→  }
   207→   207→  return (
   208→   208→    <Input
   209→   209→      value={value === null || value === undefined ? '' : String(value)}
   210→   210→      onChange={(e) => onChange(e.target.value)}
   211→   211→      disabled={disabled}
   212→   212→      className="h-8 font-mono text-xs min-w-[100px]"
   213→   213→      aria-label={col.name}
   214→   214→    />
   215→   215→  )
   216→   216→}

   547→   547→            failCount += 1
   548→   548→          }

   552→   552→      setSelectedRowIds(new Set())

   554→   554→      if (failCount === 0) {
   555→   555→        toast({
   556→   556→          title: 'Rows deleted',
   557→   557→          description: `${okCount} row${okCount !== 1 ? 's' : ''} removed`,
   558→   558→        })

   560→   560→        toast({
   561→   561→          title: 'Bulk delete partial',
   562→   562→          description: `${okCount} deleted, ${failCount} failed`,
   563→   563→          variant: 'destructive',
   564→   564→        })
   565→   565→      }
   566→   566→    } finally {
   567→   567→      setBulkDeleting(false)
   568→   568→      setBulkDeleteOpen(false)
   569→   569→    }
