   134→const emptyTrend = (rowCount: number) => {
   135→  // Demo trend derived from row count for visual interest.
   136→  // Real trend would require an API endpoint.
   137→  if (rowCount === 0) return { delta: 0, direction: 'none' as const }
   138→  const delta = Math.round((rowCount % 13) - 5)
   139→  return { delta: Math.abs(delta), direction: delta >= 0 ? ('up' as const) : ('down' as const) }
   140→}

   142→export function TablesView() {
   143→  const { toast } = useToast()
   144→  const [tables, setTables] = useState<SbTableItem[]>([])
   145→  const [loading, setLoading] = useState(true)
   146→  const [search, setSearch] = useState('')
   147→  const [selectedTable, setSelectedTable] = useState<SbTableItem | null>(null)
   148→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   149→  const [showColumnDialog, setShowColumnDialog] = useState(false)
   150→  const [deleteTarget, setDeleteTarget] = useState<SbTableItem | null>(null)
   151→  const [showDataDialog, setShowDataDialog] = useState(false)
   152→  const [dataRows, setDataRows] = useState<SbRowItem[]>([])
   153→  const [dataLoading, setDataLoading] = useState(false)
   154→
   155→  // New table form
   156→  const [newTableName, setNewTableName] = useState('')
   157→  const [newTableDesc, setNewTableDesc] = useState('')
   158→  const [newTablePriority, setNewTablePriority] = useState(2)
   159→  const [newColumns, setNewColumns] = useState<ColumnItem[]>([
   160→    { id: 'c-init', name: 'id', type: 'TEXT', nullable: false, isPrimaryKey: true, isUnique: true, isIndexed: true, defaultValue: null, order: 0 },
   161→  ])
   162→
   163→  // New column form
   164→  const [newColumnName, setNewColumnName] = useState('')
   165→  const [newColumnType, setNewColumnType] = useState('TEXT')
   166→
   167→  const [sortField, setSortField] = useState<'name' | 'rowCount' | 'priority'>('name')
   168→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
   169→
   170→  const loadTables = useCallback(async () => {
   171→    setLoading(true)
   172→    try {
   173→      const data = await apiGet<SbTableItem[]>('/api/tables')
   174→      setTables(Array.isArray(data) ? data : [])
   175→    } catch (err) {
   176→      toast({
   177→        title: 'Failed to load tables',
   178→        description: err instanceof Error ? err.message : 'Unknown error',
   179→        variant: 'destructive',
   180→      })
   181→    } finally {
   182→      setLoading(false)
   183→    }
   184→  }, [toast])
   185→
   186→  useEffect(() => {
   187→    loadTables()
   188→  }, [loadTables])
   189→
   190→  const filteredTables = tables
   191→    .filter(
   192→      (t) =>
   193→        t.name.toLowerCase().includes(search.toLowerCase()) ||
   194→        (t.displayName ?? '').toLowerCase().includes(search.toLowerCase()),
   195→    )
   196→    .sort((a, b) => {
   197→      const dir = sortDir === 'asc' ? 1 : -1
   198→      if (sortField === 'name') return a.name.localeCompare(b.name) * dir
   199→      if (sortField === 'rowCount') return (a.rowCount - b.rowCount) * dir
   200→      return (a.priority - b.priority) * dir
   201→    })
   202→
   203→  const handleSort = (field: 'name' | 'rowCount' | 'priority') => {
   204→    if (sortField === field) setSortDir((d) => (d === 'asc' ? 'desc' : 'asc'))
   205→    else {
   206→      setSortField(field)
   207→      setSortDir('asc')
   208→    }
   209→  }
   210→
   211→  const handleCreateTable = async () => {
   212→    if (!newTableName.trim()) {
   213→      toast({ title: 'Name required', variant: 'destructive' })
   214→      return
   215→    }
   216→    try {
   217→      const created = await apiPost<SbTableItem>('/api/tables', {
   218→        name: newTableName.trim().toLowerCase().replace(/\s+/g, '_'),
   219→        displayName: newTableName.trim(),
   220→        description: newTableDesc || null,
   221→        priority: newTablePriority,
   222→      })
   223→      setTables((prev) => [created, ...prev])
   224→      setShowCreateDialog(false)
   225→      setNewTableName('')
   226→      setNewTableDesc('')
   227→      setNewTablePriority(2)
   228→      setNewColumns([
   229→        { id: 'c-init', name: 'id', type: 'TEXT', nullable: false, isPrimaryKey: true, isUnique: true, isIndexed: true, defaultValue: null, order: 0 },
   230→      ])
   231→      toast({
   232→        title: 'Table created',
   233→        description: `"${created.name}" has been created successfully.`,
   234→      })
   235→    } catch (err) {
   236→      toast({
   237→        title: 'Failed to create table',
   238→        description: err instanceof Error ? err.message : 'Unknown error',
   239→        variant: 'destructive',
   240→      })
   241→    }
   242→  }
   243→
   244→  const handleDeleteTable = async (table: SbTableItem) => {
   245→    try {
   246→      await apiDelete(`/api/tables/${table.id}`)
   247→      setTables((prev) => prev.filter((t) => t.id !== table.id))
   248→      if (selectedTable?.id === table.id) setSelectedTable(null)
   249→      toast({
   250→        title: 'Table deleted',
   251→        description: `"${table.name}" has been removed.`,
   252→        variant: 'destructive',
   253→      })
   254→    } catch (err) {
   255→      toast({
   256→        title: 'Failed to delete table',
   257→        description: err instanceof Error ? err.message : 'Unknown error',
   258→        variant: 'destructive',
   259→      })
   260→    } finally {
   261→      setDeleteTarget(null)
   262→    }
   263→  }
   264→
   265→  const [addColumnLoading, setAddColumnLoading] = useState(false)
   266→
   267→  const handleAddColumn = async () => {
   268→    if (!selectedTable || !newColumnName.trim()) return
   269→    setAddColumnLoading(true)
   270→    try {
   271→      const created = await apiPost<ColumnItem>(`/api/tables/${selectedTable.id}/columns`, {
   272→        name: newColumnName.trim().toLowerCase().replace(/\s+/g, '_'),
   273→        type: newColumnType,
   274→        nullable: true,
   275→        isPrimaryKey: false,
   276→        isUnique: false,
   277→        isIndexed: false,
   278→        defaultValue: null,
   279→        order: selectedTable.columns.length,
   280→      })
   281→      const updatedCol = { ...created, id: created.id || `c${Date.now()}` }
   282→      setTables((prev) =>
   283→        prev.map((t) => (t.id === selectedTable.id ? { ...t, columns: [...t.columns, updatedCol] } : t)),
   284→      )
   285→      setSelectedTable((prev) => (prev ? { ...prev, columns: [...prev.columns, updatedCol] } : prev))
   286→      setShowColumnDialog(false)
   287→      setNewColumnName('')
   288→      setNewColumnType('TEXT')
   289→      toast({ title: 'Column added', description: `"${updatedCol.name}" added to ${selectedTable.name}` })
   290→    } catch (err) {
   291→      toast({
   292→        title: 'Failed to add column',
   293→        description: err instanceof Error ? err.message : 'Unknown error',
   294→        variant: 'destructive',
   295→      })
   296→    } finally {
   297→      setAddColumnLoading(false)
   298→    }
   299→  }
   300→
   301→  const handleViewData = async (table: SbTableItem) => {
   302→    setShowDataDialog(true)
   303→    setDataLoading(true)
   304→    try {
   305→      const rows = await apiGet<SbRowItem[]>(`/api/tables/${table.id}/rows`)
   306→      setDataRows(Array.isArray(rows) ? rows : [])
   307→    } catch (err) {
   308→      toast({
   309→        title: 'Failed to load rows',
   310→        description: err instanceof Error ? err.message : 'Unknown error',
   311→        variant: 'destructive',
   312→      })
   313→      setDataRows([])
   314→    } finally {
   315→      setDataLoading(false)
   316→    }
   317→  }
   318→
   319→  if (loading) {
   320→    return (
   321→      <div className="space-y-6">
   322→        <div className="flex justify-between">
   323→          <Skeleton className="h-8 w-40" />
   324→          <Skeleton className="h-9 w-32" />
   325→        </div>
   326→        <div className="grid gap-4 md:grid-cols-3">
   327→          {Array.from({ length: 6 }).map((_, i) => (
   328→            <Card key={i}>
   329→              <CardContent className="p-4">
   330→                <Skeleton className="h-20 w-full" />
   331→              </CardContent>
   332→            </Card>
   333→          ))}
   334→        </div>
   335→      </div>
   336→    )
   337→  }
   338→
   339→  // Detail view
   340→  if (selectedTable) {
   341→    return (
   342→      <motion.div
   343→        initial={{ opacity: 0, y: 8 }}
   344→        animate={{ opacity: 1, y: 0 }}
   345→        transition={{ duration: 0.2 }}
   346→        className="space-y-6"
   347→      >
   348→        <div className="flex items-center gap-2 flex-wrap">
   349→          <Button variant="ghost" size="sm" onClick={() => setSelectedTable(null)}>
   350→            <ChevronRight className="h-4 w-4 rotate-180" />
   351→            Back
   352→          </Button>
   353→          <Separator orientation="vertical" className="h-6" />
   354→          <div className="rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 p-2 text-white">
   355→            <Database className="h-5 w-5" />
   356→          </div>
   357→          <h1 className="text-2xl font-bold tracking-tight">
   358→            {selectedTable.displayName || selectedTable.name}
   359→          </h1>
   360→          <Badge variant="outline" className="ml-2 font-mono text-xs">
   361→            {selectedTable.name}
   362→          </Badge>
   363→          {selectedTable.isSystem && (
   364→            <Badge variant="secondary" className="gap-1">
   365→              <Lock className="h-3 w-3" /> System
   366→            </Badge>
   367→          )}
   368→          <Badge variant="outline" className={priorityColors[selectedTable.priority]}>
   369→            {priorityLabels[selectedTable.priority]}
   370→          </Badge>
   371→          <div className="ml-auto flex gap-2">
   372→            <Button variant="outline" size="sm" onClick={() => handleViewData(selectedTable)}>
   373→              <Eye className="mr-1 h-3.5 w-3.5" /> View Data
   374→            </Button>
   375→            <Button
   376→              variant="outline"
   377→              size="sm"
   378→              onClick={() => {
   379→                void loadTables()
   380→                const fresh = tables.find((t) => t.id === selectedTable.id)
   381→                if (fresh) setSelectedTable(fresh)
   382→              }}
   383→            >
   384→              <RefreshCw className="mr-1 h-3.5 w-3.5" /> Refresh
   385→            </Button>
   386→            <Button
   387→              variant="outline"
   388→              size="sm"
   389→              className="text-destructive hover:bg-destructive/10"
   390→              onClick={() => setDeleteTarget(selectedTable)}
   391→            >
   392→              <Trash2 className="mr-1 h-3.5 w-3.5" /> Delete Table
   393→            </Button>
   394→          </div>
   395→        </div>
   396→
   397→        <div className="grid gap-4 md:grid-cols-4">
   398→          <Card>
   399→            <CardContent className="pt-4">
   400→              <div className="text-sm text-muted-foreground">Rows</div>
   401→              <div className="text-2xl font-bold">{selectedTable.rowCount.toLocaleString()}</div>
   402→            </CardContent>
   403→          </Card>
   404→          <Card>
   405→            <CardContent className="pt-4">
   406→              <div className="text-sm text-muted-foreground">Columns</div>
   407→              <div className="text-2xl font-bold">{selectedTable.columns.length}</div>
   408→            </CardContent>
   409→          </Card>
   410→          <Card>
   411→            <CardContent className="pt-4">
   412→              <div className="text-sm text-muted-foreground">Version</div>
   413→              <div className="text-lg font-bold font-mono">{selectedTable.versionHash}</div>
   414→            </CardContent>
   415→          </Card>
   416→          <Card>
   417→            <CardContent className="pt-4">
   418→              <div className="text-sm text-muted-foreground">Priority</div>
   419→              <div className="text-lg font-bold">{priorityLabels[selectedTable.priority]}</div>
   420→            </CardContent>
   421→          </Card>
   422→        </div>
   423→
   424→        <Card>
   425→          <CardHeader>
   426→            <CardTitle className="text-base bg-gradient-to-r from-emerald-600 to-teal-600 bg-clip-text text-transparent">
   427→              Features
   428→            </CardTitle>
   429→            <CardDescription>Table-level feature flags</CardDescription>
   430→          </CardHeader>
   431→          <CardContent>
   432→            <div className="flex flex-wrap gap-6">
   433→              <div className="flex items-center gap-2">
   434→                <Switch checked={selectedTable.enableRealtime} />
   435→                <Label className="flex items-center gap-1">
   436→                  <Radio className="h-3.5 w-3.5 text-emerald-600" /> Realtime
   437→                </Label>
   438→              </div>
   439→              <div className="flex items-center gap-2">
   440→                <Switch checked={selectedTable.enableEmbedding} />
   441→                <Label className="flex items-center gap-1">
   442→                  <Brain className="h-3.5 w-3.5 text-emerald-600" /> Embeddings
   443→                </Label>
   444→              </div>
   445→              <div className="flex items-center gap-2">
   446→                <Switch checked={selectedTable.rlsEnabled} />
   447→                <Label className="flex items-center gap-1">
   448→                  <Shield className="h-3.5 w-3.5 text-emerald-600" /> Row-Level Security
   449→                </Label>
   450→              </div>
   451→            </div>
   452→          </CardContent>
   453→        </Card>
   454→
   455→        <Card>
   456→          <CardHeader className="flex flex-row items-center justify-between">
   457→            <div>
   458→              <CardTitle className="text-base">Columns</CardTitle>
   459→              <CardDescription>Schema definition for this table</CardDescription>
   460→            </div>
   461→            <Dialog open={showColumnDialog} onOpenChange={setShowColumnDialog}>
   462→              <DialogTrigger asChild>
   463→                <Button size="sm">
   464→                  <Plus className="mr-1 h-4 w-4" /> Add Column
   465→                </Button>
   466→              </DialogTrigger>
   467→              <DialogContent>
   468→                <DialogHeader>
   469→                  <DialogTitle>Add Column</DialogTitle>
   470→                  <DialogDescription>Add a new column to {selectedTable.name}</DialogDescription>
   471→                </DialogHeader>
   472→                <div className="space-y-4 py-2">
   473→                  <div className="space-y-2">
   474→                    <Label>Column Name</Label>
   475→                    <Input
   476→                      value={newColumnName}
   477→                      onChange={(e) => setNewColumnName(e.target.value)}
   478→                      placeholder="column_name"
   479→                    />
   480→                  </div>
   481→                  <div className="space-y-2">
   482→                    <Label>Type</Label>
   483→                    <Select value={newColumnType} onValueChange={setNewColumnType}>
   484→                      <SelectTrigger>
   485→                        <SelectValue />
   486→                      </SelectTrigger>
   487→                      <SelectContent>
   488→                        {columnTypes.map((t) => (
   489→                          <SelectItem key={t} value={t}>
   490→                            {t}
   491→                          </SelectItem>
   492→                        ))}
   493→                      </SelectContent>
   494→                    </Select>
   495→                  </div>
   496→                </div>
   497→                <DialogFooter>
   498→                  <Button variant="outline" onClick={() => setShowColumnDialog(false)}>
   499→                    Cancel
   500→                  </Button>
   501→                  <Button onClick={() => void handleAddColumn()} disabled={addColumnLoading}>Add Column</Button>
   502→                </DialogFooter>
   503→              </DialogContent>
   504→            </Dialog>
   505→          </CardHeader>
   506→          <CardContent>
   507→            {selectedTable.columns.length === 0 ? (
   508→              <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   509→                <Columns3 className="h-10 w-10 mb-2 opacity-40" />
   510→                <p className="text-sm">No columns defined yet</p>
   511→                <Button variant="outline" size="sm" className="mt-2" onClick={() => setShowColumnDialog(true)}>
   512→                  <Plus className="mr-1 h-3.5 w-3.5" /> Add first column
   513→                </Button>
   514→              </div>
   515→            ) : (
   516→              <Table>
   517→                <TableHeader>
   518→                  <TableRow>
   519→                    <TableHead className="w-8">#</TableHead>
   520→                    <TableHead>Name</TableHead>
   521→                    <TableHead>Type</TableHead>
   522→                    <TableHead>Nullable</TableHead>
   523→                    <TableHead>Primary Key</TableHead>
   524→                    <TableHead>Unique</TableHead>
   525→                    <TableHead>Indexed</TableHead>
   526→                    <TableHead>Default</TableHead>
   527→                    <TableHead className="w-8" />
   528→                  </TableRow>
   529→                </TableHeader>
   530→                <TableBody>
   531→                  {selectedTable.columns.map((col, i) => (
   532→                    <TableRow key={col.id} className="hover:bg-muted/40 transition-colors">
   533→                      <TableCell className="text-muted-foreground text-xs">{i + 1}</TableCell>
   534→                      <TableCell className="font-mono font-medium">{col.name}</TableCell>
   535→                      <TableCell>
   536→                        <Badge variant="outline" className="font-mono text-xs">
   537→                          {col.type}
   538→                        </Badge>
   539→                      </TableCell>
   540→                      <TableCell>
   541→                        {col.nullable ? (
   542→                          <span className="text-emerald-500">✓</span>
   543→                        ) : (
   544→                          <span className="text-muted-foreground">—</span>
   545→                        )}
   546→                      </TableCell>
   547→                      <TableCell>{col.isPrimaryKey ? <span>🔑</span> : <span className="text-muted-foreground">—</span>}</TableCell>
   548→                      <TableCell>
   549→                        {col.isUnique ? (
   550→                          <span className="text-emerald-500">✓</span>
   551→                        ) : (
   552→                          <span className="text-muted-foreground">—</span>
   553→                        )}
   554→                      </TableCell>
   555→                      <TableCell>
   556→                        {col.isIndexed ? (
   557→                          <span className="text-emerald-500">✓</span>
   558→                        ) : (
   559→                          <span className="text-muted-foreground">—</span>
   560→                        )}
   561→                      </TableCell>
   562→                      <TableCell className="font-mono text-xs text-muted-foreground">
   563→                        {col.defaultValue ?? '—'}
   564→                      </TableCell>
   565→                      <TableCell>
   566→                        <DropdownMenu>
   567→                          <DropdownMenuTrigger asChild>
   568→                            <Button variant="ghost" size="icon" className="h-7 w-7">
   569→                              <MoreHorizontal className="h-3.5 w-3.5" />
   570→                            </Button>
   571→                          </DropdownMenuTrigger>
   572→                          <DropdownMenuContent align="end">
   573→                            <DropdownMenuItem>
   574→                              <Pencil className="mr-2 h-3.5 w-3.5" /> Edit
   575→                            </DropdownMenuItem>
