   162→   162→async function fetchConfigMap(): Promise<Record<string, unknown>> {
   163→   163→  try {
   164→   164→    const res = await fetch('/api/config?limit=200')
   165→   165→    if (!res.ok) return {}
   166→   166→    const body = await res.json()
   167→   167→    const list: Array<{ key: string; value: unknown }> = body?.data ?? body ?? []
   168→   168→    const map: Record<string, unknown> = {}
   169→   169→    for (const item of list) {
   170→   170→      map[item.key] = item.value
   171→   171→    }
   172→   172→    return map
   173→   173→  } catch {
   174→   174→    return {}
   175→   175→  }
   176→   176→}

   178→   178→function deriveInitial<T extends Record<string, unknown>>(
   179→   179→  keys: string[],
   180→   180→  configMap: Record<string, unknown>,
   181→   181→): T {
   182→   182→  const out = {} as Record<string, unknown>
   183→   183→  for (const k of keys) {
   184→   184→    const fallback = CONFIG_DEFAULTS[k]
   185→   185→    if (k in configMap && configMap[k] !== undefined && configMap[k] !== null && configMap[k] !== '') {
   186→   186→      out[k] = configMap[k]
   187→   187→    } else {
   188→   188→      out[k] = fallback
   189→   189→    }
   190→   190→  }
   191→   191→  return out as T
   192→   192→}

   194→   194→async function saveConfig(values: Record<string, unknown>): Promise<{ ok: boolean; saved: number }> {
   195→   195→  let saved = 0
   196→   196→  let failed = 0
   197→   197→  // Send sequentially to avoid race conditions on the upsert endpoint.
   198→   198→  for (const [key, value] of Object.entries(values)) {
   199→   199→    try {
   200→   200→      const res = await fetch('/api/config', {
   201→   201→        method: 'POST',
   202→   202→        headers: { 'Content-Type': 'application/json' },
   203→   203→        body: JSON.stringify({ key, value }),
   204→   204→      })
   205→   205→      if (res.ok) saved++
   206→   206→      else failed++
   207→   207→    } catch {
   208→   208→      failed++
   209→   209→    }
   210→   210→  }
   211→   211→  return { ok: failed === 0, saved }
   212→   212→}

  1488→  1488→function formatUptime(ms: number): string {
  1489→  1489→  const seconds = Math.floor(ms / 1000)
  1490→  1490→  const d = Math.floor(seconds / 86400)
  1491→  1491→  const h = Math.floor((seconds % 86400) / 3600)
  1492→  1492→  const m = Math.floor((seconds % 3600) / 60)
  1493→  1493→  const parts: string[] = []
  1494→  1494→  if (d > 0) parts.push(`${d}d`)
