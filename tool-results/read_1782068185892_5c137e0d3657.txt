   176→/** Unwrap `{ success, data }` envelope; fall back to raw payload. */
   177→async function fetchJson<T>(url: string): Promise<T | null> {
   178→  try {
   179→    const res = await fetch(url, { cache: 'no-store' })
   180→    if (!res.ok) return null
   181→    const json = await res.json()
   182→    if (json && typeof json === 'object' && 'success' in json && 'data' in json) {
   183→      return (json as { data: T }).data as T
   184→    }
   185→    return json as T
   186→  } catch {
   187→    return null
   188→  }
   189→}

   191→/** Normalize an API response that may be wrapped or have pagination into an array. */
   192→function normalizeArray<T>(raw: T | null): T[] {
   193→  if (!raw) return []
   194→  if (Array.isArray(raw)) return raw
   195→  if (typeof raw === 'object' && raw !== null) {
   196→    const obj = raw as Record<string, unknown>
   197→    // Handle { data: [...] } envelope
   198→    if (Array.isArray(obj.data)) return obj.data as T[]
   199→    // Handle { runs: [...] } envelope (functions/runs)
   200→    if (Array.isArray(obj.runs)) return obj.runs as T[]
   201→  }
   202→  return []
   203→}

   205→function formatNumber(n: number, mode: 'compact' | 'full' = 'compact'): string {
   206→  if (!Number.isFinite(n)) return '0'
   207→  if (mode === 'full') return n.toLocaleString()
   208→  if (Math.abs(n) >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`
   209→  if (Math.abs(n) >= 1_000) return `${(n / 1_000).toFixed(1)}K`
   210→  return n.toLocaleString()
   211→}

   213→function formatRelativeTime(dateStr: string): string {
   214→  const now = Date.now()
   215→  const then = new Date(dateStr).getTime()
   216→  const diff = Math.max(0, now - then)
   217→  const sec = Math.floor(diff / 1000)
   218→  if (sec < 60) return `${sec}s ago`
   219→  const min = Math.floor(sec / 60)
   220→  if (min < 60) return `${min}m ago`
   221→  const hr = Math.floor(min / 60)
   222→  if (hr < 24) return `${hr}h ago`
   223→  const day = Math.floor(hr / 24)
   224→  if (day < 30) return `${day}d ago`
   225→  const mo = Math.floor(day / 30)
   226→  if (mo < 12) return `${mo}mo ago`
   227→  return `${Math.floor(mo / 12)}y ago`
   228→}

   230→function formatDuration(ms: number): string {
   231→  if (!Number.isFinite(ms) || ms <= 0) return '0m'
   232→  const sec = Math.floor(ms / 1000)
   233→  const days = Math.floor(sec / 86400)
   234→  const hours = Math.floor((sec % 86400) / 3600)
   235→  const mins = Math.floor((sec % 3600) / 60)
   236→  if (days > 0) return `${days}d ${hours}h ${mins}m`
   237→  if (hours > 0) return `${hours}h ${mins}m`
   238→  return `${mins}m`
   239→}

   270→function AnimatedNumber({
   271→  value,
   272→  format = 'plain',
   273→  duration = 800,
   274→  className,
   275→}: {
   276→  value: number
   277→  format?: 'plain' | 'compact' | 'percent'
   278→  duration?: number
   279→  className?: string
   280→}) {
   281→  const reduce = useReducedMotion()
   282→  const [display, setDisplay] = useState(value)
   283→  const rafRef = useRef<number | null>(null)
   284→
   285→  useEffect(() => {
   286→    if (reduce) return // reduced-motion users see raw value, no state writes
   287→    const start = display
   288→    const delta = value - start
   289→    if (delta === 0) return
   290→    const startTs = performance.now()
   291→    const tick = (now: number) => {
   292→      const t = Math.min(1, (now - startTs) / duration)
   293→      const eased = 1 - Math.pow(1 - t, 3)
   294→      setDisplay(start + delta * eased)
   295→      if (t < 1) {
   296→        rafRef.current = requestAnimationFrame(tick)
   297→      } else {
   298→        setDisplay(value)
   299→      }
   300→    }
   301→    rafRef.current = requestAnimationFrame(tick)
   302→    return () => {
   303→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   304→    }
   305→  }, [value, duration, reduce])
   306→
   307→  const shown = reduce ? value : display
   308→  if (format === 'compact') {
   309→    return <span className={className}>{formatNumber(Math.round(shown))}</span>
   310→  }
   311→  if (format === 'percent') {
   312→    return <span className={className}>{shown.toFixed(1)}%</span>
   313→  }
   314→  return <span className={className}>{Math.round(shown).toLocaleString()}</span>
   315→}

   322→/** Build a smooth Catmull-Rom (cardinal) spline through the given points. */
   323→function buildSplinePath(points: Array<{ x: number; y: number }>, tension = 0.18): string {
   324→  if (points.length < 2) return ''
   325→  let d = `M ${points[0].x.toFixed(2)} ${points[0].y.toFixed(2)}`
   326→  for (let i = 0; i < points.length - 1; i++) {
   327→    const p0 = points[i === 0 ? 0 : i - 1]
   328→    const p1 = points[i]
   329→    const p2 = points[i + 1]
   330→    const p3 = points[i + 2 < points.length ? i + 2 : points.length - 1]
   331→    const cp1x = p1.x + (p2.x - p0.x) * tension
   332→    const cp1y = p1.y + (p2.y - p0.y) * tension
   333→    const cp2x = p2.x - (p3.x - p1.x) * tension
   334→    const cp2y = p2.y - (p3.y - p1.y) * tension
   335→    d += ` C ${cp1x.toFixed(2)} ${cp1y.toFixed(2)}, ${cp2x.toFixed(2)} ${cp2y.toFixed(2)}, ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`
   336→  }
   337→  return d
   338→}

   340→function Sparkline({
   341→  data,
   342→  color,
   343→  height = 36,
   344→}: {
   345→  data: number[]
   346→  color: string
   347→  height?: number
   348→}) {
   349→  const reduce = useReducedMotion()
   350→  if (!data || data.length < 2) {
   351→    return <div style={{ height }} className="w-full" aria-hidden />
   352→  }
   353→
   354→  const width = 120 // viewBox width (scales to container via preserveAspectRatio)
   355→  const pad = 3
   356→  const innerW = width - pad * 2
   357→  const innerH = height - pad * 2
   358→  const min = Math.min(...data)
   359→  const max = Math.max(...data)
   360→  const range = max - min || 1
   361→  const stepX = innerW / (data.length - 1)
   362→
   363→  const points = data.map((v, i) => ({
   364→    x: pad + i * stepX,
   365→    y: pad + innerH * (1 - (v - min) / range),
   366→  }))
   367→
   368→  const linePath = buildSplinePath(points)
   369→  const fillPath = `${linePath} L ${points[points.length - 1].x.toFixed(2)} ${height} L ${points[0].x.toFixed(2)} ${height} Z`
   370→  const last = points[points.length - 1]
   371→  const gradId = `spark-fill-${color.replace('#', '')}`
   372→  const clipId = `spark-clip-${color.replace('#', '')}`
   373→
   374→  return (
   375→    <svg
   376→      viewBox={`0 0 ${width} ${height}`}
   377→      width="100%"
   378→      height={height}
   379→      preserveAspectRatio="none"
   380→      className="block overflow-visible"
   381→      aria-hidden
   382→    >
   383→      <defs>
   384→        <linearGradient id={gradId} x1="0" y1="0" x2="0" y2="1">
   385→          <stop offset="0%" stopColor={color} stopOpacity={0.45} />
   386→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   387→        </linearGradient>
   388→        <clipPath id={clipId}>
   389→          <rect x="0" y="0" width={width} height={height} />
   390→        </clipPath>
   391→      </defs>
   392→      <g clipPath={`url(#${clipId})`}>
   393→        <motion.path
   394→          d={fillPath}
   395→          fill={`url(#${gradId})`}
   396→          initial={{ opacity: 0 }}
   397→          animate={{ opacity: 1 }}
   398→          transition={{ duration: 0.6, ease: 'easeOut' }}
   399→        />
   400→        <motion.path
   401→          d={linePath}
   402→          fill="none"
   403→          stroke={color}
   404→          strokeWidth={1.5}
   405→          strokeLinecap="round"
   406→          strokeLinejoin="round"
   407→          vectorEffect="non-scaling-stroke"
   408→          initial={reduce ? false : { pathLength: 0 }}
   409→          animate={{ pathLength: 1 }}
   410→          transition={{ duration: 0.8, ease: 'easeOut' }}
   411→        />
   412→        {/* Last-point pulse dot */}
   413→        <motion.circle
   414→          cx={last.x}
   415→          cy={last.y}
   416→          r={2}
   417→          fill={color}
   418→          initial={reduce ? false : { scale: 0, opacity: 0 }}
   419→          animate={{ scale: 1, opacity: 1 }}
   420→          transition={{ duration: 0.3, delay: reduce ? 0 : 0.55 }}
   421→          style={{ transformOrigin: `${last.x}px ${last.y}px` }}
   422→        />
   423→      </g>
   424→    </svg>
   425→  )
   426→}

   477→function ChangePill({
   478→  direction,
   479→  label,
   480→  invertColor = false,
   481→}: {
   482→  direction: 'up' | 'down' | 'neutral'
   483→  label: string
   484→  invertColor?: boolean
   485→}) {
   486→  // invertColor = true means "up is bad" (e.g. error rate)
   487→  const good = direction === 'up' ? !invertColor : direction === 'down' ? invertColor : null
   488→  const isNeutral = direction === 'neutral'
   489→  const colorClass = isNeutral
   490→    ? 'text-muted-foreground bg-muted/60'
   491→    : good
   492→      ? 'text-emerald-700 dark:text-emerald-400 bg-emerald-500/10'
   493→      : 'text-red-700 dark:text-red-400 bg-red-500/10'
   494→  const Icon = direction === 'up' ? ArrowUpRight : direction === 'down' ? ArrowDownRight : Minus
   495→  return (
   496→    <span
   497→      className={`inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[11px] font-medium transition-colors ${colorClass}`}
   498→    >
   499→      <Icon className="h-3 w-3" />
   500→      {label}
   501→    </span>
   502→  )
   503→}

   511→export function DashboardView() {
   512→  const [data, setData] = useState<DashboardData>({
   513→    load: null,
   514→    uptime: null,
   515→    heartbeats: [],
   516→    tables: [],
   517→    pipelines: [],
   518→    functions: [],
   519→    queue: null,
   520→    logs: [],
   521→    pipelineRuns: [],
   522→    functionRuns: [],
   523→  })
   524→  const [loading, setLoading] = useState(true)
   525→  const [refreshing, setRefreshing] = useState(false)
   526→  const [lastUpdated, setLastUpdated] = useState<Date | null>(null)
   527→
   528→  const fetchAll = useCallback(async () => {
   529→    const [load, uptime, heartbeats, tables, pipelines, functions, queue, logs, pipelineRuns, functionRuns] =
   530→      await Promise.all([
   531→        fetchJson<LoadData>('/api/monitoring/load'),
   532→        fetchJson<UptimeData>('/api/monitoring/uptime'),
   533→        fetchJson<Heartbeat[]>('/api/monitoring/heartbeat?limit=60'),
   534→        fetchJson<SbTable[]>('/api/tables'),
   535→        fetchJson<PipelineSource[]>('/api/pipelines?isActive=true'),
   536→        fetchJson<SbFunction[]>('/api/functions'),
   537→        fetchJson<QueueData>('/api/queue'),
   538→        fetchJson<SourceError[]>('/api/logs?limit=8'),
   539→        fetchJson<PipelineRun[]>('/api/pipelines/runs?limit=5'),
   540→        fetchJson<FunctionRunItem[]>('/api/functions/runs?limit=5'),
   541→      ])
   542→
   543→    setData({
   544→      load: load ?? null,
   545→      uptime: uptime ?? null,
   546→      heartbeats: heartbeats ?? [],
   547→      tables: tables ?? [],
   548→      pipelines: pipelines ?? [],
   549→      functions: functions ?? [],
   550→      queue: queue ?? null,
   551→      logs: logs ?? [],
   552→      pipelineRuns: normalizeArray(pipelineRuns) ?? [],
   553→      functionRuns: normalizeArray(functionRuns) ?? [],
   554→    })
   555→    setLastUpdated(new Date())
   556→  }, [])
   557→
   558→  useEffect(() => {
   559→    let cancelled = false
   560→    const run = async () => {
   561→      await fetchAll()
   562→      if (!cancelled) setLoading(false)
   563→    }
   564→    run()
   565→    const id = setInterval(() => {
   566→      fetchAll()
   567→    }, REFRESH_INTERVAL_MS)
   568→    return () => {
   569→      cancelled = true
   570→      clearInterval(id)
   571→    }
   572→  }, [fetchAll])
   573→
   574→  const handleRefresh = useCallback(() => {
   575→    setRefreshing(true)
   576→    fetchAll().finally(() => setRefreshing(false))
   577→  }, [fetchAll])
   578→
   579→  if (loading) {
   580→    return <DashboardSkeleton />
   581→  }
   582→
   583→  return <DashboardContent data={data} lastUpdated={lastUpdated} onRefresh={handleRefresh} refreshing={refreshing} />
   584→}

   590→function DashboardContent({
   591→  data,
   592→  lastUpdated,
   593→  onRefresh,
   594→  refreshing,
   595→}: {
   596→  data: DashboardData
   597→  lastUpdated: Date | null
   598→  onRefresh: () => void
   599→  refreshing: boolean
   600→}) {
   601→  const reduce = useReducedMotion()
   602→
   603→  // ---- computed values -------------------------------------------------
   604→  const computed = useMemo(() => {
   605→    const load = data.load
   606→    const loadScore = load?.loadScore ?? 0
   607→    const cpuTotal = load?.cpu.total ?? 0
   608→    const cpuScraper = load?.cpu.scraper ?? 0
   609→    const cpuApi = load?.cpu.api ?? 0
   610→    const cpuFunctions = load?.cpu.functions ?? 0
   611→    const ramPercent = load?.memory.percent ?? 0
   612→    const ramUsedMb = load?.memory.usedMb ?? 0
   613→    const ramTotalMb = load?.memory.totalMb ?? 0
   614→    const connections = load?.connections ?? 0
   615→    const reqPerSec = load?.requestsPerSecond ?? 0
   616→    const activePipelines = data.pipelines.length
