   337→const ALL_METHODS: HttpMethod[] = ['GET', 'POST', 'PUT', 'PATCH', 'DELETE']

   339→let _rowIdCounter = 0
   340→function newRowId() {
   341→  _rowIdCounter += 1
   342→  return `row-${Date.now()}-${_rowIdCounter}`
   343→}
   344→
   345→function toKv(arr: Array<{ key: string; value: string }> | undefined): KeyValueRow[] {
   346→  if (!arr || arr.length === 0) {
   347→    return [{ id: newRowId(), key: '', value: '' }]
   348→  }
   349→  return arr.map((r) => ({ id: newRowId(), key: r.key, value: r.value }))
   350→}

   366→function StatusBadge({ status }: { status: number }) {
   367→  const firstDigit = String(status).charAt(0)
   368→  const colorMap: Record<string, string> = {
   369→    '2': 'bg-emerald-500/10 text-emerald-700 border-emerald-200 dark:text-emerald-400 dark:border-emerald-900',
   370→    '3': 'bg-blue-500/10 text-blue-700 border-blue-200 dark:text-blue-400 dark:border-blue-900',
   371→    '4': 'bg-amber-500/10 text-amber-700 border-amber-200 dark:text-amber-400 dark:border-amber-900',
   372→    '5': 'bg-red-500/10 text-red-700 border-red-200 dark:text-red-400 dark:border-red-900',
   373→  }
   374→  const cls = colorMap[firstDigit] ?? 'bg-muted text-muted-foreground border-border'
   375→  return (
   376→    <span className={`inline-flex items-center justify-center rounded-md border px-2 py-0.5 text-xs font-bold ${cls}`}>
   377→      {status}
   378→    </span>
   379→  )
   380→}

   382→function formatBytes(n: number) {
   383→  if (n < 1024) return `${n} B`
   384→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
   385→  return `${(n / (1024 * 1024)).toFixed(2)} MB`
   386→}
   387→
   388→function formatDuration(ms: number) {
   389→  if (ms < 1000) return `${ms} ms`
   390→  return `${(ms / 1000).toFixed(2)} s`
   391→}

   393→// =====================================================================

   396→
   397→interface KvEditorProps {
   398→  rows: KeyValueRow[]
   399→  onChange: (rows: KeyValueRow[]) => void
   400→  keyPlaceholder?: string
   401→  valuePlaceholder?: string
   402→}
   403→
   404→function KvEditor({ rows, onChange, keyPlaceholder = 'key', valuePlaceholder = 'value' }: KvEditorProps) {
   405→  const updateRow = (id: string, field: 'key' | 'value', val: string) => {
   406→    onChange(rows.map((r) => (r.id === id ? { ...r, [field]: val } : r)))
   407→  }
   408→  const removeRow = (id: string) => {
   409→    onChange(rows.filter((r) => r.id !== id))
   410→  }
   411→  const addRow = () => {
   412→    onChange([...rows, { id: newRowId(), key: '', value: '' }])
   413→  }
   414→
   415→  return (
   416→    <div className="space-y-2">
   417→      {rows.length === 0 && (
   418→        <p className="text-xs text-muted-foreground italic py-2">No entries. Click &quot;Add row&quot; to add one.</p>
   419→      )}
   420→      {rows.map((row) => (
   421→        <div key={row.id} className="flex items-center gap-2">
   422→          <Input
   423→            value={row.key}
   424→            onChange={(e) => updateRow(row.id, 'key', e.target.value)}
   425→            placeholder={keyPlaceholder}
   426→            className="h-8 font-mono text-xs flex-1"
   427→          />
   428→          <span className="text-muted-foreground text-xs">:</span>
   429→          <Input
   430→            value={row.value}
   431→            onChange={(e) => updateRow(row.id, 'value', e.target.value)}
   432→            placeholder={valuePlaceholder}
   433→            className="h-8 font-mono text-xs flex-1"
   434→          />
   435→          <Button
   436→            variant="ghost"
   437→            size="icon"
   438→            className="h-8 w-8 shrink-0 text-muted-foreground hover:text-destructive"
   439→            onClick={() => removeRow(row.id)}
   440→            aria-label="Remove row"
   441→          >
   442→            <Trash2 className="h-3.5 w-3.5" />
   443→          </Button>
   444→        </div>
   445→      ))}
   446→      <Button
   447→        variant="outline"
   448→        size="sm"
   449→        className="h-7 text-xs gap-1"
   450→        onClick={addRow}
   451→      >
   452→        <Plus className="h-3 w-3" />
   453→        Add row
   454→      </Button>
   455→    </div>
   456→  )
   457→}

   463→export function PlaygroundView() {
   464→  const { toast } = useToast()
   465→
   466→  const [selectedTemplate, setSelectedTemplate] = useState<EndpointTemplate | null>(null)
   467→  const [method, setMethod] = useState<HttpMethod>('GET')
   468→  const [url, setUrl] = useState('')
   469→  const [params, setParams] = useState<KeyValueRow[]>(toKv([]))
   470→  const [headers, setHeaders] = useState<KeyValueRow[]>(toKv([]))
   471→  const [body, setBody] = useState('')
   472→  const [activeTab, setActiveTab] = useState<'params' | 'headers' | 'body'>('params')
   473→  const [response, setResponse] = useState<ResponseData | null>(null)
   474→  const [loading, setLoading] = useState(false)
   475→  const [error, setError] = useState<string | null>(null)
   476→  const [search, setSearch] = useState('')
   477→  const [history, setHistory] = useState<HistoryEntry[]>([])
   478→  const [headersOpen, setHeadersOpen] = useState(false)
   479→
   480→  // Stats
   481→  const successRate = useMemo(() => {
   482→    if (history.length === 0) return 0
   483→    const ok = history.filter((h) => h.ok).length
   484→    return Math.round((ok / history.length) * 100)
   485→  }, [history])
   486→  const lastDuration = useMemo(() => {
   487→    if (history.length === 0) return null
   488→    return history[history.length - 1].durationMs
   489→  }, [history])
   490→
   491→  // Filtered templates
   492→  const filteredTemplates = useMemo(() => {
   493→    const q = search.trim().toLowerCase()
   494→    if (!q) return ENDPOINT_TEMPLATES
   495→    return ENDPOINT_TEMPLATES.filter(
   496→      (t) =>
   497→        t.path.toLowerCase().includes(q) ||
   498→        t.description.toLowerCase().includes(q) ||
   499→        t.category.toLowerCase().includes(q) ||
   500→        t.method.toLowerCase().includes(q),
   501→    )
   502→  }, [search])
   503→
   504→  const groupedTemplates = useMemo(() => {
   505→    const map = new Map<string, EndpointTemplate[]>()
   506→    filteredTemplates.forEach((t) => {
   507→      if (!map.has(t.category)) map.set(t.category, [])
   508→      map.get(t.category)!.push(t)
   509→    })
   510→    return Array.from(map.entries())
   511→  }, [filteredTemplates])
   512→
   513→  // Load template into request builder
   514→  const loadTemplate = useCallback((template: EndpointTemplate) => {
   515→    setSelectedTemplate(template)
   516→    setMethod(template.method)
   517→    setUrl(template.path)
   518→    setParams(toKv(template.defaultParams))
   519→    setHeaders(toKv(template.defaultHeaders))
   520→    setBody(template.defaultBody ?? '')
   521→    setResponse(null)
   522→    setError(null)
   523→    if (template.method === 'GET' || template.method === 'DELETE') {
   524→      setActiveTab('params')
   525→    } else {
   526→      setActiveTab('body')
   527→    }
   528→  }, [])
   529→
   530→  // Send request
   531→  const sendRequest = useCallback(async () => {
   532→    if (!url) {
   533→      toast({
   534→        title: 'URL is required',
   535→        description: 'Pick a template from the library or type a URL.',
   536→        variant: 'destructive',
   537→      })
   538→      return
   539→    }
   540→    setLoading(true)
   541→    setError(null)
   542→    setResponse(null)
   543→    const startTime = performance.now()
   544→    try {
   545→      // Build URL with params
   546→      const urlObj = new URL(url, window.location.origin)
   547→      params.forEach((p) => {
   548→        if (p.key.trim()) {
   549→          urlObj.searchParams.set(p.key, p.value)
   550→        }
   551→      })
   552→
   553→      // Build headers
   554→      const headerObj: Record<string, string> = {}
   555→      headers.forEach((h) => {
   556→        if (h.key.trim()) headerObj[h.key] = h.value
   557→      })
   558→      const hasBody = method === 'POST' || method === 'PUT' || method === 'PATCH'
   559→      if (!headerObj['Content-Type'] && hasBody) {
   560→        headerObj['Content-Type'] = 'application/json'
   561→      }
   562→
   563→      const res = await fetch(urlObj.toString(), {
   564→        method,
   565→        headers: headerObj,
   566→        body: hasBody && body ? body : undefined,
   567→      })
   568→
   569→      const responseText = await res.text()
   570→      const durationMs = Math.round(performance.now() - startTime)
   571→
   572→      // Parse response headers
   573→      const responseHeaders: Record<string, string> = {}
   574→      res.headers.forEach((value, key) => {
   575→        responseHeaders[key] = value
   576→      })
   577→
   578→      // Try to pretty-print JSON
   579→      let responseBody = responseText
   580→      try {
   581→        const parsed = JSON.parse(responseText)
   582→        responseBody = JSON.stringify(parsed, null, 2)
   583→      } catch {
   584→        // not JSON, keep raw text
   585→      }
   586→
   587→      const sizeBytes = new Blob([responseText]).size
   588→      const newResponse: ResponseData = {
   589→        status: res.status,
   590→        statusText: res.statusText,
   591→        headers: responseHeaders,
   592→        body: responseBody,
   593→        durationMs,
   594→        sizeBytes,
   595→        ok: res.ok,
   596→      }
   597→      setResponse(newResponse)
   598→      setHistory((prev) => [...prev, {
   599→        status: res.status,
   600→        durationMs,
   601→        timestamp: Date.now(),
   602→        ok: res.ok,
   603→      }].slice(-20))
   604→      toast({
   605→        title: res.ok ? 'Request completed' : 'Request returned non-OK',
   606→        description: `${res.status} ${res.statusText} · ${formatDuration(durationMs)} · ${formatBytes(sizeBytes)}`,
