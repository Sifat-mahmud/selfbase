   124→function mapLevel(errorType: string): 'error' | 'warning' | 'info' {
   125→  if (errorType === 'network_error' || errorType === 'execution_error') return 'error'
   126→  if (errorType === 'fetch_error' || errorType === 'validation_error' || errorType === 'transform_error' || errorType === 'timeout')
   127→    return 'warning'
   128→  return 'info'
   129→}

   131→function downloadCsv(filename: string, rows: Record<string, unknown>[]) {
   132→  if (rows.length === 0) return
   133→  const keys = Object.keys(rows[0])
   134→  const csv = [
   135→    keys.join(','),
   136→    ...rows.map((r) =>
   137→      keys
   138→        .map((k) => {
   139→          const v = r[k]
   140→          if (v == null) return ''
   141→          const s = typeof v === 'object' ? JSON.stringify(v) : String(v)
   142→          return `"${s.replace(/"/g, '""')}"`
   143→        })
   144→        .join(','),
   145→    ),
   146→  ].join('\n')
   147→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   148→  const url = URL.createObjectURL(blob)
   149→  const a = document.createElement('a')
   150→  a.href = url
   151→  a.download = filename
   152→  a.click()
   153→  URL.revokeObjectURL(url)
   154→}

   532→                                <ChevronRight className="h-4 w-4 text-muted-foreground" />
   533→                              )}
   534→                            </div>
   535→                          )}
   536→                        </div>
   537→                        {isExpanded && log.rawPayload && (
   538→                          <div className="px-4 pb-4 pl-11">
   539→                            <pre className="rounded-md bg-slate-950 text-emerald-400 p-3 text-xs font-mono whitespace-pre-wrap overflow-auto max-h-[200px]">
   540→                              {log.rawPayload}
   541→                            </pre>
   542→                          </div>
   543→                        )}
   544→                      </div>
   545→                    )
   546→                  })}
   547→                </div>
   548→              )}
   549→            </CardContent>
   550→          </Card>
   551→        </TabsContent>
   552→
   553→        {/* Function Errors Tab */}
   554→        <TabsContent value="function" className="space-y-4">
   555→          <Card>

   557→              <CardTitle className="text-base">Function Run Errors</CardTitle>

   560→              </CardDescription>
   561→            </CardHeader>

   563→              {functionErrors.length === 0 ? (
   564→                <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">

   568→              ) : (
   569→                <div className="divide-y max-h-[600px] overflow-y-auto">
   570→                  {functionErrors.map((err) => {
   571→                    const isExpanded = expandedLog === err.id
   572→                    const StatusIcon = err.status === 'timeout' ? AlertTriangle : XCircle

   574→                      <div key={err.id} className="hover:bg-muted/30 transition-colors">
   575→                        <div
   576→                          className="flex items-start gap-3 p-4 cursor-pointer"

   579→                          <StatusIcon className={`h-4 w-4 mt-0.5 shrink-0 ${err.status === 'timeout' ? 'text-amber-500' : 'text-red-500'}`} />
   580→                          <div className="flex-1 min-w-0">
   581→                            <div className="text-sm font-medium">
   582→                              {err.functionName ?? 'Unknown function'} — {err.status}

   597→                            </div>
   598→                          </div>
   599→                        </div>
   600→                        {isExpanded && (err.errorPayload || err.input) && (
   601→                          <div className="px-4 pb-4 pl-11 space-y-2">
   602→                            {err.errorPayload && (
   603→                              <div>
   604→                                <p className="text-xs text-muted-foreground mb-1">Error:</p>
   605→                                <pre className="rounded-md bg-slate-950 text-red-400 p-3 text-xs font-mono whitespace-pre-wrap overflow-auto max-h-[200px]">
   606→                                  {err.errorPayload}
   607→                                </pre>
   608→                              </div>
   609→                            )}
   610→                            {err.input && (
   611→                              <div>
   612→                                <p className="text-xs text-muted-foreground mb-1">Input:</p>
   613→                                <pre className="rounded-md bg-slate-950 text-emerald-400 p-3 text-xs font-mono whitespace-pre-wrap overflow-auto max-h-[150px]">
   614→                                  {err.input}
   615→                                </pre>
   616→                              </div>
   617→                            )}
   618→                          </div>
   619→                        )}
   620→                      </div>
   621→                    )
   622→                  })}
   623→                </div>
   624→              )}
   625→            </CardContent>
   626→          </Card>
   627→        </TabsContent>
   628→      </Tabs>
   629→
   630→      <AlertDialog open={showClearConfirm} onOpenChange={setShowClearConfirm}>
   631→        <AlertDialogContent>
   632→          <AlertDialogHeader>
   633→            <AlertDialogTitle>Clear all logs?</AlertDialogTitle>

   635→              This will dismiss all currently loaded log entries from view. The underlying database
   636→              records are not affected. Use this to clear your current view.
   637→            </AlertDialogDescription>
   638→          </AlertDialogHeader>
   639→          <AlertDialogFooter>
   640→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   641→            <AlertDialogAction
   642→              onClick={() => {

   644→                setSourceErrorsDetailed([])
   645→                setFunctionErrors([])
   646→                setShowClearConfirm(false)
   647→                toast({ title: 'Logs cleared', description: 'Log view has been cleared.' })
   648→              }}
   649→            >
   650→              Clear Logs
   651→            </AlertDialogAction>
   652→          </AlertDialogFooter>
   653→        </AlertDialogContent>
   654→      </AlertDialog>
   655→    </motion.div>
   656→  )
