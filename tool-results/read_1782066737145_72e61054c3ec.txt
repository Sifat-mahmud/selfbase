   118→   118→function getInitials(name: string | null, email: string): string {
   119→   119→  if (name && name.trim().length > 0) {
   120→   120→    const parts = name.trim().split(/\s+/)
   121→   121→    return (parts[0][0] + (parts[1]?.[0] ?? '')).toUpperCase()
   122→   122→  }
   123→   123→  return email[0]?.toUpperCase() ?? '?'
   124→   124→}

   126→   126→function avatarColor(email: string): string {
   127→   127→  const colors = [
   128→   128→    'bg-emerald-500/10 text-emerald-700',
   129→   129→    'bg-teal-500/10 text-teal-700',
   130→   130→    'bg-amber-500/10 text-amber-700',
   131→   131→    'bg-purple-500/10 text-purple-700',
   132→   132→    'bg-rose-500/10 text-rose-700',
   133→   133→    'bg-cyan-500/10 text-cyan-700',
   134→   134→  ]
   135→   135→  let hash = 0
   136→   136→  for (let i = 0; i < email.length; i++) hash = (hash * 31 + email.charCodeAt(i)) >>> 0
   137→   137→  return colors[hash % colors.length]
   138→   138→}

   140→   140→function generateApiKey(): string {
   141→   141→  const chars = 'abcdefghijklmnopqrstuvwxyz0123456789'
   142→   142→  let s = ''
   143→   143→  for (let i = 0; i < 32; i++) s += chars[Math.floor(Math.random() * chars.length)]
   144→   144→  return `sb_${s}`
   145→   145→}
