   195→async function fetchConfigMap(): Promise<Record<string, unknown>> {
   196→  try {
   197→    const res = await fetch('/api/config?limit=200')
   198→    if (!res.ok) return {}
   199→    const body = await res.json()
   200→    const list: Array<{ key: string; value: unknown }> = body?.data ?? body ?? []
   201→    const map: Record<string, unknown> = {}
   202→    for (const item of list) {
   203→      map[item.key] = item.value
   204→    }
   205→    return map
   206→  } catch {
   207→    return {}
   208→  }
   209→}
   210→
   211→function deriveInitial<T extends Record<string, unknown>>(
   212→  keys: string[],
   213→  configMap: Record<string, unknown>,
   214→): T {
   215→  const out = {} as Record<string, unknown>
   216→  for (const k of keys) {
   217→    const fallback = CONFIG_DEFAULTS[k]
   218→    if (k in configMap && configMap[k] !== undefined && configMap[k] !== null && configMap[k] !== '') {
   219→      out[k] = configMap[k]
   220→    } else {
   221→      out[k] = fallback
   222→    }
   223→  }
   224→  return out as T
   225→}

   227→async function saveConfig(values: Record<string, unknown>): Promise<{ ok: boolean; saved: number }> {
   228→  let saved = 0
   229→  let failed = 0
   230→  // Send sequentially to avoid race conditions on the upsert endpoint.
   231→  for (const [key, value] of Object.entries(values)) {
   232→    try {
   233→      const res = await fetch('/api/config', {
   234→        method: 'POST',
   235→        headers: { 'Content-Type': 'application/json' },
   236→        body: JSON.stringify({ key, value }),
   237→      })
   238→      if (res.ok) saved++
   239→      else failed++
   240→    } catch {
   241→      failed++
   242→    }
   243→  }
   244→  return { ok: failed === 0, saved }
   245→}

   371→function ValidationIndicator({ value, keyName }: { value: unknown; keyName: string }) {
   372→  const defaults: Record<string, unknown> = CONFIG_DEFAULTS
   373→  const isInsecureDefault =
   374→    (keyName === 'security.corsOrigins' && value === '*') ||
   375→    (keyName === 'security.ipWhitelist' && (value === '' || value == null)) ||
   376→    (keyName === 'general.adminEmail' && value === 'admin@selfbase.local') ||
   377→    (keyName === 'storage.cdnBaseUrl' && value === 'https://cdn.selfbase.local')
   378→  const isMissing =
   379→    value === '' || value === undefined || value === null
   380→  const isDefault =
   381→    keyName in defaults && value === defaults[keyName]
   382→
   383→  if (isMissing) {
   384→    return (
   385→      <Tooltip>
   386→        <TooltipTrigger asChild>
   387→          <XCircle className="h-3.5 w-3.5 text-red-500 shrink-0" />
   388→        </TooltipTrigger>
   389→        <TooltipContent>Required field is missing</TooltipContent>
   390→      </Tooltip>
   391→    )
   392→  }
   393→  if (isInsecureDefault) {
   394→    return (
   395→      <Tooltip>
   396→        <TooltipTrigger asChild>
   397→          <AlertTriangle className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   398→        </TooltipTrigger>
   399→        <TooltipContent>Using default or insecure value</TooltipContent>
   400→      </Tooltip>
   401→    )
   402→  }
   403→  if (isDefault) {
   404→    return (
   405→      <Tooltip>
   406→        <TooltipTrigger asChild>
   407→          <AlertTriangle className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   408→        </TooltipTrigger>
   409→        <TooltipContent>Using default value</TooltipContent>
   410→      </Tooltip>
   411→    )
   412→  }
   413→  return (
   414→    <Tooltip>
   415→      <TooltipTrigger asChild>
   416→        <CheckCircle2 className="h-3.5 w-3.5 text-emerald-500 shrink-0" />
   417→      </TooltipTrigger>
   418→      <TooltipContent>Properly configured</TooltipContent>
   419→    </Tooltip>
   420→  )
   421→}

  1879→function formatWebhookTime(dateStr: string): string {
  1880→  const diff = Date.now() - new Date(dateStr).getTime()
  1881→  const mins = Math.floor(diff / 60000)
  1882→  if (mins < 1) return 'Just now'
  1883→  if (mins < 60) return `${mins}m ago`
  1884→  const hrs = Math.floor(mins / 60)
  1885→  if (hrs < 24) return `${hrs}h ago`
  1886→  const days = Math.floor(hrs / 24)
  1887→  return `${days}d ago`
  1888→}
