   214→export function PipelineView() {
   215→  const { toast } = useToast()
   216→  const [pipelines, setPipelines] = useState<PipelineSourceItem[]>([])
   217→  const [allRuns, setAllRuns] = useState<PipelineRunItem[]>([])
   218→  const [loading, setLoading] = useState(true)
   219→  const [search, setSearch] = useState('')
   220→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   221→  const [selectedPipeline, setSelectedPipeline] = useState<PipelineSourceItem | null>(null)
   222→  const [previewTarget, setPreviewTarget] = useState<PipelineSourceItem | null>(null)
   223→  const [previewResult, setPreviewResult] = useState<PreviewResult | null>(null)
   224→  const [previewLoading, setPreviewLoading] = useState(false)
   225→  const [runningId, setRunningId] = useState<string | null>(null)
   226→
   227→  // Create form
   228→  const [newName, setNewName] = useState('')
   229→  const [newDesc, setNewDesc] = useState('')
   230→  const [newType, setNewType] = useState<'rest' | 'rss' | 'websocket' | 'scraper'>('rest')
   231→  const [newUrl, setNewUrl] = useState('')
   232→  const [newMethod, setNewMethod] = useState('GET')
   233→  const [newHeaders, setNewHeaders] = useState('{}')
   234→  const [newJsonPath, setNewJsonPath] = useState('')
   235→  const [newInterval, setNewInterval] = useState('300')
   236→  const [newOnConflict, setNewOnConflict] = useState('update')
   237→  const [newMappings, setNewMappings] = useState<ColumnMapping[]>([])
   238→
   239→  const loadAll = useCallback(async () => {
   240→    setLoading(true)
   241→    try {
   242→      const [p, r] = await Promise.all([
   243→        apiGet<PipelineSourceItem[]>('/api/pipelines'),
   244→        apiGet<PipelineRunItem[] | PipelineRunsResponse>('/api/pipelines/runs?limit=100').catch(() => null),
   245→      ])
   246→      const list = Array.isArray(p) ? p : []
   247→      // Map targetTable.name -> targetTableName for convenience, normalize columnMappings
   248→      const normalized = list.map((pl) => ({
   249→        ...pl,
   250→        targetTableName: pl.targetTableName ?? pl.targetTable?.name ?? null,
   251→        lastRun: pl.pipelineRuns && pl.pipelineRuns.length > 0 ? pl.pipelineRuns[0] : null,
   252→      }))
   253→      setPipelines(normalized)
   254→      // apiGet already unwraps `{ success, data }` envelopes, so the result is usually
   255→      // the raw array. Handle both shapes defensively.
   256→      const runs = Array.isArray(r) ? r : (r?.data ?? [])
   257→      setAllRuns(Array.isArray(runs) ? runs : [])
   258→    } catch (err) {
   259→      toast({
   260→        title: 'Failed to load pipelines',
   261→        description: err instanceof Error ? err.message : 'Unknown error',
   262→        variant: 'destructive',
   263→      })
   264→    } finally {
   265→      setLoading(false)
   266→    }
   267→  }, [toast])
   268→
   269→  useEffect(() => {
   270→    void loadAll()
   271→  }, [loadAll])
   272→
   273→  const handleCreate = async () => {
   274→    if (!newName.trim() || !newUrl.trim()) {
   275→      toast({ title: 'Name and URL required', variant: 'destructive' })
   276→      return
   277→    }
   278→    try {
   279→      const payload: Record<string, unknown> = {
   280→        name: newName.trim(),
   281→        description: newDesc || null,
   282→        sourceType: newType,
   283→        url: newUrl.trim(),
   284→        method: newMethod,
   285→        headers: newHeaders || null,
   286→        jsonPath: newJsonPath || null,
   287→        fetchInterval: Number(newInterval) || 300,
   288→        onConflict: newOnConflict,
   289→        isActive: true,
   290→        columnMappings: JSON.stringify(newMappings),
   291→      }
   292→      const created = await apiPost<PipelineSourceItem>('/api/pipelines', payload)
   293→      setPipelines((prev) => [
   294→        {
   295→          ...created,
   296→          targetTableName: created.targetTable?.name ?? null,
   297→          lastRun: null,
   298→        },
   299→        ...prev,
   300→      ])
   301→      setShowCreateDialog(false)
   302→      setNewName('')
   303→      setNewDesc('')
   304→      setNewUrl('')
   305→      setNewMethod('GET')
   306→      setNewHeaders('{}')
   307→      setNewJsonPath('')
   308→      setNewInterval('300')
   309→      setNewOnConflict('update')
   310→      setNewMappings([])
   311→      toast({ title: 'Pipeline created', description: `"${created.name}" has been created.` })
   312→    } catch (err) {
   313→      toast({
   314→        title: 'Failed to create pipeline',
   315→        description: err instanceof Error ? err.message : 'Unknown error',
   316→        variant: 'destructive',
   317→      })
   318→    }
   319→  }
   320→
   321→  const handleToggleActive = async (pipeline: PipelineSourceItem) => {
   322→    try {
   323→      await apiPut(`/api/pipelines/${pipeline.id}`, { isActive: !pipeline.isActive })
   324→      setPipelines((prev) =>
   325→        prev.map((p) => (p.id === pipeline.id ? { ...p, isActive: !p.isActive } : p)),
   326→      )
   327→      toast({ title: `Pipeline ${pipeline.isActive ? 'paused' : 'activated'}` })
   328→    } catch (err) {
   329→      toast({
   330→        title: 'Failed to update pipeline',
   331→        description: err instanceof Error ? err.message : 'Unknown error',
   332→        variant: 'destructive',
   333→      })
   334→    }
   335→  }
   336→
   337→  const handleRunNow = async (pipeline: PipelineSourceItem) => {
   338→    setRunningId(pipeline.id)
   339→    try {
   340→      const result = await apiPost<{ runId: string; status: string; rowsFetched?: number; rowsWritten?: number; error?: string }>(
   341→        `/api/pipelines/${pipeline.id}/run`,
   342→        {},
   343→      )
   344→      toast({
   345→        title: result.status === 'success' ? 'Pipeline run completed' : 'Pipeline run finished',
   346→        description:
   347→          result.status === 'success'
   348→            ? `Fetched ${result.rowsFetched ?? 0} rows, wrote ${result.rowsWritten ?? 0}.`
   349→            : result.error ?? `Status: ${result.status}`,
   350→        variant: result.status === 'success' ? 'default' : 'destructive',
   351→      })
   352→      await loadAll()
   353→    } catch (err) {
   354→      toast({
   355→        title: 'Pipeline run failed',
   356→        description: err instanceof Error ? err.message : 'Unknown error',
   357→        variant: 'destructive',
   358→      })
   359→    } finally {
   360→      setRunningId(null)
   361→    }
   362→  }
   363→
   364→  const handlePreview = async (pipeline: PipelineSourceItem) => {
   365→    setPreviewTarget(pipeline)
   366→    setPreviewResult(null)
   367→    setPreviewLoading(true)
   368→    try {
   369→      const result = await apiPost<PreviewResult>(`/api/pipelines/${pipeline.id}/preview`, {})
   370→      setPreviewResult(result)
   371→    } catch (err) {
   372→      setPreviewResult({
   373→        success: false,
   374→        error: err instanceof Error ? err.message : 'Unknown error',
   375→      })
   376→    } finally {
   377→      setPreviewLoading(false)
   378→    }
   379→  }
   380→
   381→  const filteredPipelines = pipelines.filter(
   382→    (p) =>
   383→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   384→      (p.description ?? '').toLowerCase().includes(search.toLowerCase()) ||
   385→      p.url.toLowerCase().includes(search.toLowerCase()),
   386→  )
   387→
   388→  // Build chart data from allRuns grouped by day
   389→  const chartData = (() => {
   390→    const map = new Map<string, { day: string; success: number; failed: number; rows: number }>()
   391→    for (const r of allRuns) {
   392→      const d = new Date(r.startedAt)
   393→      const key = d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   394→      const entry = map.get(key) ?? { day: key, success: 0, failed: 0, rows: 0 }
   395→      if (r.status === 'success') entry.success += 1
   396→      if (r.status === 'failed' || r.status === 'timeout') entry.failed += 1
   397→      entry.rows += r.rowsWritten ?? 0
   398→      map.set(key, entry)
   399→    }
   400→    return Array.from(map.values()).slice(-7)
   401→  })()
   402→
   403→  // ----- Pipeline Analytics (Task 12) -----
   404→  // Most recent 20 runs, oldest first so the timeline reads left → right.
   405→  const timelineData = allRuns
   406→    .slice(0, 20)
   407→    .reverse()
   408→    .map((run) => {
   409→      const d = new Date(run.startedAt)
   410→      const time = d.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: false })
   411→      return {
   412→        time,
   413→        duration: run.durationMs ?? 0,
   414→        status: run.status,
   415→        rows: run.rowsWritten ?? 0,
   416→        name: run.source?.name ?? 'Pipeline',
   417→      }
   418→    })
   419→
   420→  // Status distribution for the donut chart.
   421→  const statusCounts = allRuns.reduce((acc, run) => {
   422→    acc[run.status] = (acc[run.status] || 0) + 1
   423→    return acc
   424→  }, {} as Record<string, number>)
   425→
   426→  const pieData = Object.entries(statusCounts).map(([name, value]) => ({
   427→    name,
   428→    value,
   429→    color: STATUS_COLORS[name] ?? STATUS_COLORS.pending,
   430→  }))
   431→
   432→  // Stat cards
   433→  const totalRuns = allRuns.length
   434→  const successCount = allRuns.filter((r) => r.status === 'success').length
   435→  const successRate = totalRuns > 0 ? Math.round((successCount / totalRuns) * 100) : 0
   436→  const completedRuns = allRuns.filter((r) => r.durationMs != null)
   437→  const avgDuration =
   438→    completedRuns.length > 0
   439→      ? Math.round(completedRuns.reduce((s, r) => s + (r.durationMs ?? 0), 0) / completedRuns.length)
   440→      : 0
   441→  const totalRowsWritten = allRuns.reduce((s, r) => s + (r.rowsWritten ?? 0), 0)
   442→
   443→  if (loading) {
   444→    return (
   445→      <div className="space-y-6">
   446→        <div className="flex justify-between">
   447→          <Skeleton className="h-8 w-48" />
   448→          <Skeleton className="h-9 w-36" />
   449→        </div>
   450→        <div className="grid gap-4 md:grid-cols-2">
   451→          {Array.from({ length: 4 }).map((_, i) => (
   452→            <Card key={i}>
   453→              <CardContent className="p-4">
   454→                <Skeleton className="h-32 w-full" />
   455→              </CardContent>
   456→            </Card>
   457→          ))}
   458→        </div>
   459→      </div>
   460→    )
   461→  }
   462→
   463→  if (selectedPipeline) {
   464→    const SourceIcon = sourceIcons[selectedPipeline.sourceType]
   465→    const lastRun = selectedPipeline.lastRun ?? selectedPipeline.pipelineRuns?.[0] ?? null
   466→    const LastStatusIcon = lastRun ? statusIcons[lastRun.status] : Clock
   467→    const mappings = parseJsonField<ColumnMapping[]>(selectedPipeline.columnMappings, [])
   468→    const sourceRuns = allRuns.filter((r) => r.sourceId === selectedPipeline.id)
   469→
   470→    return (
   471→      <motion.div
   472→        initial={{ opacity: 0, y: 8 }}
   473→        animate={{ opacity: 1, y: 0 }}
   474→        transition={{ duration: 0.2 }}
   475→        className="space-y-6"
   476→      >
   477→        <div className="flex items-center gap-2 flex-wrap">
   478→          <Button variant="ghost" size="sm" onClick={() => setSelectedPipeline(null)}>
   479→            <ArrowRight className="h-4 w-4 rotate-180" /> Back
   480→          </Button>
   481→          <Separator orientation="vertical" className="h-6" />
   482→          <div className="rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 p-2 text-white">
   483→            <SourceIcon className="h-5 w-5" />
   484→          </div>
   485→          <h1 className="text-2xl font-bold tracking-tight">{selectedPipeline.name}</h1>
   486→          <Badge variant="outline" className={`capitalize ${sourceTypeBadge[selectedPipeline.sourceType]}`}>
   487→            {selectedPipeline.sourceType}
   488→          </Badge>
   489→          <Badge variant={selectedPipeline.isActive ? 'default' : 'secondary'}>
   490→            {selectedPipeline.isActive ? 'Active' : 'Paused'}
   491→          </Badge>
   492→          <div className="ml-auto flex gap-2">
   493→            <Button variant="outline" size="sm" onClick={() => void handlePreview(selectedPipeline)}>
   494→              <Eye className="mr-1 h-3.5 w-3.5" /> Preview
   495→            </Button>
   496→            <Button
   497→              size="sm"
   498→              onClick={() => void handleRunNow(selectedPipeline)}
   499→              disabled={runningId === selectedPipeline.id}
   500→            >
   501→              <Play className="mr-1 h-3.5 w-3.5" />
   502→              {runningId === selectedPipeline.id ? 'Running...' : 'Run Now'}
   503→            </Button>
   504→          </div>
   505→        </div>
   506→
   507→        <div className="grid gap-4 md:grid-cols-4">
   508→          <Card>
   509→            <CardContent className="pt-4">
   510→              <div className="text-sm text-muted-foreground">Source Type</div>
   511→              <div className="text-lg font-bold capitalize">{selectedPipeline.sourceType}</div>
   512→            </CardContent>
   513→          </Card>
   514→          <Card>
   515→            <CardContent className="pt-4">
   516→              <div className="text-sm text-muted-foreground">Fetch Interval</div>
   517→              <div className="text-lg font-bold">
   518→                {selectedPipeline.fetchInterval > 0 ? `${selectedPipeline.fetchInterval}s` : 'Real-time'}
   519→              </div>
   520→            </CardContent>
   521→          </Card>
   522→          <Card>
   523→            <CardContent className="pt-4">
   524→              <div className="text-sm text-muted-foreground">Target Table</div>
   525→              <div className="text-lg font-bold font-mono">{selectedPipeline.targetTableName ?? '—'}</div>
   526→            </CardContent>
   527→          </Card>
   528→          <Card>
   529→            <CardContent className="pt-4">
   530→              <div className="text-sm text-muted-foreground">Last Run</div>
   531→              <div className="flex items-center gap-1">
   532→                {lastRun && <LastStatusIcon className={`h-4 w-4 ${statusColors[lastRun.status]}`} />}
   533→                <span className="text-lg font-bold capitalize">{lastRun?.status ?? 'Never'}</span>
   534→              </div>
   535→              {lastRun?.durationMs != null && (
   536→                <p className="text-xs text-muted-foreground mt-0.5">{lastRun.durationMs}ms · {lastRun.rowsWritten ?? 0} rows written</p>
   537→              )}
   538→            </CardContent>
   539→          </Card>
   540→        </div>
   541→
   542→        <Card>
   543→          <CardHeader>
   544→            <CardTitle className="text-base">Source URL & Configuration</CardTitle>
   545→            <CardDescription>Pipeline source endpoint and extraction settings</CardDescription>
   546→          </CardHeader>
   547→          <CardContent className="space-y-3">
   548→            <div className="flex items-center gap-2">
   549→              <Badge variant="outline" className="font-mono">
   550→                {selectedPipeline.method}
   551→              </Badge>
   552→              <code className="text-sm font-mono break-all">{selectedPipeline.url}</code>
   553→              <a
   554→                href={selectedPipeline.url}
   555→                target="_blank"
   556→                rel="noopener noreferrer"
   557→                className="ml-auto text-muted-foreground hover:text-emerald-600"
   558→              >
   559→                <ExternalLink className="h-3.5 w-3.5" />
   560→              </a>
   561→            </div>
   562→            {selectedPipeline.jsonPath && (
   563→              <div className="text-sm">
   564→                <span className="text-muted-foreground">JSONPath: </span>
   565→                <code className="font-mono">{selectedPipeline.jsonPath}</code>
   566→              </div>
   567→            )}
