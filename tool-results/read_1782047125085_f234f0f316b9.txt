   149→/** Unwrap `{ success, data }` envelope; fall back to raw payload. */
   150→async function fetchJson<T>(url: string): Promise<T | null> {
   151→  try {
   152→    const res = await fetch(url, { cache: 'no-store' })
   153→    if (!res.ok) return null
   154→    const json = await res.json()
   155→    if (json && typeof json === 'object' && 'success' in json && 'data' in json) {
   156→      return (json as { data: T }).data as T
   157→    }
   158→    return json as T
   159→  } catch {
   160→    return null
   161→  }
   162→}

   164→function formatNumber(n: number, mode: 'compact' | 'full' = 'compact'): string {
   165→  if (!Number.isFinite(n)) return '0'
   166→  if (mode === 'full') return n.toLocaleString()
   167→  if (Math.abs(n) >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`
   168→  if (Math.abs(n) >= 1_000) return `${(n / 1_000).toFixed(1)}K`
   169→  return n.toLocaleString()
   170→}

   172→function formatRelativeTime(dateStr: string): string {
   173→  const now = Date.now()
   174→  const then = new Date(dateStr).getTime()
   175→  const diff = Math.max(0, now - then)
   176→  const sec = Math.floor(diff / 1000)
   177→  if (sec < 60) return `${sec}s ago`
   178→  const min = Math.floor(sec / 60)
   179→  if (min < 60) return `${min}m ago`
   180→  const hr = Math.floor(min / 60)
   181→  if (hr < 24) return `${hr}h ago`
   182→  const day = Math.floor(hr / 24)
   183→  if (day < 30) return `${day}d ago`
   184→  const mo = Math.floor(day / 30)
   185→  if (mo < 12) return `${mo}mo ago`
   186→  return `${Math.floor(mo / 12)}y ago`
   187→}
   188→
   189→function formatDuration(ms: number): string {
   190→  if (!Number.isFinite(ms) || ms <= 0) return '0m'
   191→  const sec = Math.floor(ms / 1000)
   192→  const days = Math.floor(sec / 86400)
   193→  const hours = Math.floor((sec % 86400) / 3600)
   194→  const mins = Math.floor((sec % 3600) / 60)
   195→  if (days > 0) return `${days}d ${hours}h ${mins}m`
   196→  if (hours > 0) return `${hours}h ${mins}m`
   197→  return `${mins}m`
   198→}

   229→function AnimatedNumber({
   230→  value,
   231→  format = 'plain',
   232→  duration = 800,
   233→  className,
   234→}: {
   235→  value: number
   236→  format?: 'plain' | 'compact' | 'percent'
   237→  duration?: number
   238→  className?: string
   239→}) {
   240→  const reduce = useReducedMotion()
   241→  const [display, setDisplay] = useState(value)
   242→  const rafRef = useRef<number | null>(null)
   243→
   244→  useEffect(() => {
   245→    if (reduce) return // reduced-motion users see raw value, no state writes
   246→    const start = display
   247→    const delta = value - start
   248→    if (delta === 0) return
   249→    const startTs = performance.now()
   250→    const tick = (now: number) => {
   251→      const t = Math.min(1, (now - startTs) / duration)
   252→      const eased = 1 - Math.pow(1 - t, 3)
   253→      setDisplay(start + delta * eased)
   254→      if (t < 1) {
   255→        rafRef.current = requestAnimationFrame(tick)
   256→      } else {
   257→        setDisplay(value)
   258→      }
   259→    }
   260→    rafRef.current = requestAnimationFrame(tick)
   261→    return () => {
   262→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   263→    }
   264→  }, [value, duration, reduce])
   265→
   266→  const shown = reduce ? value : display
   267→  if (format === 'compact') {
   268→    return <span className={className}>{formatNumber(Math.round(shown))}</span>
   269→  }
   270→  if (format === 'percent') {
   271→    return <span className={className}>{shown.toFixed(1)}%</span>
   272→  }
   273→  return <span className={className}>{Math.round(shown).toLocaleString()}</span>
   274→}

   280→function Sparkline({
   281→  data,
   282→  color,
   283→  height = 36,
   284→}: {
   285→  data: number[]
   286→  color: string
   287→  height?: number
   288→}) {
   289→  if (data.length === 0) {
   290→    return <div style={{ height }} className="w-full" />
   291→  }
   292→  const chartData = data.map((v, i) => ({ i, v }))
   293→  const id = `spark-${color.replace('#', '')}`
   294→  return (
   295→    <div style={{ height }} className="w-full">
   296→      <ResponsiveContainer width="100%" height="100%">
   297→        <AreaChart data={chartData} margin={{ top: 2, right: 0, bottom: 0, left: 0 }}>
   298→          <defs>
   299→            <linearGradient id={id} x1="0" y1="0" x2="0" y2="1">
   300→              <stop offset="0%" stopColor={color} stopOpacity={0.45} />
   301→              <stop offset="100%" stopColor={color} stopOpacity={0} />
   302→            </linearGradient>
   303→          </defs>
   304→          <Area
   305→            type="monotone"
   306→            dataKey="v"
   307→            stroke={color}
   308→            strokeWidth={1.5}
   309→            fill={`url(#${id})`}
   310→            isAnimationActive={false}
   311→            dot={false}
   312→          />
   313→        </AreaChart>
   314→      </ResponsiveContainer>
   315→    </div>
   316→  )
   317→}

   323→function ChangePill({
   324→  direction,
   325→  label,
   326→  invertColor = false,
   327→}: {
   328→  direction: 'up' | 'down' | 'neutral'
   329→  label: string
   330→  invertColor?: boolean
   331→}) {
   332→  // invertColor = true means "up is bad" (e.g. error rate)
   333→  const good = direction === 'up' ? !invertColor : direction === 'down' ? invertColor : null
   334→  const colorClass =
   335→    direction === 'neutral'
   336→      ? 'text-muted-foreground'
   337→      : good
   338→        ? 'text-emerald-600'
   339→        : 'text-red-600'
   340→  const Icon = direction === 'up' ? ArrowUpRight : direction === 'down' ? ArrowDownRight : Minus
   341→  return (
   342→    <span className={`flex items-center gap-0.5 font-medium ${colorClass}`}>
   343→      <Icon className="h-3 w-3" />
   344→      {label}
   345→    </span>
   346→  )
   347→}

   355→export function DashboardView() {
   356→  const [data, setData] = useState<DashboardData>({
   357→    load: null,
   358→    uptime: null,
   359→    heartbeats: [],
   360→    tables: [],
   361→    pipelines: [],
   362→    functions: [],
   363→    queue: null,
   364→    logs: [],
   365→  })
   366→  const [loading, setLoading] = useState(true)
   367→  const [refreshing, setRefreshing] = useState(false)
   368→  const [lastUpdated, setLastUpdated] = useState<Date | null>(null)
   369→
   370→  const fetchAll = useCallback(async () => {
   371→    const [load, uptime, heartbeats, tables, pipelines, functions, queue, logs] =
   372→      await Promise.all([
   373→        fetchJson<LoadData>('/api/monitoring/load'),
   374→        fetchJson<UptimeData>('/api/monitoring/uptime'),
   375→        fetchJson<Heartbeat[]>('/api/monitoring/heartbeat?limit=60'),
   376→        fetchJson<SbTable[]>('/api/tables'),
   377→        fetchJson<PipelineSource[]>('/api/pipelines?isActive=true'),
   378→        fetchJson<SbFunction[]>('/api/functions'),
   379→        fetchJson<QueueData>('/api/queue'),
   380→        fetchJson<SourceError[]>('/api/logs?limit=8'),
   381→      ])
   382→
   383→    setData({
   384→      load: load ?? null,
   385→      uptime: uptime ?? null,
   386→      heartbeats: heartbeats ?? [],
   387→      tables: tables ?? [],
   388→      pipelines: pipelines ?? [],
   389→      functions: functions ?? [],
   390→      queue: queue ?? null,
   391→      logs: logs ?? [],
   392→    })
   393→    setLastUpdated(new Date())
   394→  }, [])
   395→
   396→  useEffect(() => {
   397→    let cancelled = false
   398→    const run = async () => {
   399→      await fetchAll()
   400→      if (!cancelled) setLoading(false)
   401→    }
   402→    run()
   403→    const id = setInterval(() => {
   404→      fetchAll()
   405→    }, REFRESH_INTERVAL_MS)
   406→    return () => {
   407→      cancelled = true
   408→      clearInterval(id)
   409→    }
   410→  }, [fetchAll])
   411→
   412→  const handleRefresh = useCallback(() => {
   413→    setRefreshing(true)
   414→    fetchAll().finally(() => setRefreshing(false))
   415→  }, [fetchAll])
   416→
   417→  if (loading) {
   418→    return <DashboardSkeleton />
   419→  }
   420→
   421→  return <DashboardContent data={data} lastUpdated={lastUpdated} onRefresh={handleRefresh} refreshing={refreshing} />
   422→}

   428→function DashboardContent({
   429→  data,
   430→  lastUpdated,
   431→  onRefresh,
   432→  refreshing,
   433→}: {
   434→  data: DashboardData
   435→  lastUpdated: Date | null
   436→  onRefresh: () => void
   437→  refreshing: boolean
   438→}) {
   439→  const reduce = useReducedMotion()
   440→
   441→  // ---- computed values -------------------------------------------------
   442→  const computed = useMemo(() => {
   443→    const load = data.load
   444→    const loadScore = load?.loadScore ?? 0
   445→    const cpuTotal = load?.cpu.total ?? 0
   446→    const cpuScraper = load?.cpu.scraper ?? 0
   447→    const cpuApi = load?.cpu.api ?? 0
   448→    const cpuFunctions = load?.cpu.functions ?? 0
   449→    const ramPercent = load?.memory.percent ?? 0
   450→    const ramUsedMb = load?.memory.usedMb ?? 0
   451→    const ramTotalMb = load?.memory.totalMb ?? 0
   452→    const connections = load?.connections ?? 0
   453→    const reqPerSec = load?.requestsPerSecond ?? 0
   454→    const activePipelines = data.pipelines.length
   455→    const activeFunctions = data.functions.filter((f) => f.isActive).length
   456→
   457→    const totalTables = data.tables.length
   458→    const totalRows = data.tables.reduce((sum, t) => sum + (t.rowCount ?? 0), 0)
   459→    const recentErrorsTotal = load?.recentErrors.total ?? 0
   460→
   461→    // error rate: errors in last hour vs estimated requests in last hour
   462→    const estimatedHourlyRequests = reqPerSec * 3600
   463→    const errorRate =
   464→      estimatedHourlyRequests > 0
   465→        ? Math.min((recentErrorsTotal / estimatedHourlyRequests) * 100, 100)
   466→        : recentErrorsTotal > 0
   467→          ? 100
   468→          : 0
   469→
   470→    // uptime
   471→    const uptimePercent = data.uptime?.uptimePercent ?? 100
   472→    const heartbeatCount = data.uptime?.heartbeatCount ?? 0
   473→
   474→    // uptime duration — oldest heartbeat to now
   475→    const oldestHeartbeat = data.heartbeats.length > 0
   476→      ? data.heartbeats[data.heartbeats.length - 1]
   477→      : null
   478→    const uptimeDurationMs = oldestHeartbeat
   479→      ? Date.now() - new Date(oldestHeartbeat.recordedAt).getTime()
   480→      : 0
   481→    const uptimeDuration = formatDuration(uptimeDurationMs)
   482→
   483→    // queue totals
   484→    const queueStats = data.queue?.stats ?? []
   485→    const queuedCount = queueStats.find((s) => s.status === 'queued')?.count ?? 0
   486→    const processingCount = queueStats.find((s) => s.status === 'processing')?.count ?? 0
   487→    const queueTotal = queueStats.reduce((sum, s) => sum + (s.count ?? 0), 0)
   488→
   489→    // direction (latest vs previous heartbeat)
   490→    const hb = data.heartbeats
   491→    const latest = hb[0] ?? null
   492→    const prev = hb[1] ?? null
   493→    const dir = (curr?: number | null, p?: number | null): 'up' | 'down' | 'neutral' => {
   494→      if (curr == null || p == null || curr === p) return 'neutral'
   495→      return curr > p ? 'up' : 'down'
   496→    }
   497→    const connectionsDir = dir(latest?.activeConnections, prev?.activeConnections)
   498→    const reqDir = dir(latest?.reqPerSec, prev?.reqPerSec)
   499→    const loadDir = dir(latest?.loadScore, prev?.loadScore)
   500→    const cpuDir = dir(latest?.cpuTotal, prev?.cpuTotal)
   501→
   502→    // server health from load level
   503→    // Only consider uptime a concern if there are enough heartbeats to make
   504→    // the measurement meaningful (at least 60 = 1 hour of 1-min intervals)
   505→    const uptimeIsConcern = uptimePercent < 95 && heartbeatCount > 60
   506→    const health: 'healthy' | 'degraded' | 'down' =
   507→      loadScore >= 85
   508→        ? 'down'
   509→        : loadScore >= 60 || uptimeIsConcern
   510→          ? 'degraded'
   511→          : 'healthy'
   512→
   513→    // sparkline series (oldest → newest)
   514→    const sparkConnections = hb.slice().reverse().map((h) => h.activeConnections)
   515→    const sparkReq = hb.slice().reverse().map((h) => h.reqPerSec)
   516→    const sparkLoad = hb.slice().reverse().map((h) => h.loadScore)
   517→    const sparkCpu = hb.slice().reverse().map((h) => h.cpuTotal)
   518→    const sparkRam = hb.slice().reverse().map((h) => h.ramUsedMb)
   519→
   520→    return {
   521→      loadScore,
   522→      loadLevel: load?.loadLevel ?? 'low',
   523→      cpuTotal,
   524→      cpuScraper,
   525→      cpuApi,
   526→      cpuFunctions,
   527→      ramPercent,
   528→      ramUsedMb,
   529→      ramTotalMb,
   530→      connections,
   531→      reqPerSec,
   532→      activePipelines,
   533→      activeFunctions,
   534→      activeScrapers: load?.activeJobs.scrapers ?? 0,
   535→      activePipelineRuns: load?.activeJobs.pipelines ?? 0,
   536→      totalTables,
   537→      totalRows,
   538→      recentErrorsTotal,
   539→      errorRate,
   540→      uptimePercent,
   541→      heartbeatCount,
   542→      uptimeDuration,
   543→      queuedCount,
   544→      processingCount,
   545→      queueTotal,
   546→      connectionsDir,
   547→      reqDir,
   548→      loadDir,
   549→      cpuDir,
   550→      health,
   551→      sparkConnections,
   552→      sparkReq,
   553→      sparkLoad,
   554→      sparkCpu,
   555→      sparkRam,
   556→    }
   557→  }, [data])
   558→
   559→  // chart data — reverse so oldest is first
   560→  const chartData = useMemo(() => {
   561→    return data.heartbeats
   562→      .slice()
   563→      .reverse()
   564→      .map((h) => ({
   565→        time: new Date(h.recordedAt).toLocaleTimeString('en-US', {
   566→          hour: '2-digit',
   567→          minute: '2-digit',
   568→        }),
   569→        cpu: h.cpuTotal,
   570→        ram: Math.round((h.ramUsedMb / (data.load?.memory.totalMb || 4096)) * 100),
   571→        reqPerSec: h.reqPerSec,
   572→        loadScore: Math.round(h.loadScore),
   573→        connections: h.activeConnections,
   574→      }))
   575→  }, [data.heartbeats, data.load])
   576→
   577→  // framer-motion variants
   578→  const containerVariants: Variants = {
   579→    hidden: { opacity: 0 },
   580→    visible: {
   581→      opacity: 1,
   582→      transition: { staggerChildren: reduce ? 0 : 0.06, delayChildren: reduce ? 0 : 0.05 },
   583→    },
   584→  }
   585→  const itemVariants: Variants = {
   586→    hidden: { opacity: 0, y: 12 },
   587→    visible: {
   588→      opacity: 1,
   589→      y: 0,
   590→      transition: { duration: 0.4, ease: 'easeOut' },
   591→    },
   592→  }
   593→
   594→  const healthLabel =
   595→    computed.health === 'healthy'
   596→      ? 'All Systems Operational'
   597→      : computed.health === 'degraded'
   598→        ? 'Degraded Performance'
   599→        : 'Critical Load'
   600→
   601→  return (
   602→    <motion.div
   603→      variants={containerVariants}
   604→      initial="hidden"
   605→      animate="visible"
   606→      className="space-y-6"
   607→    >
   608→      {/* ====================== HERO HEADER ====================== */}
   609→      <motion.div variants={itemVariants}>
   610→        <HeroHeader
   611→          health={computed.health}
   612→          healthLabel={healthLabel}
   613→          loadScore={computed.loadScore}
   614→          loadLevel={computed.loadLevel}
   615→          uptimePercent={computed.uptimePercent}
   616→          uptimeDuration={computed.uptimeDuration}
   617→          lastUpdated={lastUpdated}
   618→          onRefresh={onRefresh}
   619→          refreshing={refreshing}
   620→        />
   621→      </motion.div>
   622→
   623→      {/* ====================== KPI CARDS ====================== */}
   624→      <motion.div
   625→        variants={itemVariants}
   626→        className="grid gap-4 md:grid-cols-2 lg:grid-cols-4"
   627→      >
   628→        <KPICard
   629→          title="Active Connections"
   630→          value={computed.connections}
   631→          icon={<Users className="h-4 w-4" />}
