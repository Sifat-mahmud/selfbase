   195→/**
   196→ * Render a type-aware editor cell for inline editing.
   197→ * - BOOLEAN  -> Switch
   198→ * - INTEGER/DECIMAL -> number Input
   199→ * - JSON/TIMESTAMP -> wider Input
   200→ * - TEXT/default  -> Input
   201→ */
   202→function renderCellInput(
   203→  col: ColumnItem,
   204→  value: unknown,
   205→  onChange: (v: unknown) => void,
   206→  disabled?: boolean,
   207→) {
   208→  const typeUpper = col.type.toUpperCase()
   209→  if (typeUpper === 'BOOLEAN') {
   210→    return (
   211→      <Switch
   212→        checked={Boolean(value)}
   213→        onCheckedChange={(v) => onChange(v)}
   214→        disabled={disabled}
   215→        aria-label={col.name}
   216→      />
   217→    )
   218→  }
   219→  if (typeUpper === 'INTEGER' || typeUpper === 'DECIMAL') {
   220→    return (
   221→      <Input
   222→        type="number"
   223→        value={value === null || value === undefined ? '' : String(value)}
   224→        onChange={(e) => {
   225→          const raw = e.target.value
   226→          if (raw === '') {
   227→            onChange(null)
   228→          } else if (typeUpper === 'INTEGER') {
   229→            const n = parseInt(raw, 10)
   230→            onChange(Number.isNaN(n) ? null : n)
   231→          } else {
   232→            const n = parseFloat(raw)
   233→            onChange(Number.isNaN(n) ? null : n)
   234→          }
   235→        }}
   236→        disabled={disabled}
   237→        className="h-8 font-mono text-xs min-w-[80px]"
   238→        aria-label={col.name}
   239→      />
   240→    )
   241→  }
   242→  if (typeUpper === 'JSON' || typeUpper === 'TIMESTAMP') {
   243→    return (
   244→      <Input
   245→        value={value === null || value === undefined ? '' : String(value)}
   246→        onChange={(e) => onChange(e.target.value)}
   247→        disabled={disabled}
   248→        className="h-8 font-mono text-xs min-w-[140px]"
   249→        placeholder={typeUpper === 'JSON' ? '{...}' : '2024-01-01T00:00:00Z'}
   250→        aria-label={col.name}
   251→      />
   252→    )
   253→  }
   254→  return (
   255→    <Input
   256→      value={value === null || value === undefined ? '' : String(value)}
   257→      onChange={(e) => onChange(e.target.value)}
   258→      disabled={disabled}
   259→      className="h-8 font-mono text-xs min-w-[100px]"
   260→      aria-label={col.name}
   261→    />
   262→  )
   263→}

   788→  function parseCsv(text: string): { headers: string[]; rows: Record<string, string>[] } {
   789→    const lines = text.split(/\r?\n/).filter((l) => l.trim())
   790→    if (lines.length === 0) return { headers: [], rows: [] }
   791→    // Simple CSV parser (handles quoted fields)
   792→    function splitLine(line: string): string[] {
   793→      const result: string[] = []
   794→      let current = ''
   795→      let inQuotes = false
   796→      for (let i = 0; i < line.length; i++) {
   797→        const ch = line[i]
   798→        if (inQuotes) {
   799→          if (ch === '"' && line[i + 1] === '"') { current += '"'; i++ }
   800→          else if (ch === '"') { inQuotes = false }
   801→          else { current += ch }
   802→        } else {
   803→          if (ch === '"') { inQuotes = true }
   804→          else if (ch === ',') { result.push(current); current = '' }
   805→          else { current += ch }
   806→        }
   807→      }
   808→      result.push(current)
   809→      return result
   810→    }
   811→    const headers = splitLine(lines[0])
   812→    const rows = lines.slice(1).map((line) => {
   813→      const vals = splitLine(line)
   814→      const obj: Record<string, string> = {}
   815→      headers.forEach((h, i) => { obj[h] = vals[i] ?? '' })
   816→      return obj
   817→    })
   818→    return { headers, rows }
   819→  }
