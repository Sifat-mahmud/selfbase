   216→export function PipelineView() {
   217→  const { toast } = useToast()
   218→  const [pipelines, setPipelines] = useState<PipelineSourceItem[]>([])
   219→  const [allRuns, setAllRuns] = useState<PipelineRunItem[]>([])
   220→  const [loading, setLoading] = useState(true)
   221→  const [search, setSearch] = useState('')
   222→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   223→  const [selectedPipeline, setSelectedPipeline] = useState<PipelineSourceItem | null>(null)
   224→  const [previewTarget, setPreviewTarget] = useState<PipelineSourceItem | null>(null)
   225→  const [previewResult, setPreviewResult] = useState<PreviewResult | null>(null)
   226→  const [previewLoading, setPreviewLoading] = useState(false)
   227→  const [runningId, setRunningId] = useState<string | null>(null)
   228→
   229→  // Create form
   230→  const [newName, setNewName] = useState('')
   231→  const [newDesc, setNewDesc] = useState('')
   232→  const [newType, setNewType] = useState<'rest' | 'rss' | 'websocket' | 'scraper'>('rest')
   233→  const [newUrl, setNewUrl] = useState('')
   234→  const [newMethod, setNewMethod] = useState('GET')
   235→  const [newHeaders, setNewHeaders] = useState('{}')
   236→  const [newJsonPath, setNewJsonPath] = useState('')
   237→  const [newInterval, setNewInterval] = useState('300')
   238→  const [newOnConflict, setNewOnConflict] = useState('update')
   239→  const [newTargetTableId, setNewTargetTableId] = useState('')
   240→  const [newPreRunAction, setNewPreRunAction] = useState('none')
   241→  const [newPrimaryKeyCols, setNewPrimaryKeyCols] = useState<string[]>([])
   242→  const [newMappings, setNewMappings] = useState<ColumnMapping[]>([])
   243→  const [tables, setTables] = useState<Array<{ id: string; name: string; displayName?: string | null; columns: Array<{ id: string; name: string; type: string }> }>>([])
   244→
   245→  const loadAll = useCallback(async () => {
   246→    setLoading(true)
   247→    try {
   248→      const [p, r, t] = await Promise.all([
   249→        apiGet<PipelineSourceItem[]>('/api/pipelines'),
   250→        apiGet<PipelineRunItem[] | PipelineRunsResponse>('/api/pipelines/runs?limit=100').catch(() => null),
   251→        apiGet<Array<{ id: string; name: string; displayName?: string | null; columns: Array<{ id: string; name: string; type: string }> }>>('/api/tables').catch(() => []),
   252→      ])
   253→      if (Array.isArray(t)) setTables(t)
   254→      const list = Array.isArray(p) ? p : []
   255→      // Map targetTable.name -> targetTableName for convenience, normalize columnMappings
   256→      const normalized = list.map((pl) => ({
   257→        ...pl,
   258→        targetTableName: pl.targetTableName ?? pl.targetTable?.name ?? null,
   259→        lastRun: pl.pipelineRuns && pl.pipelineRuns.length > 0 ? pl.pipelineRuns[0] : null,
   260→      }))
   261→      setPipelines(normalized)
   262→      // apiGet already unwraps `{ success, data }` envelopes, so the result is usually
   263→      // the raw array. Handle both shapes defensively.
   264→      const runs = Array.isArray(r) ? r : (r?.data ?? [])
   265→      setAllRuns(Array.isArray(runs) ? runs : [])
   266→    } catch (err) {
   267→      toast({
   268→        title: 'Failed to load pipelines',
   269→        description: err instanceof Error ? err.message : 'Unknown error',
   270→        variant: 'destructive',
   271→      })
   272→    } finally {
   273→      setLoading(false)
   274→    }
   275→  }, [toast])
   276→
   277→  useEffect(() => {
   278→    void loadAll()
   279→  }, [loadAll])
   280→
   281→  const handleCreate = async () => {
   282→    if (!newName.trim() || !newUrl.trim()) {
   283→      toast({ title: 'Name and URL required', variant: 'destructive' })
   284→      return
   285→    }
   286→    try {
   287→      const payload: Record<string, unknown> = {
   288→        name: newName.trim(),
   289→        description: newDesc || null,
   290→        sourceType: newType,
   291→        url: newUrl.trim(),
   292→        method: newMethod,
   293→        headers: newHeaders || null,
   294→        jsonPath: newJsonPath || null,
   295→        fetchInterval: Number(newInterval) || 300,
   296→        onConflict: newOnConflict,
   297→        targetTableId: newTargetTableId && newTargetTableId !== '_none' ? newTargetTableId : null,
   298→        preRunAction: newPreRunAction,
   299→        primaryKeyCols: JSON.stringify(newPrimaryKeyCols),
   300→        isActive: true,
   301→        columnMappings: JSON.stringify(newMappings),
   302→      }
   303→      const created = await apiPost<PipelineSourceItem>('/api/pipelines', payload)
   304→      setPipelines((prev) => [
   305→        {
   306→          ...created,
   307→          targetTableName: created.targetTable?.name ?? null,
   308→          lastRun: null,
   309→        },
   310→        ...prev,
   311→      ])
   312→      setShowCreateDialog(false)
   313→      setNewName('')
   314→      setNewDesc('')
   315→      setNewUrl('')
   316→      setNewMethod('GET')
   317→      setNewHeaders('{}')
   318→      setNewJsonPath('')
   319→      setNewInterval('300')
   320→      setNewOnConflict('update')
   321→      setNewTargetTableId('')
   322→      setNewPreRunAction('none')
   323→      setNewPrimaryKeyCols([])
   324→      setNewMappings([])
   325→      toast({ title: 'Pipeline created', description: `"${created.name}" has been created.` })
   326→    } catch (err) {
   327→      toast({
   328→        title: 'Failed to create pipeline',
   329→        description: err instanceof Error ? err.message : 'Unknown error',
   330→        variant: 'destructive',
   331→      })
   332→    }
   333→  }
   334→
   335→  const handleToggleActive = async (pipeline: PipelineSourceItem) => {
   336→    try {
   337→      await apiPut(`/api/pipelines/${pipeline.id}`, { isActive: !pipeline.isActive })
   338→      setPipelines((prev) =>
   339→        prev.map((p) => (p.id === pipeline.id ? { ...p, isActive: !p.isActive } : p)),
   340→      )
   341→      toast({ title: `Pipeline ${pipeline.isActive ? 'paused' : 'activated'}` })
   342→    } catch (err) {
   343→      toast({
   344→        title: 'Failed to update pipeline',
   345→        description: err instanceof Error ? err.message : 'Unknown error',
   346→        variant: 'destructive',
   347→      })
   348→    }
   349→  }
   350→
   351→  const handleRunNow = async (pipeline: PipelineSourceItem) => {
   352→    setRunningId(pipeline.id)
   353→    try {
   354→      const result = await apiPost<{ runId: string; status: string; rowsFetched?: number; rowsWritten?: number; error?: string }>(
   355→        `/api/pipelines/${pipeline.id}/run`,
   356→        {},
   357→      )
   358→      toast({
   359→        title: result.status === 'success' ? 'Pipeline run completed' : 'Pipeline run finished',
   360→        description:
   361→          result.status === 'success'
   362→            ? `Fetched ${result.rowsFetched ?? 0} rows, wrote ${result.rowsWritten ?? 0}.`
   363→            : result.error ?? `Status: ${result.status}`,
   364→        variant: result.status === 'success' ? 'default' : 'destructive',
   365→      })
   366→      await loadAll()
   367→    } catch (err) {
   368→      toast({
   369→        title: 'Pipeline run failed',
   370→        description: err instanceof Error ? err.message : 'Unknown error',
   371→        variant: 'destructive',
   372→      })
   373→    } finally {
   374→      setRunningId(null)
   375→    }
   376→  }
   377→
   378→  const handlePreview = async (pipeline: PipelineSourceItem) => {
   379→    setPreviewTarget(pipeline)
   380→    setPreviewResult(null)
   381→    setPreviewLoading(true)
   382→    try {
   383→      const result = await apiPost<PreviewResult>(`/api/pipelines/${pipeline.id}/preview`, {})
   384→      setPreviewResult(result)
   385→    } catch (err) {
   386→      setPreviewResult({
   387→        success: false,
   388→        error: err instanceof Error ? err.message : 'Unknown error',
   389→      })
   390→    } finally {
   391→      setPreviewLoading(false)
   392→    }
   393→  }
   394→
   395→  const filteredPipelines = pipelines.filter(
   396→    (p) =>
   397→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   398→      (p.description ?? '').toLowerCase().includes(search.toLowerCase()) ||
   399→      p.url.toLowerCase().includes(search.toLowerCase()),
   400→  )
   401→
   402→  // Build chart data from allRuns grouped by day
   403→  const chartData = (() => {
   404→    const map = new Map<string, { day: string; success: number; failed: number; rows: number }>()
   405→    for (const r of allRuns) {
   406→      const d = new Date(r.startedAt)
   407→      const key = d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   408→      const entry = map.get(key) ?? { day: key, success: 0, failed: 0, rows: 0 }
   409→      if (r.status === 'success') entry.success += 1
   410→      if (r.status === 'failed' || r.status === 'timeout') entry.failed += 1
   411→      entry.rows += r.rowsWritten ?? 0
   412→      map.set(key, entry)
   413→    }
   414→    return Array.from(map.values()).slice(-7)
   415→  })()
   416→
   417→  // ----- Pipeline Analytics (Task 12) -----
   418→  // Most recent 20 runs, oldest first so the timeline reads left → right.
   419→  const timelineData = allRuns
   420→    .slice(0, 20)
   421→    .reverse()
   422→    .map((run) => {
   423→      const d = new Date(run.startedAt)
   424→      const time = d.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: false })
   425→      return {
   426→        time,
   427→        duration: run.durationMs ?? 0,
   428→        status: run.status,
   429→        rows: run.rowsWritten ?? 0,
   430→        name: run.source?.name ?? 'Pipeline',
   431→      }
   432→    })
   433→
   434→  // Status distribution for the donut chart.
   435→  const statusCounts = allRuns.reduce((acc, run) => {
   436→    acc[run.status] = (acc[run.status] || 0) + 1
   437→    return acc
   438→  }, {} as Record<string, number>)
   439→
   440→  const pieData = Object.entries(statusCounts).map(([name, value]) => ({
   441→    name,
   442→    value,
   443→    color: STATUS_COLORS[name] ?? STATUS_COLORS.pending,
   444→  }))
   445→
   446→  // Stat cards
   447→  const totalRuns = allRuns.length
   448→  const successCount = allRuns.filter((r) => r.status === 'success').length
   449→  const successRate = totalRuns > 0 ? Math.round((successCount / totalRuns) * 100) : 0
   450→  const completedRuns = allRuns.filter((r) => r.durationMs != null)
   451→  const avgDuration =
   452→    completedRuns.length > 0
   453→      ? Math.round(completedRuns.reduce((s, r) => s + (r.durationMs ?? 0), 0) / completedRuns.length)
   454→      : 0
   455→  const totalRowsWritten = allRuns.reduce((s, r) => s + (r.rowsWritten ?? 0), 0)
   456→
   457→  if (loading) {
   458→    return (
   459→      <div className="space-y-6">
   460→        <div className="flex justify-between">
   461→          <Skeleton className="h-8 w-48" />
   462→          <Skeleton className="h-9 w-36" />
   463→        </div>
   464→        <div className="grid gap-4 md:grid-cols-2">
   465→          {Array.from({ length: 4 }).map((_, i) => (
   466→            <Card key={i}>
   467→              <CardContent className="p-4">
   468→                <Skeleton className="h-32 w-full" />
   469→              </CardContent>
   470→            </Card>
   471→          ))}
   472→        </div>
   473→      </div>
   474→    )
   475→  }
   476→
   477→  if (selectedPipeline) {
   478→    const SourceIcon = sourceIcons[selectedPipeline.sourceType]
   479→    const lastRun = selectedPipeline.lastRun ?? selectedPipeline.pipelineRuns?.[0] ?? null
   480→    const LastStatusIcon = lastRun ? statusIcons[lastRun.status] : Clock
   481→    const mappings = parseJsonField<ColumnMapping[]>(selectedPipeline.columnMappings, [])
   482→    const sourceRuns = allRuns.filter((r) => r.sourceId === selectedPipeline.id)
   483→
   484→    return (
   485→      <motion.div
   486→        initial={{ opacity: 0, y: 8 }}
   487→        animate={{ opacity: 1, y: 0 }}
   488→        transition={{ duration: 0.2 }}
   489→        className="space-y-6"
   490→      >
   491→        <div className="flex items-center gap-2 flex-wrap">
   492→          <Button variant="ghost" size="sm" onClick={() => setSelectedPipeline(null)}>
   493→            <ArrowRight className="h-4 w-4 rotate-180" /> Back
   494→          </Button>
   495→          <Separator orientation="vertical" className="h-6" />
   496→          <div className="rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 p-2 text-white">
   497→            <SourceIcon className="h-5 w-5" />
   498→          </div>
   499→          <h1 className="text-2xl font-bold tracking-tight">{selectedPipeline.name}</h1>
   500→          <Badge variant="outline" className={`capitalize ${sourceTypeBadge[selectedPipeline.sourceType]}`}>
   501→            {selectedPipeline.sourceType}
   502→          </Badge>
   503→          <Badge variant={selectedPipeline.isActive ? 'default' : 'secondary'}>
   504→            {selectedPipeline.isActive ? 'Active' : 'Paused'}
   505→          </Badge>
   506→          <div className="ml-auto flex gap-2">
   507→            <Button variant="outline" size="sm" onClick={() => void handlePreview(selectedPipeline)}>
   508→              <Eye className="mr-1 h-3.5 w-3.5" /> Preview
   509→            </Button>
   510→            <Button
   511→              size="sm"
   512→              onClick={() => void handleRunNow(selectedPipeline)}
   513→              disabled={runningId === selectedPipeline.id}
   514→            >
   515→              <Play className="mr-1 h-3.5 w-3.5" />
   516→              {runningId === selectedPipeline.id ? 'Running...' : 'Run Now'}
   517→            </Button>
   518→          </div>
   519→        </div>
   520→
   521→        <div className="grid gap-4 md:grid-cols-4">
   522→          <Card>
   523→            <CardContent className="pt-4">
   524→              <div className="text-sm text-muted-foreground">Source Type</div>
   525→              <div className="text-lg font-bold capitalize">{selectedPipeline.sourceType}</div>
   526→            </CardContent>
   527→          </Card>
   528→          <Card>
   529→            <CardContent className="pt-4">
   530→              <div className="text-sm text-muted-foreground">Fetch Interval</div>
   531→              <div className="text-lg font-bold">
   532→                {selectedPipeline.fetchInterval > 0 ? `${selectedPipeline.fetchInterval}s` : 'Real-time'}
   533→              </div>
   534→            </CardContent>
   535→          </Card>
   536→          <Card>
   537→            <CardContent className="pt-4">
   538→              <div className="text-sm text-muted-foreground">Target Table</div>
   539→              <div className="text-lg font-bold font-mono">{selectedPipeline.targetTableName ?? '—'}</div>
   540→            </CardContent>
   541→          </Card>
   542→          <Card>
   543→            <CardContent className="pt-4">
   544→              <div className="text-sm text-muted-foreground">Last Run</div>
   545→              <div className="flex items-center gap-1">
   546→                {lastRun && <LastStatusIcon className={`h-4 w-4 ${statusColors[lastRun.status]}`} />}
   547→                <span className="text-lg font-bold capitalize">{lastRun?.status ?? 'Never'}</span>
   548→              </div>
   549→              {lastRun?.durationMs != null && (
   550→                <p className="text-xs text-muted-foreground mt-0.5">{lastRun.durationMs}ms · {lastRun.rowsWritten ?? 0} rows written</p>
   551→              )}
   552→            </CardContent>
   553→          </Card>
   554→        </div>
   555→
   556→        <Card>
   557→          <CardHeader>
   558→            <CardTitle className="text-base">Source URL & Configuration</CardTitle>
