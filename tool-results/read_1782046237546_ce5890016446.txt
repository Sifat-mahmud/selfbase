    84→interface ColumnMapping {
    85→  src: string
    86→  target: string
    87→  type?: string
    88→  transform?: string
    89→}
    90→
    91→interface PipelineRunItem {
    92→  id: string
    93→  sourceId: string
    94→  status: 'pending' | 'running' | 'success' | 'failed' | 'timeout'
    95→  startedAt: string
    96→  completedAt: string | null
    97→  durationMs: number | null
    98→  rowsFetched: number
    99→  rowsWritten: number
   100→  rowsFailed: number
   101→  errorPayload?: string | null
   102→  isManual: boolean
   103→  source?: { id: string; name: string; sourceType: string; url: string }
   104→}
   105→
   106→interface PipelineSourceItem {
   107→  id: string
   108→  name: string
   109→  description: string | null
   110→  sourceType: 'rest' | 'rss' | 'websocket' | 'scraper'
   111→  url: string
   112→  method: string
   113→  headers?: string | null
   114→  isActive: boolean
   115→  fetchInterval: number
   116→  jsonPath?: string | null
   117→  onConflict?: string
   118→  targetTableId?: string | null
   119→  targetTableName?: string | null
   120→  targetTable?: { name: string } | null
   121→  columnMappings: string
   122→  lastRun?: PipelineRunItem | null
   123→  pipelineRuns?: PipelineRunItem[]
   124→  createdAt: string
   125→}
   126→
   127→interface PipelineRunsResponse {
   128→  data?: PipelineRunItem[]
   129→  meta?: { page: number; limit: number; total: number }
   130→}
   131→
   132→interface PreviewResult {
   133→  success?: boolean
   134→  url?: string
   135→  method?: string
   136→  statusCode?: number
   137→  durationMs?: number
   138→  totalRows?: number
   139→  previewRows?: Array<Record<string, unknown>>
   140→  columns?: Array<{ name: string; type: string }>
   141→  error?: string
   142→}

   182→export function PipelineView() {
   183→  const { toast } = useToast()
   184→  const [pipelines, setPipelines] = useState<PipelineSourceItem[]>([])
   185→  const [allRuns, setAllRuns] = useState<PipelineRunItem[]>([])
   186→  const [loading, setLoading] = useState(true)
   187→  const [search, setSearch] = useState('')
   188→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   189→  const [selectedPipeline, setSelectedPipeline] = useState<PipelineSourceItem | null>(null)
   190→  const [previewTarget, setPreviewTarget] = useState<PipelineSourceItem | null>(null)
   191→  const [previewResult, setPreviewResult] = useState<PreviewResult | null>(null)
   192→  const [previewLoading, setPreviewLoading] = useState(false)
   193→  const [runningId, setRunningId] = useState<string | null>(null)
   194→
   195→  // Create form
   196→  const [newName, setNewName] = useState('')
   197→  const [newDesc, setNewDesc] = useState('')
   198→  const [newType, setNewType] = useState<'rest' | 'rss' | 'websocket' | 'scraper'>('rest')
   199→  const [newUrl, setNewUrl] = useState('')
   200→  const [newMethod, setNewMethod] = useState('GET')
   201→  const [newHeaders, setNewHeaders] = useState('{}')
   202→  const [newJsonPath, setNewJsonPath] = useState('')
   203→  const [newInterval, setNewInterval] = useState('300')
   204→  const [newOnConflict, setNewOnConflict] = useState('update')
   205→  const [newMappings, setNewMappings] = useState<ColumnMapping[]>([])
   206→
   207→  const loadAll = useCallback(async () => {
   208→    setLoading(true)
   209→    try {
   210→      const [p, r] = await Promise.all([
   211→        apiGet<PipelineSourceItem[]>('/api/pipelines'),
   212→        apiGet<PipelineRunsResponse>('/api/pipelines/runs?limit=100').catch(() => null),
   213→      ])
   214→      const list = Array.isArray(p) ? p : []
   215→      // Map targetTable.name -> targetTableName for convenience, normalize columnMappings
   216→      const normalized = list.map((pl) => ({
   217→        ...pl,
   218→        targetTableName: pl.targetTableName ?? pl.targetTable?.name ?? null,
   219→        lastRun: pl.pipelineRuns && pl.pipelineRuns.length > 0 ? pl.pipelineRuns[0] : null,
   220→      }))
   221→      setPipelines(normalized)
   222→      const runs = r?.data ?? []
   223→      setAllRuns(Array.isArray(runs) ? runs : [])
   224→    } catch (err) {
   225→      toast({
   226→        title: 'Failed to load pipelines',
   227→        description: err instanceof Error ? err.message : 'Unknown error',
   228→        variant: 'destructive',
   229→      })
   230→    } finally {
   231→      setLoading(false)
   232→    }
   233→  }, [toast])
   234→
   235→  useEffect(() => {
   236→    void loadAll()
   237→  }, [loadAll])
   238→
   239→  const handleCreate = async () => {
   240→    if (!newName.trim() || !newUrl.trim()) {
   241→      toast({ title: 'Name and URL required', variant: 'destructive' })
   242→      return
   243→    }
   244→    try {
   245→      const payload: Record<string, unknown> = {
   246→        name: newName.trim(),
   247→        description: newDesc || null,
   248→        sourceType: newType,
   249→        url: newUrl.trim(),
   250→        method: newMethod,
   251→        headers: newHeaders || null,
   252→        jsonPath: newJsonPath || null,
   253→        fetchInterval: Number(newInterval) || 300,
   254→        onConflict: newOnConflict,
   255→        isActive: true,
   256→        columnMappings: JSON.stringify(newMappings),
   257→      }
   258→      const created = await apiPost<PipelineSourceItem>('/api/pipelines', payload)
   259→      setPipelines((prev) => [
   260→        {
   261→          ...created,
   262→          targetTableName: created.targetTable?.name ?? null,
   263→          lastRun: null,
   264→        },
   265→        ...prev,
   266→      ])
   267→      setShowCreateDialog(false)
   268→      setNewName('')
   269→      setNewDesc('')
   270→      setNewUrl('')
   271→      setNewMethod('GET')
   272→      setNewHeaders('{}')
   273→      setNewJsonPath('')
   274→      setNewInterval('300')
   275→      setNewOnConflict('update')
   276→      setNewMappings([])
   277→      toast({ title: 'Pipeline created', description: `"${created.name}" has been created.` })
   278→    } catch (err) {
   279→      toast({
   280→        title: 'Failed to create pipeline',
   281→        description: err instanceof Error ? err.message : 'Unknown error',
   282→        variant: 'destructive',
   283→      })
   284→    }
   285→  }
   286→
   287→  const handleToggleActive = async (pipeline: PipelineSourceItem) => {
   288→    try {
   289→      await apiPut(`/api/pipelines/${pipeline.id}`, { isActive: !pipeline.isActive })
   290→      setPipelines((prev) =>
   291→        prev.map((p) => (p.id === pipeline.id ? { ...p, isActive: !p.isActive } : p)),
   292→      )
   293→      toast({ title: `Pipeline ${pipeline.isActive ? 'paused' : 'activated'}` })
   294→    } catch (err) {
   295→      toast({
   296→        title: 'Failed to update pipeline',
   297→        description: err instanceof Error ? err.message : 'Unknown error',
   298→        variant: 'destructive',
   299→      })
   300→    }
   301→  }
   302→
   303→  const handleRunNow = async (pipeline: PipelineSourceItem) => {
   304→    setRunningId(pipeline.id)
   305→    try {
   306→      const result = await apiPost<{ runId: string; status: string; rowsFetched?: number; rowsWritten?: number; error?: string }>(
   307→        `/api/pipelines/${pipeline.id}/run`,
   308→        {},
   309→      )
   310→      toast({
   311→        title: result.status === 'success' ? 'Pipeline run completed' : 'Pipeline run finished',
   312→        description:
   313→          result.status === 'success'
   314→            ? `Fetched ${result.rowsFetched ?? 0} rows, wrote ${result.rowsWritten ?? 0}.`
   315→            : result.error ?? `Status: ${result.status}`,
   316→        variant: result.status === 'success' ? 'default' : 'destructive',
   317→      })
   318→      await loadAll()
   319→    } catch (err) {
   320→      toast({
   321→        title: 'Pipeline run failed',
   322→        description: err instanceof Error ? err.message : 'Unknown error',
   323→        variant: 'destructive',
   324→      })
   325→    } finally {
   326→      setRunningId(null)
   327→    }
   328→  }
   329→
   330→  const handlePreview = async (pipeline: PipelineSourceItem) => {
   331→    setPreviewTarget(pipeline)
   332→    setPreviewResult(null)
   333→    setPreviewLoading(true)
   334→    try {
   335→      const result = await apiPost<PreviewResult>(`/api/pipelines/${pipeline.id}/preview`, {})
   336→      setPreviewResult(result)
   337→    } catch (err) {
   338→      setPreviewResult({
   339→        success: false,
   340→        error: err instanceof Error ? err.message : 'Unknown error',
   341→      })
   342→    } finally {
   343→      setPreviewLoading(false)
   344→    }
   345→  }
   346→
   347→  const filteredPipelines = pipelines.filter(
   348→    (p) =>
   349→      p.name.toLowerCase().includes(search.toLowerCase()) ||
   350→      (p.description ?? '').toLowerCase().includes(search.toLowerCase()) ||
   351→      p.url.toLowerCase().includes(search.toLowerCase()),
   352→  )
   353→
   354→  // Build chart data from allRuns grouped by day
   355→  const chartData = (() => {
   356→    const map = new Map<string, { day: string; success: number; failed: number; rows: number }>()
   357→    for (const r of allRuns) {
   358→      const d = new Date(r.startedAt)
   359→      const key = d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   360→      const entry = map.get(key) ?? { day: key, success: 0, failed: 0, rows: 0 }
   361→      if (r.status === 'success') entry.success += 1
   362→      if (r.status === 'failed' || r.status === 'timeout') entry.failed += 1
   363→      entry.rows += r.rowsWritten ?? 0
   364→      map.set(key, entry)
   365→    }
   366→    return Array.from(map.values()).slice(-7)
   367→  })()
   368→
   369→  if (loading) {
   370→    return (
   371→      <div className="space-y-6">
   372→        <div className="flex justify-between">
   373→          <Skeleton className="h-8 w-48" />
   374→          <Skeleton className="h-9 w-36" />
   375→        </div>
   376→        <div className="grid gap-4 md:grid-cols-2">
   377→          {Array.from({ length: 4 }).map((_, i) => (
   378→            <Card key={i}>
   379→              <CardContent className="p-4">
   380→                <Skeleton className="h-32 w-full" />
   381→              </CardContent>
   382→            </Card>
   383→          ))}
   384→        </div>
   385→      </div>
   386→    )
   387→  }
   388→
   389→  if (selectedPipeline) {
   390→    const SourceIcon = sourceIcons[selectedPipeline.sourceType]
   391→    const lastRun = selectedPipeline.lastRun ?? selectedPipeline.pipelineRuns?.[0] ?? null
   392→    const LastStatusIcon = lastRun ? statusIcons[lastRun.status] : Clock
   393→    const mappings = parseJsonField<ColumnMapping[]>(selectedPipeline.columnMappings, [])
   394→    const sourceRuns = allRuns.filter((r) => r.sourceId === selectedPipeline.id)
   395→
   396→    return (
   397→      <motion.div
   398→        initial={{ opacity: 0, y: 8 }}
   399→        animate={{ opacity: 1, y: 0 }}
   400→        transition={{ duration: 0.2 }}
   401→        className="space-y-6"
   402→      >
   403→        <div className="flex items-center gap-2 flex-wrap">
   404→          <Button variant="ghost" size="sm" onClick={() => setSelectedPipeline(null)}>
   405→            <ArrowRight className="h-4 w-4 rotate-180" /> Back
   406→          </Button>
   407→          <Separator orientation="vertical" className="h-6" />
   408→          <div className="rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 p-2 text-white">
   409→            <SourceIcon className="h-5 w-5" />
   410→          </div>
   411→          <h1 className="text-2xl font-bold tracking-tight">{selectedPipeline.name}</h1>
   412→          <Badge variant="outline" className={`capitalize ${sourceTypeBadge[selectedPipeline.sourceType]}`}>
   413→            {selectedPipeline.sourceType}
   414→          </Badge>
   415→          <Badge variant={selectedPipeline.isActive ? 'default' : 'secondary'}>
   416→            {selectedPipeline.isActive ? 'Active' : 'Paused'}
   417→          </Badge>
   418→          <div className="ml-auto flex gap-2">
   419→            <Button variant="outline" size="sm" onClick={() => void handlePreview(selectedPipeline)}>
   420→              <Eye className="mr-1 h-3.5 w-3.5" /> Preview
   421→            </Button>
   422→            <Button
   423→              size="sm"
   424→              onClick={() => void handleRunNow(selectedPipeline)}
   425→              disabled={runningId === selectedPipeline.id}
   426→            >
   427→              <Play className="mr-1 h-3.5 w-3.5" />
   428→              {runningId === selectedPipeline.id ? 'Running...' : 'Run Now'}
   429→            </Button>
   430→          </div>
   431→        </div>
   432→
   433→        <div className="grid gap-4 md:grid-cols-4">
   434→          <Card>
   435→            <CardContent className="pt-4">
   436→              <div className="text-sm text-muted-foreground">Source Type</div>
   437→              <div className="text-lg font-bold capitalize">{selectedPipeline.sourceType}</div>
   438→            </CardContent>
   439→          </Card>
   440→          <Card>
   441→            <CardContent className="pt-4">
   442→              <div className="text-sm text-muted-foreground">Fetch Interval</div>
   443→              <div className="text-lg font-bold">
   444→                {selectedPipeline.fetchInterval > 0 ? `${selectedPipeline.fetchInterval}s` : 'Real-time'}
   445→              </div>
   446→            </CardContent>
   447→          </Card>
   448→          <Card>
   449→            <CardContent className="pt-4">
   450→              <div className="text-sm text-muted-foreground">Target Table</div>
   451→              <div className="text-lg font-bold font-mono">{selectedPipeline.targetTableName ?? '—'}</div>
   452→            </CardContent>
   453→          </Card>
   454→          <Card>
   455→            <CardContent className="pt-4">
   456→              <div className="text-sm text-muted-foreground">Last Run</div>
   457→              <div className="flex items-center gap-1">
   458→                {lastRun && <LastStatusIcon className={`h-4 w-4 ${statusColors[lastRun.status]}`} />}
   459→                <span className="text-lg font-bold capitalize">{lastRun?.status ?? 'Never'}</span>
   460→              </div>
   461→              {lastRun?.durationMs != null && (
   462→                <p className="text-xs text-muted-foreground mt-0.5">{lastRun.durationMs}ms · {lastRun.rowsWritten ?? 0} rows written</p>
   463→              )}
   464→            </CardContent>
   465→          </Card>
   466→        </div>
   467→
   468→        <Card>
   469→          <CardHeader>
   470→            <CardTitle className="text-base">Source URL & Configuration</CardTitle>
   471→            <CardDescription>Pipeline source endpoint and extraction settings</CardDescription>
   472→          </CardHeader>
   473→          <CardContent className="space-y-3">
   474→            <div className="flex items-center gap-2">
   475→              <Badge variant="outline" className="font-mono">
   476→                {selectedPipeline.method}
   477→              </Badge>
   478→              <code className="text-sm font-mono break-all">{selectedPipeline.url}</code>
   479→              <a
   480→                href={selectedPipeline.url}
   481→                target="_blank"
   482→                rel="noopener noreferrer"
   483→                className="ml-auto text-muted-foreground hover:text-emerald-600"
   484→              >
   485→                <ExternalLink className="h-3.5 w-3.5" />
   486→              </a>
   487→            </div>
   488→            {selectedPipeline.jsonPath && (
   489→              <div className="text-sm">
   490→                <span className="text-muted-foreground">JSONPath: </span>
   491→                <code className="font-mono">{selectedPipeline.jsonPath}</code>
   492→              </div>
   493→            )}
   494→            {selectedPipeline.onConflict && (
   495→              <div className="text-sm">
   496→                <span className="text-muted-foreground">On Conflict: </span>
   497→                <Badge variant="outline" className="capitalize">
   498→                  {selectedPipeline.onConflict}
   499→                </Badge>
   500→              </div>
   501→            )}
   502→          </CardContent>
   503→        </Card>
   504→
   505→        <Card>
   506→          <CardHeader>
   507→            <CardTitle className="text-base">Column Mappings</CardTitle>
   508→            <CardDescription>Source-to-target field mappings</CardDescription>
   509→          </CardHeader>
   510→          <CardContent>
   511→            {mappings.length === 0 ? (
   512→              <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   513→                <GitBranch className="h-10 w-10 mb-2 opacity-40" />
   514→                <p className="text-sm">No column mappings configured</p>
   515→              </div>
   516→            ) : (
   517→              <Table>
   518→                <TableHeader>
   519→                  <TableRow>
   520→                    <TableHead>Source Field</TableHead>
   521→                    <TableHead className="w-8" />
   522→                    <TableHead>Target Column</TableHead>
   523→                    <TableHead>Type</TableHead>
   524→                    <TableHead>Transform</TableHead>
   525→                  </TableRow>
   526→                </TableHeader>
   527→                <TableBody>
   528→                  {mappings.map((m, i) => (
   529→                    <TableRow key={i} className="hover:bg-muted/40">
   530→                      <TableCell className="font-mono text-sm">{m.src}</TableCell>
   531→                      <TableCell>
   532→                        <ArrowRight className="h-3.5 w-3.5 text-muted-foreground" />
   533→                      </TableCell>
   534→                      <TableCell className="font-mono text-sm font-medium">{m.target}</TableCell>
   535→                      <TableCell>
   536→                        {m.type && (
   537→                          <Badge variant="outline" className="font-mono text-xs">
   538→                            {m.type}
   539→                          </Badge>
   540→                        )}
   541→                      </TableCell>
   542→                      <TableCell className="text-sm text-muted-foreground">{m.transform ?? '—'}</TableCell>
   543→                    </TableRow>
   544→                  ))}
   545→                </TableBody>
   546→              </Table>
   547→            )}
   548→          </CardContent>
   549→        </Card>
   550→
   551→        <Card>
   552→          <CardHeader>
   553→            <CardTitle className="text-base">Run History</CardTitle>
   554→            <CardDescription>Recent pipeline executions ({sourceRuns.length} total)</CardDescription>
   555→          </CardHeader>
   556→          <CardContent>
   557→            {sourceRuns.length === 0 ? (
   558→              <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   559→                <Clock className="h-10 w-10 mb-2 opacity-40" />
   560→                <p className="text-sm">No runs yet</p>
   561→                <p className="text-xs mt-1">Click "Run Now" to trigger a pipeline execution.</p>
   562→              </div>
   563→            ) : (
   564→              <div className="rounded-md border max-h-96 overflow-y-auto">
   565→                <Table>
   566→                  <TableHeader className="sticky top-0 bg-background">
   567→                    <TableRow>
   568→                      <TableHead>Status</TableHead>
   569→                      <TableHead>Started</TableHead>
   570→                      <TableHead>Duration</TableHead>
