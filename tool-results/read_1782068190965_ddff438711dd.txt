   155→/**
   156→ * Render a type-aware editor cell for inline editing.
   157→ * - BOOLEAN  -> Switch
   158→ * - INTEGER/DECIMAL -> number Input
   159→ * - JSON/TIMESTAMP -> wider Input
   160→ * - TEXT/default  -> Input
   161→ */
   162→function renderCellInput(
   163→  col: ColumnItem,
   164→  value: unknown,
   165→  onChange: (v: unknown) => void,
   166→  disabled?: boolean,
   167→) {
   168→  const typeUpper = col.type.toUpperCase()
   169→  if (typeUpper === 'BOOLEAN') {
   170→    return (
   171→      <Switch
   172→        checked={Boolean(value)}
   173→        onCheckedChange={(v) => onChange(v)}
   174→        disabled={disabled}
   175→        aria-label={col.name}
   176→      />
   177→    )
   178→  }
   179→  if (typeUpper === 'INTEGER' || typeUpper === 'DECIMAL') {
   180→    return (
   181→      <Input
   182→        type="number"
   183→        value={value === null || value === undefined ? '' : String(value)}
   184→        onChange={(e) => {
   185→          const raw = e.target.value
   186→          if (raw === '') {
   187→            onChange(null)
   188→          } else if (typeUpper === 'INTEGER') {
   189→            const n = parseInt(raw, 10)
   190→            onChange(Number.isNaN(n) ? null : n)
   191→          } else {
   192→            const n = parseFloat(raw)
   193→            onChange(Number.isNaN(n) ? null : n)
   194→          }
   195→        }}
   196→        disabled={disabled}
   197→        className="h-8 font-mono text-xs min-w-[80px]"
   198→        aria-label={col.name}
   199→      />
   200→    )
   201→  }
   202→  if (typeUpper === 'JSON' || typeUpper === 'TIMESTAMP') {
   203→    return (
   204→      <Input
   205→        value={value === null || value === undefined ? '' : String(value)}
   206→        onChange={(e) => onChange(e.target.value)}
   207→        disabled={disabled}
   208→        className="h-8 font-mono text-xs min-w-[140px]"
   209→        placeholder={typeUpper === 'JSON' ? '{...}' : '2024-01-01T00:00:00Z'}
   210→        aria-label={col.name}
   211→      />
   212→    )
   213→  }
   214→  return (
   215→    <Input
   216→      value={value === null || value === undefined ? '' : String(value)}
   217→      onChange={(e) => onChange(e.target.value)}
   218→      disabled={disabled}
   219→      className="h-8 font-mono text-xs min-w-[100px]"
   220→      aria-label={col.name}
   221→    />
   222→  )
   223→}

   549→  async function handleBulkDelete() {
   550→    if (!selectedTable || selectedRowIds.size === 0) return
   551→    setBulkDeleting(true)
   552→    let okCount = 0
   553→    let failCount = 0
   554→    try {
   555→      const ids = Array.from(selectedRowIds)
   556→      await Promise.all(
   557→        ids.map(async (id) => {
   558→          try {
   559→            await apiDelete(`/api/tables/${selectedTable.id}/rows/${id}`)
   560→            okCount += 1
   561→          } catch {
   562→            failCount += 1
   563→          }
   564→        }),
   565→      )
   566→      if (okCount > 0) adjustRowCount(-okCount)
   567→      setSelectedRowIds(new Set())
   568→      await refreshRows()
   569→      if (failCount === 0) {
   570→        toast({
   571→          title: 'Rows deleted',
   572→          description: `${okCount} row${okCount !== 1 ? 's' : ''} removed`,
   573→        })
   574→      } else {
   575→        toast({
   576→          title: 'Bulk delete partial',
   577→          description: `${okCount} deleted, ${failCount} failed`,
   578→          variant: 'destructive',
   579→        })
   580→      }
   581→    } finally {
   582→      setBulkDeleting(false)
   583→      setBulkDeleteOpen(false)
   584→    }
   585→  }

   635→        av = JSON.parse(a.data)[sortColumn]

   637→        av = undefined
   638→      }

   640→        bv = JSON.parse(b.data)[sortColumn]

   642→        bv = undefined
   643→      }

   645→      if (bv === null || bv === undefined) return -1
   646→      if (typeof av === 'number' && typeof bv === 'number') {
   647→        return sortDirection === 'asc' ? av - bv : bv - av
   648→      }
   649→      const cmp = String(av).localeCompare(String(bv))
   650→      return sortDirection === 'asc' ? cmp : -cmp
   651→    })
