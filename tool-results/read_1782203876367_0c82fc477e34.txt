    93→function readFromLocalStorage<T>(key: string, fallback: T): T {
    94→  if (typeof window === 'undefined') return fallback
    95→  try {
    96→    const raw = localStorage.getItem(key)
    97→    return raw ? JSON.parse(raw) : fallback
    98→  } catch { return fallback }
    99→}
   100→
   101→function writeToLocalStorage(key: string, value: unknown) {
   102→  if (typeof window === 'undefined') return
   103→  try {
   104→    if (value === null || value === undefined) {
   105→      localStorage.removeItem(key)
   106→    } else {
   107→      localStorage.setItem(key, JSON.stringify(value))
   108→    }
   109→  } catch { /* ignore */ }
   110→}

   889→function KvEditor({ rows, onChange, keyPlaceholder = 'key', valuePlaceholder = 'value' }: {
   890→  rows: KeyValueRow[]; onChange: (rows: KeyValueRow[]) => void; keyPlaceholder?: string; valuePlaceholder?: string
   891→}) {
   892→  const updateRow = (id: string, field: 'key' | 'value', val: string) => onChange(rows.map(r => r.id === id ? { ...r, [field]: val } : r))
   893→  const removeRow = (id: string) => onChange(rows.filter(r => r.id !== id))
   894→  const addRow = () => onChange([...rows, { id: newRowId(), key: '', value: '' }])
   895→
   896→  return (
   897→    <div className="space-y-2">
   898→      {rows.map(row => (
   899→        <div key={row.id} className="flex items-center gap-2">
   900→          <Input value={row.key} onChange={e => updateRow(row.id, 'key', e.target.value)} placeholder={keyPlaceholder} className="h-8 font-mono text-xs flex-1" />
   901→          <span className="text-muted-foreground text-xs">:</span>
   902→          <Input value={row.value} onChange={e => updateRow(row.id, 'value', e.target.value)} placeholder={valuePlaceholder} className="h-8 font-mono text-xs flex-1" />
   903→          <Button variant="ghost" size="icon" className="h-8 w-8 shrink-0 text-muted-foreground hover:text-destructive" onClick={() => removeRow(row.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   904→        </div>
   905→      ))}
   906→      <Button variant="outline" size="sm" className="h-7 text-xs gap-1" onClick={addRow}><Plus className="h-3 w-3" />Add row</Button>
   907→    </div>
   908→  )
   909→}

   915→function AuthFlowSection({ onTokenObtained, initialToken, initialTokenInfo }: {
   916→  onTokenObtained: (token: string) => void
   917→  initialToken: string | null
   918→  initialTokenInfo: { expiresAt: string; permissions: string[]; app: { name: string } } | null
   919→}) {
   920→  const { toast } = useToast()
   921→  const [apiKeys, setApiKeys] = useState<ApiKeyInfo[]>([])
   922→  const [creating, setCreating] = useState(false)
   923→  const [newKeyName, setNewKeyName] = useState('')
   924→  const [newKeyPerms, setNewKeyPerms] = useState('read,write')
   925→  const [createdKey, setCreatedKey] = useState<string | null>(null)
   926→  const [showCreatedKey, setShowCreatedKey] = useState(false)
   927→  const [activeToken, setActiveToken] = useState<string | null>(initialToken)
   928→  const [tokenInfo, setTokenInfo] = useState<{ expiresAt: string; permissions: string[]; app: { name: string } } | null>(initialTokenInfo)
   929→  const [loginLoading, setLoginLoading] = useState(false)
   930→  const [selectedApiKey, setSelectedApiKey] = useState<string>(readFromLocalStorage<string>(STORAGE_KEY_API_KEY, ''))
   931→
   932→  useEffect(() => { loadApiKeys() }, [])
   933→
   934→  async function loadApiKeys() {
   935→    try {
   936→      const data = await apiGet<{ data: ApiKeyInfo[] }>('/api/api-keys')
   937→      const list = Array.isArray(data) ? data : (data as any)?.data ?? []
   938→      setApiKeys(list)
   939→    } catch { /* ignore */ }
   940→  }
   941→
   942→  async function createApiKey() {
   943→    if (!newKeyName.trim()) return
   944→    setCreating(true)
   945→    try {
   946→      const result = await apiPost<{ success: boolean; apiKey: { key: string } }>('/api/api-keys', {
   947→        name: newKeyName.trim(),
   948→        permissions: newKeyPerms,
   949→      })
   950→      if ((result as any).apiKey?.key) {
   951→        setCreatedKey((result as any).apiKey.key)
   952→        setShowCreatedKey(true)
   953→        setNewKeyName('')
   954→        await loadApiKeys()
   955→        toast({ title: 'API Key created', description: 'Copy the key now — it won\'t be shown again!' })
   956→      }
   957→    } catch (err) {
   958→      toast({ title: 'Failed to create key', variant: 'destructive' })
   959→    } finally {
   960→      setCreating(false)
   961→    }
   962→  }
   963→
   964→  async function revokeKey(id: string) {
   965→    try {
   966→      await apiDelete(`/api/api-keys/${id}`)
   967→      await loadApiKeys()
   968→      toast({ title: 'API Key revoked' })
   969→    } catch {
   970→      toast({ title: 'Failed to revoke key', variant: 'destructive' })
   971→    }
   972→  }
   973→
   974→  async function loginWithApiKey() {
   975→    if (!selectedApiKey) {
   976→      toast({ title: 'Select an API key first', variant: 'destructive' })
   977→      return
   978→    }
   979→    setLoginLoading(true)
   980→    try {
   981→      const res = await fetch('/api/v1/auth/login', {
   982→        method: 'POST',
   983→        headers: { Authorization: `Bearer ${selectedApiKey}` },
   984→      })
   985→      const data = await res.json()
   986→      if (res.ok && data.token) {
   987→        setActiveToken(data.token)
   988→        const info = {
   989→          expiresAt: data.expiresAt,
   990→          permissions: data.permissions,
   991→          app: data.app,
   992→        }
   993→        setTokenInfo(info)
   994→        writeToLocalStorage(STORAGE_KEY_TOKEN, data.token)
   995→        writeToLocalStorage(STORAGE_KEY_TOKEN_INFO, info)
   996→        writeToLocalStorage(STORAGE_KEY_API_KEY, selectedApiKey)
   997→        onTokenObtained(data.token)
   998→        toast({ title: 'Token obtained!', description: `Valid for ${Math.round(data.expiresIn / 60)} minutes` })
   999→      } else {
  1000→        toast({ title: 'Login failed', description: data.error || 'Invalid API key', variant: 'destructive' })
  1001→      }
  1002→    } catch {
  1003→      toast({ title: 'Login failed', variant: 'destructive' })
  1004→    } finally {
  1005→      setLoginLoading(false)
  1006→    }
  1007→  }
  1008→
  1009→  async function validateToken() {
  1010→    if (!activeToken) return
  1011→    try {
  1012→      const res = await fetch('/api/v1/auth/validate', {
  1013→        method: 'POST',
  1014→        headers: { Authorization: `Bearer ${activeToken}` },
  1015→      })
  1016→      const data = await res.json()
  1017→      if (data.valid) {
  1018→        toast({ title: 'Token is valid', description: `Expires: ${new Date(data.expiresAt).toLocaleTimeString()}` })
  1019→      } else {
  1020→        toast({ title: 'Token invalid', description: data.error, variant: 'destructive' })
  1021→        setActiveToken(null)
  1022→        setTokenInfo(null)
  1023→        writeToLocalStorage(STORAGE_KEY_TOKEN, null)
  1024→        writeToLocalStorage(STORAGE_KEY_TOKEN_INFO, null)
  1025→      }
  1026→    } catch {
  1027→      toast({ title: 'Validation failed', variant: 'destructive' })
  1028→    }
  1029→  }

  1041→      writeToLocalStorage(STORAGE_KEY_TOKEN_INFO, null)
  1042→      toast({ title: 'Token revoked' })
  1043→    } catch { /* ignore */ }
  1044→  }
