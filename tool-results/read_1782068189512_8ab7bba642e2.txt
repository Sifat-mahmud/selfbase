   171→   171→async function fetchConfigMap(): Promise<Record<string, unknown>> {
   172→   172→  try {
   173→   173→    const res = await fetch('/api/config?limit=200')
   174→   174→    if (!res.ok) return {}
   175→   175→    const body = await res.json()
   176→   176→    const list: Array<{ key: string; value: unknown }> = body?.data ?? body ?? []
   177→   177→    const map: Record<string, unknown> = {}
   178→   178→    for (const item of list) {
   179→   179→      map[item.key] = item.value
   180→   180→    }
   181→   181→    return map
   182→   182→  } catch {
   183→   183→    return {}
   184→   184→  }
   185→   185→}

   187→   187→function deriveInitial<T extends Record<string, unknown>>(
   188→   188→  keys: string[],
   189→   189→  configMap: Record<string, unknown>,
   190→   190→): T {
   191→   191→  const out = {} as Record<string, unknown>
   192→   192→  for (const k of keys) {
   193→   193→    const fallback = CONFIG_DEFAULTS[k]
   194→   194→    if (k in configMap && configMap[k] !== undefined && configMap[k] !== null && configMap[k] !== '') {
   195→   195→      out[k] = configMap[k]
   196→   196→    } else {
   197→   197→      out[k] = fallback
   198→   198→    }
   199→   199→  }
   200→   200→  return out as T
   201→   201→}

   203→   203→async function saveConfig(values: Record<string, unknown>): Promise<{ ok: boolean; saved: number }> {
   204→   204→  let saved = 0
   205→   205→  let failed = 0
   206→   206→  // Send sequentially to avoid race conditions on the upsert endpoint.
   207→   207→  for (const [key, value] of Object.entries(values)) {
   208→   208→    try {
   209→   209→      const res = await fetch('/api/config', {
   210→   210→        method: 'POST',
   211→   211→        headers: { 'Content-Type': 'application/json' },
   212→   212→        body: JSON.stringify({ key, value }),
   213→   213→      })
   214→   214→      if (res.ok) saved++
   215→   215→      else failed++
   216→   216→    } catch {
   217→   217→      failed++
   218→   218→    }
   219→   219→  }
   220→   220→  return { ok: failed === 0, saved }
   221→   221→}

   347→   347→function ValidationIndicator({ value, keyName }: { value: unknown; keyName: string }) {
   348→   348→  const defaults: Record<string, unknown> = CONFIG_DEFAULTS
   349→   349→  const isInsecureDefault =
   350→   350→    (keyName === 'security.corsOrigins' && value === '*') ||
   351→   351→    (keyName === 'security.ipWhitelist' && (value === '' || value == null)) ||
   352→   352→    (keyName === 'general.adminEmail' && value === 'admin@selfbase.local') ||
   353→   353→    (keyName === 'storage.cdnBaseUrl' && value === 'https://cdn.selfbase.local')
   354→   354→  const isMissing =
   355→   355→    value === '' || value === undefined || value === null
   356→   356→  const isDefault =
   357→   357→    keyName in defaults && value === defaults[keyName]
   358→   358→
   359→   359→  if (isMissing) {
   360→   360→    return (
   361→   361→      <Tooltip>
   362→   362→        <TooltipTrigger asChild>
   363→   363→          <XCircle className="h-3.5 w-3.5 text-red-500 shrink-0" />
   364→   364→        </TooltipTrigger>
   365→   365→        <TooltipContent>Required field is missing</TooltipContent>
   366→   366→      </Tooltip>
   367→   367→    )
   368→   368→  }
   369→   369→  if (isInsecureDefault) {
   370→   370→    return (
   371→   371→      <Tooltip>
   372→   372→        <TooltipTrigger asChild>
   373→   373→          <AlertTriangle className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   374→   374→        </TooltipTrigger>
   375→   375→        <TooltipContent>Using default or insecure value</TooltipContent>
   376→   376→      </Tooltip>
   377→   377→    )
   378→   378→  }
   379→   379→  if (isDefault) {
   380→   380→    return (
   381→   381→      <Tooltip>
   382→   382→        <TooltipTrigger asChild>
   383→   383→          <AlertTriangle className="h-3.5 w-3.5 text-amber-500 shrink-0" />
   384→   384→        </TooltipTrigger>
   385→   385→        <TooltipContent>Using default value</TooltipContent>
   386→   386→      </Tooltip>
   387→   387→    )
   388→   388→  }
   389→   389→  return (
   390→   390→    <Tooltip>
   391→   391→      <TooltipTrigger asChild>
   392→   392→        <CheckCircle2 className="h-3.5 w-3.5 text-emerald-500 shrink-0" />
   393→   393→      </TooltipTrigger>
   394→   394→      <TooltipContent>Properly configured</TooltipContent>
   395→   395→    </Tooltip>
   396→   396→  )
   397→   397→}

  1583→  1583→function formatUptime(ms: number): string {
  1584→  1584→  const seconds = Math.floor(ms / 1000)
  1585→  1585→  const d = Math.floor(seconds / 86400)
  1586→  1586→  const h = Math.floor((seconds % 86400) / 3600)
  1587→  1587→  const m = Math.floor((seconds % 3600) / 60)
  1588→  1588→  const parts: string[] = []
  1589→  1589→  if (d > 0) parts.push(`${d}d`)
  1590→  1590→  parts.push(`${h}h`)
  1591→  1591→  parts.push(`${m}m`)
  1592→  1592→  return parts.join(' ')
  1593→  1593→}

  1595→  1595→function formatBytes(kb: number): string {
  1596→  1596→  if (kb < 1024) return `${kb.toFixed(0)} KB`
  1597→  1597→  const mb = kb / 1024
  1598→  1598→  if (mb < 1024) return `${mb.toFixed(1)} MB`
  1599→  1599→  const gb = mb / 1024
  1600→  1600→  return `${gb.toFixed(2)} GB`
  1601→  1601→}
