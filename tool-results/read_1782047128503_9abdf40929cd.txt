   149→   149→/** Unwrap `{ success, data }` envelope; fall back to raw payload. */
   150→   150→async function fetchJson<T>(url: string): Promise<T | null> {
   151→   151→  try {
   152→   152→    const res = await fetch(url, { cache: 'no-store' })
   153→   153→    if (!res.ok) return null
   154→   154→    const json = await res.json()
   155→   155→    if (json && typeof json === 'object' && 'success' in json && 'data' in json) {
   156→   156→      return (json as { data: T }).data as T
   157→   157→    }
   158→   158→    return json as T
   159→   159→  } catch {
   160→   160→    return null
   161→   161→  }
   162→   162→}

   164→   164→function formatNumber(n: number, mode: 'compact' | 'full' = 'compact'): string {
   165→   165→  if (!Number.isFinite(n)) return '0'
   166→   166→  if (mode === 'full') return n.toLocaleString()
   167→   167→  if (Math.abs(n) >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`
   168→   168→  if (Math.abs(n) >= 1_000) return `${(n / 1_000).toFixed(1)}K`
   169→   169→  return n.toLocaleString()
   170→   170→}

   172→   172→function formatRelativeTime(dateStr: string): string {
   173→   173→  const now = Date.now()
   174→   174→  const then = new Date(dateStr).getTime()
   175→   175→  const diff = Math.max(0, now - then)
   176→   176→  const sec = Math.floor(diff / 1000)
   177→   177→  if (sec < 60) return `${sec}s ago`
   178→   178→  const min = Math.floor(sec / 60)
   179→   179→  if (min < 60) return `${min}m ago`
   180→   180→  const hr = Math.floor(min / 60)
   181→   181→  if (hr < 24) return `${hr}h ago`
   182→   182→  const day = Math.floor(hr / 24)
   183→   183→  if (day < 30) return `${day}d ago`
   184→   184→  const mo = Math.floor(day / 30)
   185→   185→  if (mo < 12) return `${mo}mo ago`
   186→   186→  return `${Math.floor(mo / 12)}y ago`
   187→   187→}

   189→   189→function formatDuration(ms: number): string {
   190→   190→  if (!Number.isFinite(ms) || ms <= 0) return '0m'
   191→   191→  const sec = Math.floor(ms / 1000)
   192→   192→  const days = Math.floor(sec / 86400)
   193→   193→  const hours = Math.floor((sec % 86400) / 3600)
   194→   194→  const mins = Math.floor((sec % 3600) / 60)
   195→   195→  if (days > 0) return `${days}d ${hours}h ${mins}m`
   196→   196→  if (hours > 0) return `${hours}h ${mins}m`
   197→   197→  return `${mins}m`
   198→   198→}

   229→   229→function AnimatedNumber({
   230→   230→  value,
   231→   231→  format = 'plain',
   232→   232→  duration = 800,
   233→   233→  className,
   234→   234→}: {
   235→   235→  value: number
   236→   236→  format?: 'plain' | 'compact' | 'percent'
   237→   237→  duration?: number
   238→   238→  className?: string
   239→   239→}) {
   240→   240→  const reduce = useReducedMotion()
   241→   241→  const [display, setDisplay] = useState(value)
   242→   242→  const rafRef = useRef<number | null>(null)
   243→   243→
   244→   244→  useEffect(() => {
   245→   245→    if (reduce) return // reduced-motion users see raw value, no state writes
   246→   246→    const start = display
   247→   247→    const delta = value - start
   248→   248→    if (delta === 0) return
   249→   249→    const startTs = performance.now()
   250→   250→    const tick = (now: number) => {
   251→   251→      const t = Math.min(1, (now - startTs) / duration)
   252→   252→      const eased = 1 - Math.pow(1 - t, 3)
   253→   253→      setDisplay(start + delta * eased)
   254→   254→      if (t < 1) {
   255→   255→        rafRef.current = requestAnimationFrame(tick)
   256→   256→      } else {
   257→   257→        setDisplay(value)
   258→   258→      }
   259→   259→    }
   260→   260→    rafRef.current = requestAnimationFrame(tick)
   261→   261→    return () => {
   262→   262→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   263→   263→    }
   264→   264→  }, [value, duration, reduce])
   265→   265→
   266→   266→  const shown = reduce ? value : display
   267→   267→  if (format === 'compact') {
   268→   268→    return <span className={className}>{formatNumber(Math.round(shown))}</span>
   269→   269→  }
   270→   270→  if (format === 'percent') {
   271→   271→    return <span className={className}>{shown.toFixed(1)}%</span>
   272→   272→  }
   273→   273→  return <span className={className}>{Math.round(shown).toLocaleString()}</span>
   274→   274→}

   355→   355→export function DashboardView() {

   387→   387→      tables: tables ?? [],

   389→   389→      functions: functions ?? [],
   390→   390→      queue: queue ?? null,

   399→   399→      await fetchAll()

   402→   402→    run()

   407→   407→      cancelled = true
   408→   408→      clearInterval(id)
   409→   409→    }
   410→   410→  }, [fetchAll])
   411→   411→
   412→   412→  const handleRefresh = useCallback(() => {

   414→   414→    fetchAll().finally(() => setRefreshing(false))

   416→   416→
   417→   417→  if (loading) {
   418→   418→    return <DashboardSkeleton />
   419→   419→  }
   420→   420→
   421→   421→  return <DashboardContent data={data} lastUpdated={lastUpdated} onRefresh={handleRefresh} refreshing={refreshing} />

   428→   428→function DashboardContent({
   429→   429→  data,
   430→   430→  lastUpdated,
   431→   431→  onRefresh,
   432→   432→  refreshing,
   433→   433→}: {
   434→   434→  data: DashboardData
   435→   435→  lastUpdated: Date | null
   436→   436→  onRefresh: () => void
   437→   437→  refreshing: boolean
   438→   438→}) {
   439→   439→  const reduce = useReducedMotion()
   440→   440→
   441→   441→  // ---- computed values -------------------------------------------------
   442→   442→  const computed = useMemo(() => {
   443→   443→    const load = data.load
   444→   444→    const loadScore = load?.loadScore ?? 0
   445→   445→    const cpuTotal = load?.cpu.total ?? 0
   446→   446→    const cpuScraper = load?.cpu.scraper ?? 0
   447→   447→    const cpuApi = load?.cpu.api ?? 0
   448→   448→    const cpuFunctions = load?.cpu.functions ?? 0
   449→   449→    const ramPercent = load?.memory.percent ?? 0
   450→   450→    const ramUsedMb = load?.memory.usedMb ?? 0
   451→   451→    const ramTotalMb = load?.memory.totalMb ?? 0
   452→   452→    const connections = load?.connections ?? 0
   453→   453→    const reqPerSec = load?.requestsPerSecond ?? 0
   454→   454→    const activePipelines = data.pipelines.length
   455→   455→    const activeFunctions = data.functions.filter((f) => f.isActive).length
   456→   456→
   457→   457→    const totalTables = data.tables.length
   458→   458→    const totalRows = data.tables.reduce((sum, t) => sum + (t.rowCount ?? 0), 0)
   459→   459→    const recentErrorsTotal = load?.recentErrors.total ?? 0
   460→   460→
   461→   461→    // error rate: errors in last hour vs estimated requests in last hour
   462→   462→    const estimatedHourlyRequests = reqPerSec * 3600
   463→   463→    const errorRate =
   464→   464→      estimatedHourlyRequests > 0
   465→   465→        ? Math.min((recentErrorsTotal / estimatedHourlyRequests) * 100, 100)
   466→   466→        : recentErrorsTotal > 0
   467→   467→          ? 100
   468→   468→          : 0
   469→   469→
   470→   470→    // uptime
   471→   471→    const uptimePercent = data.uptime?.uptimePercent ?? 100
   472→   472→    const heartbeatCount = data.uptime?.heartbeatCount ?? 0
   473→   473→
   474→   474→    // uptime duration — oldest heartbeat to now
   475→   475→    const oldestHeartbeat = data.heartbeats.length > 0
   476→   476→      ? data.heartbeats[data.heartbeats.length - 1]
   477→   477→      : null
   478→   478→    const uptimeDurationMs = oldestHeartbeat
   479→   479→      ? Date.now() - new Date(oldestHeartbeat.recordedAt).getTime()
   480→   480→      : 0
   481→   481→    const uptimeDuration = formatDuration(uptimeDurationMs)
   482→   482→
   483→   483→    // queue totals
   484→   484→    const queueStats = data.queue?.stats ?? []
   485→   485→    const queuedCount = queueStats.find((s) => s.status === 'queued')?.count ?? 0
   486→   486→    const processingCount = queueStats.find((s) => s.status === 'processing')?.count ?? 0
   487→   487→    const queueTotal = queueStats.reduce((sum, s) => sum + (s.count ?? 0), 0)
   488→   488→
   489→   489→    // direction (latest vs previous heartbeat)
   490→   490→    const hb = data.heartbeats
   491→   491→    const latest = hb[0] ?? null
   492→   492→    const prev = hb[1] ?? null
   493→   493→    const dir = (curr?: number | null, p?: number | null): 'up' | 'down' | 'neutral' => {
   494→   494→      if (curr == null || p == null || curr === p) return 'neutral'
   495→   495→      return curr > p ? 'up' : 'down'
   496→   496→    }
   497→   497→    const connectionsDir = dir(latest?.activeConnections, prev?.activeConnections)
   498→   498→    const reqDir = dir(latest?.reqPerSec, prev?.reqPerSec)
   499→   499→    const loadDir = dir(latest?.loadScore, prev?.loadScore)
   500→   500→    const cpuDir = dir(latest?.cpuTotal, prev?.cpuTotal)
   501→   501→
   502→   502→    // server health from load level
   503→   503→    // Only consider uptime a concern if there are enough heartbeats to make
   504→   504→    // the measurement meaningful (at least 60 = 1 hour of 1-min intervals)
   505→   505→    const uptimeIsConcern = uptimePercent < 95 && heartbeatCount > 60
   506→   506→    const health: 'healthy' | 'degraded' | 'down' =
   507→   507→      loadScore >= 85
   508→   508→        ? 'down'
   509→   509→        : loadScore >= 60 || uptimeIsConcern
   510→   510→          ? 'degraded'
