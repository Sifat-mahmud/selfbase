   235→let _rowIdCounter = 0
   236→function newRowId() { _rowIdCounter += 1; return `row-${Date.now()}-${_rowIdCounter}` }
   237→
   238→function toKv(arr: Array<{ key: string; value: string }> | undefined): KeyValueRow[] {
   239→  if (!arr || arr.length === 0) return [{ id: newRowId(), key: '', value: '' }]
   240→  return arr.map(r => ({ id: newRowId(), key: r.key, value: r.value }))
   241→}

   255→function StatusBadge({ status }: { status: number }) {
   256→  const firstDigit = String(status).charAt(0)
   257→  const colorMap: Record<string, string> = {
   258→    '2': 'bg-emerald-500/10 text-emerald-700 border-emerald-200 dark:text-emerald-400 dark:border-emerald-900',
   259→    '3': 'bg-blue-500/10 text-blue-700 border-blue-200 dark:text-blue-400 dark:border-blue-900',
   260→    '4': 'bg-amber-500/10 text-amber-700 border-amber-200 dark:text-amber-400 dark:border-amber-900',
   261→    '5': 'bg-red-500/10 text-red-700 border-red-200 dark:text-red-400 dark:border-red-900',
   262→  }
   263→  const cls = colorMap[firstDigit] ?? 'bg-muted text-muted-foreground border-border'
   264→  return <span className={`inline-flex items-center justify-center rounded-md border px-2 py-0.5 text-xs font-bold ${cls}`}>{status}</span>
   265→}

   274→function KvEditor({ rows, onChange, keyPlaceholder = 'key', valuePlaceholder = 'value' }: {
   275→  rows: KeyValueRow[]; onChange: (rows: KeyValueRow[]) => void; keyPlaceholder?: string; valuePlaceholder?: string
   276→}) {
   277→  const updateRow = (id: string, field: 'key' | 'value', val: string) => onChange(rows.map(r => r.id === id ? { ...r, [field]: val } : r))
   278→  const removeRow = (id: string) => onChange(rows.filter(r => r.id !== id))
   279→  const addRow = () => onChange([...rows, { id: newRowId(), key: '', value: '' }])
   280→
   281→  return (
   282→    <div className="space-y-2">
   283→      {rows.map(row => (
   284→        <div key={row.id} className="flex items-center gap-2">
   285→          <Input value={row.key} onChange={e => updateRow(row.id, 'key', e.target.value)} placeholder={keyPlaceholder} className="h-8 font-mono text-xs flex-1" />
   286→          <span className="text-muted-foreground text-xs">:</span>
   287→          <Input value={row.value} onChange={e => updateRow(row.id, 'value', e.target.value)} placeholder={valuePlaceholder} className="h-8 font-mono text-xs flex-1" />
   288→          <Button variant="ghost" size="icon" className="h-8 w-8 shrink-0 text-muted-foreground hover:text-destructive" onClick={() => removeRow(row.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   289→        </div>
   290→      ))}
   291→      <Button variant="outline" size="sm" className="h-7 text-xs gap-1" onClick={addRow}><Plus className="h-3 w-3" />Add row</Button>
   292→    </div>
   293→  )
   294→}

   300→function AuthFlowSection({ onTokenObtained }: { onTokenObtained: (token: string) => void }) {
   301→  const { toast } = useToast()
   302→  const [apiKeys, setApiKeys] = useState<ApiKeyInfo[]>([])
   303→  const [creating, setCreating] = useState(false)
   304→  const [newKeyName, setNewKeyName] = useState('')
   305→  const [newKeyPerms, setNewKeyPerms] = useState('read,write')
   306→  const [createdKey, setCreatedKey] = useState<string | null>(null)
   307→  const [showCreatedKey, setShowCreatedKey] = useState(false)
   308→  const [activeToken, setActiveToken] = useState<string | null>(null)
   309→  const [tokenInfo, setTokenInfo] = useState<{ expiresAt: string; permissions: string[]; app: { name: string } } | null>(null)
   310→  const [loginLoading, setLoginLoading] = useState(false)
   311→  const [selectedApiKey, setSelectedApiKey] = useState<string>('')
   312→
   313→  useEffect(() => { loadApiKeys() }, [])
   314→
   315→  async function loadApiKeys() {
   316→    try {
   317→      const data = await apiGet<{ data: ApiKeyInfo[] }>('/api/api-keys')
   318→      const list = Array.isArray(data) ? data : (data as any)?.data ?? []
   319→      setApiKeys(list)
   320→    } catch { /* ignore */ }
   321→  }
   322→
   323→  async function createApiKey() {
   324→    if (!newKeyName.trim()) return
   325→    setCreating(true)
   326→    try {
   327→      const result = await apiPost<{ success: boolean; apiKey: { key: string } }>('/api/api-keys', {
   328→        name: newKeyName.trim(),
   329→        permissions: newKeyPerms,
   330→      })
   331→      if ((result as any).apiKey?.key) {
   332→        setCreatedKey((result as any).apiKey.key)
   333→        setShowCreatedKey(true)
   334→        setNewKeyName('')
   335→        await loadApiKeys()
   336→        toast({ title: 'API Key created', description: 'Copy the key now — it won\'t be shown again!' })
   337→      }
   338→    } catch (err) {
   339→      toast({ title: 'Failed to create key', variant: 'destructive' })
   340→    } finally {
   341→      setCreating(false)
   342→    }
   343→  }
   344→
   345→  async function revokeKey(id: string) {
   346→    try {
   347→      await apiDelete(`/api/api-keys/${id}`)
   348→      await loadApiKeys()
   349→      toast({ title: 'API Key revoked' })
   350→    } catch {
   351→      toast({ title: 'Failed to revoke key', variant: 'destructive' })
   352→    }
   353→  }
   354→
   355→  async function loginWithApiKey() {
   356→    if (!selectedApiKey) {
   357→      toast({ title: 'Select an API key first', variant: 'destructive' })
   358→      return
   359→    }
   360→    setLoginLoading(true)
   361→    try {
   362→      const res = await fetch('/api/v1/auth/login', {
   363→        method: 'POST',
   364→        headers: { Authorization: `Bearer ${selectedApiKey}` },
   365→      })
   366→      const data = await res.json()
   367→      if (res.ok && data.token) {
   368→        setActiveToken(data.token)
   369→        setTokenInfo({
   370→          expiresAt: data.expiresAt,
   371→          permissions: data.permissions,
   372→          app: data.app,
   373→        })
   374→        onTokenObtained(data.token)
   375→        toast({ title: 'Token obtained!', description: `Valid for ${Math.round(data.expiresIn / 60)} minutes` })
   376→      } else {
   377→        toast({ title: 'Login failed', description: data.error || 'Invalid API key', variant: 'destructive' })
   378→      }
   379→    } catch {
   380→      toast({ title: 'Login failed', variant: 'destructive' })
   381→    } finally {
   382→      setLoginLoading(false)
   383→    }
   384→  }
   385→
   386→  async function validateToken() {
   387→    if (!activeToken) return
   388→    try {
   389→      const res = await fetch('/api/v1/auth/validate', {
   390→        method: 'POST',
   391→        headers: { Authorization: `Bearer ${activeToken}` },
   392→      })
   393→      const data = await res.json()
   394→      if (data.valid) {
   395→        toast({ title: 'Token is valid', description: `Expires: ${new Date(data.expiresAt).toLocaleTimeString()}` })
   396→      } else {
   397→        toast({ title: 'Token invalid', description: data.error, variant: 'destructive' })
   398→        setActiveToken(null)
   399→        setTokenInfo(null)
   400→      }
   401→    } catch {
   402→      toast({ title: 'Validation failed', variant: 'destructive' })
   403→    }
   404→  }
   405→
   406→  async function logoutToken() {
   407→    if (!activeToken) return
   408→    try {
   409→      await fetch('/api/v1/auth/logout', {
   410→        method: 'POST',
   411→        headers: { Authorization: `Bearer ${activeToken}` },
   412→      })
   413→      setActiveToken(null)
   414→      setTokenInfo(null)
   415→      toast({ title: 'Token revoked' })
   416→    } catch { /* ignore */ }
   417→  }
   418→

   422→      <Card className="border-2 border-emerald-200 dark:border-emerald-800 bg-gradient-to-br from-emerald-50/50 via-teal-50/30 to-background dark:from-emerald-950/20 dark:via-teal-950/10 dark:to-background">
   423→        <CardHeader className="pb-3">
   424→          <CardTitle className="text-base flex items-center gap-2">
   425→            <Shield className="h-5 w-5 text-emerald-600" />
   426→            API Authentication Flow
   427→          </CardTitle>
   428→          <CardDescription>
   429→            How external apps (iOS, Android, web) authenticate with SelfBase
   430→          </CardDescription>
   431→        </CardHeader>
   432→        <CardContent>
   433→          {/* Visual Flow Diagram */}
   434→          <div className="flex flex-col md:flex-row items-stretch gap-3">
   435→            {/* Step 1 */}
   436→            <div className="flex-1 rounded-lg border bg-background p-3 space-y-2">
   437→              <div className="flex items-center gap-2">
   438→                <div className="flex h-6 w-6 items-center justify-center rounded-full bg-emerald-500 text-white text-xs font-bold">1</div>
   439→                <span className="text-sm font-semibold">Generate API Key</span>
   440→              </div>
   441→              <div className="flex items-center gap-2">
   442→                <Smartphone className="h-4 w-4 text-muted-foreground" />
   443→                <p className="text-xs text-muted-foreground">Create a key for your app in the dashboard below</p>
   444→              </div>
   445→              <div className="rounded bg-muted p-2">
   446→                <code className="text-[10px] font-mono text-emerald-700 dark:text-emerald-400">
   447→                  sb_live_a1b2c3d4e5f6...
   448→                </code>
   449→              </div>
   450→            </div>
   451→
   452→            <div className="flex items-center justify-center">
   453→              <ArrowRight className="h-5 w-5 text-emerald-400 hidden md:block" />
   454→              <ChevronDown className="h-5 w-5 text-emerald-400 md:hidden" />
   455→            </div>

   714→  const [error, setError] = useState<string | null>(null)
   715→  const [search, setSearch] = useState('')
   716→  const [history, setHistory] = useState<HistoryEntry[]>([])
   717→
   718→  // Auth token for playground requests
   719→  const [authToken, setAuthToken] = useState<string | null>(null)
   720→
   721→  const filteredTemplates = useMemo(() => {
   722→    const q = search.trim().toLowerCase()
   723→    if (!q) return ENDPOINT_TEMPLATES
   724→    return ENDPOINT_TEMPLATES.filter(t =>
   725→      t.path.toLowerCase().includes(q) || t.description.toLowerCase().includes(q) ||
   726→      t.category.toLowerCase().includes(q) || t.method.toLowerCase().includes(q)
   727→    )
   728→  }, [search])
   729→

   732→    filteredTemplates.forEach(t => {
   733→      if (!map.has(t.category)) map.set(t.category, [])
   734→      map.get(t.category)!.push(t)

   736→    return Array.from(map.entries())

   741→    setMethod(template.method)
   742→    setUrl(template.path)
   743→    setParams(toKv(template.defaultParams))
   744→    // Auto-inject auth token into headers if template requires auth
   745→    const defaultHeaders = toKv(template.defaultHeaders)

   747→      const hasAuth = defaultHeaders.some(h => h.key.toLowerCase() === 'authorization')
   748→      if (!hasAuth) {

   768→    const startTime = performance.now()
   769→    try {
   770→      const urlObj = new URL(url, window.location.origin)

   772→

   777→
   778→      const res = await fetch(urlObj.toString(), {
   779→        method, headers: headerObj, body: hasBody && body ? body : undefined,
   780→      })
   781→
   782→      const responseText = await res.text()
   783→      const durationMs = Math.round(performance.now() - startTime)
   784→      const responseHeaders: Record<string, string> = {}
   785→      res.headers.forEach((v, k) => { responseHeaders[k] = v })
   786→
   787→      let responseBody = responseText
   788→      try { responseBody = JSON.stringify(JSON.parse(responseText), null, 2) } catch { /* keep raw */ }
   789→
   790→      const sizeBytes = new Blob([responseText]).size

   792→      setHistory(prev => [...prev, { status: res.status, durationMs, timestamp: Date.now(), ok: res.ok }].slice(-20))
