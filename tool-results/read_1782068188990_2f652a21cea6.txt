   176→   176→/** Unwrap `{ success, data }` envelope; fall back to raw payload. */
   177→   177→async function fetchJson<T>(url: string): Promise<T | null> {
   178→   178→  try {
   179→   179→    const res = await fetch(url, { cache: 'no-store' })
   180→   180→    if (!res.ok) return null
   181→   181→    const json = await res.json()
   182→   182→    if (json && typeof json === 'object' && 'success' in json && 'data' in json) {
   183→   183→      return (json as { data: T }).data as T
   184→   184→    }
   185→   185→    return json as T
   186→   186→  } catch {
   187→   187→    return null
   188→   188→  }
   189→   189→}

   191→   191→/** Normalize an API response that may be wrapped or have pagination into an array. */
   192→   192→function normalizeArray<T>(raw: T | null): T[] {
   193→   193→  if (!raw) return []
   194→   194→  if (Array.isArray(raw)) return raw
   195→   195→  if (typeof raw === 'object' && raw !== null) {
   196→   196→    const obj = raw as Record<string, unknown>
   197→   197→    // Handle { data: [...] } envelope
   198→   198→    if (Array.isArray(obj.data)) return obj.data as T[]
   199→   199→    // Handle { runs: [...] } envelope (functions/runs)
   200→   200→    if (Array.isArray(obj.runs)) return obj.runs as T[]
   201→   201→  }
   202→   202→  return []
   203→   203→}

   205→   205→function formatNumber(n: number, mode: 'compact' | 'full' = 'compact'): string {
   206→   206→  if (!Number.isFinite(n)) return '0'
   207→   207→  if (mode === 'full') return n.toLocaleString()
   208→   208→  if (Math.abs(n) >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`
   209→   209→  if (Math.abs(n) >= 1_000) return `${(n / 1_000).toFixed(1)}K`
   210→   210→  return n.toLocaleString()
   211→   211→}

   213→   213→function formatRelativeTime(dateStr: string): string {
   214→   214→  const now = Date.now()
   215→   215→  const then = new Date(dateStr).getTime()
   216→   216→  const diff = Math.max(0, now - then)
   217→   217→  const sec = Math.floor(diff / 1000)
   218→   218→  if (sec < 60) return `${sec}s ago`
   219→   219→  const min = Math.floor(sec / 60)
   220→   220→  if (min < 60) return `${min}m ago`
   221→   221→  const hr = Math.floor(min / 60)
   222→   222→  if (hr < 24) return `${hr}h ago`
   223→   223→  const day = Math.floor(hr / 24)
   224→   224→  if (day < 30) return `${day}d ago`
   225→   225→  const mo = Math.floor(day / 30)
   226→   226→  if (mo < 12) return `${mo}mo ago`
   227→   227→  return `${Math.floor(mo / 12)}y ago`
   228→   228→}

   230→   230→function formatDuration(ms: number): string {
   231→   231→  if (!Number.isFinite(ms) || ms <= 0) return '0m'
   232→   232→  const sec = Math.floor(ms / 1000)
   233→   233→  const days = Math.floor(sec / 86400)
   234→   234→  const hours = Math.floor((sec % 86400) / 3600)
   235→   235→  const mins = Math.floor((sec % 3600) / 60)
   236→   236→  if (days > 0) return `${days}d ${hours}h ${mins}m`
   237→   237→  if (hours > 0) return `${hours}h ${mins}m`
   238→   238→  return `${mins}m`
   239→   239→}

   270→   270→function AnimatedNumber({
   271→   271→  value,
   272→   272→  format = 'plain',
   273→   273→  duration = 800,
   274→   274→  className,
   275→   275→}: {
   276→   276→  value: number
   277→   277→  format?: 'plain' | 'compact' | 'percent'
   278→   278→  duration?: number
   279→   279→  className?: string
   280→   280→}) {
   281→   281→  const reduce = useReducedMotion()
   282→   282→  const [display, setDisplay] = useState(value)
   283→   283→  const rafRef = useRef<number | null>(null)
   284→   284→
   285→   285→  useEffect(() => {
   286→   286→    if (reduce) return // reduced-motion users see raw value, no state writes
   287→   287→    const start = display
   288→   288→    const delta = value - start
   289→   289→    if (delta === 0) return
   290→   290→    const startTs = performance.now()
   291→   291→    const tick = (now: number) => {
   292→   292→      const t = Math.min(1, (now - startTs) / duration)
   293→   293→      const eased = 1 - Math.pow(1 - t, 3)
   294→   294→      setDisplay(start + delta * eased)
   295→   295→      if (t < 1) {
   296→   296→        rafRef.current = requestAnimationFrame(tick)
   297→   297→      } else {
   298→   298→        setDisplay(value)
   299→   299→      }
   300→   300→    }
   301→   301→    rafRef.current = requestAnimationFrame(tick)
   302→   302→    return () => {
   303→   303→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   304→   304→    }
   305→   305→  }, [value, duration, reduce])
   306→   306→
   307→   307→  const shown = reduce ? value : display
   308→   308→  if (format === 'compact') {
   309→   309→    return <span className={className}>{formatNumber(Math.round(shown))}</span>
   310→   310→  }
   311→   311→  if (format === 'percent') {
   312→   312→    return <span className={className}>{shown.toFixed(1)}%</span>
   313→   313→  }
   314→   314→  return <span className={className}>{Math.round(shown).toLocaleString()}</span>
   315→   315→}

   322→   322→/** Build a smooth Catmull-Rom (cardinal) spline through the given points. */
   323→   323→function buildSplinePath(points: Array<{ x: number; y: number }>, tension = 0.18): string {
   324→   324→  if (points.length < 2) return ''
   325→   325→  let d = `M ${points[0].x.toFixed(2)} ${points[0].y.toFixed(2)}`
   326→   326→  for (let i = 0; i < points.length - 1; i++) {
   327→   327→    const p0 = points[i === 0 ? 0 : i - 1]
   328→   328→    const p1 = points[i]
   329→   329→    const p2 = points[i + 1]
   330→   330→    const p3 = points[i + 2 < points.length ? i + 2 : points.length - 1]
   331→   331→    const cp1x = p1.x + (p2.x - p0.x) * tension
   332→   332→    const cp1y = p1.y + (p2.y - p0.y) * tension
   333→   333→    const cp2x = p2.x - (p3.x - p1.x) * tension
   334→   334→    const cp2y = p2.y - (p3.y - p1.y) * tension
   335→   335→    d += ` C ${cp1x.toFixed(2)} ${cp1y.toFixed(2)}, ${cp2x.toFixed(2)} ${cp2y.toFixed(2)}, ${p2.x.toFixed(2)} ${p2.y.toFixed(2)}`
   336→   336→  }
   337→   337→  return d
   338→   338→}

   340→   340→function Sparkline({
   341→   341→  data,
   342→   342→  color,
   343→   343→  height = 36,
   344→   344→}: {
   345→   345→  data: number[]
   346→   346→  color: string
   347→   347→  height?: number
   348→   348→}) {
   349→   349→  const reduce = useReducedMotion()
   350→   350→  if (!data || data.length < 2) {
   351→   351→    return <div style={{ height }} className="w-full" aria-hidden />
   352→   352→  }
   353→   353→
   354→   354→  const width = 120 // viewBox width (scales to container via preserveAspectRatio)
   355→   355→  const pad = 3
   356→   356→  const innerW = width - pad * 2
   357→   357→  const innerH = height - pad * 2
   358→   358→  const min = Math.min(...data)
   359→   359→  const max = Math.max(...data)
   360→   360→  const range = max - min || 1
   361→   361→  const stepX = innerW / (data.length - 1)
   362→   362→
   363→   363→  const points = data.map((v, i) => ({
   364→   364→    x: pad + i * stepX,
   365→   365→    y: pad + innerH * (1 - (v - min) / range),
   366→   366→  }))
   367→   367→
   368→   368→  const linePath = buildSplinePath(points)
   369→   369→  const fillPath = `${linePath} L ${points[points.length - 1].x.toFixed(2)} ${height} L ${points[0].x.toFixed(2)} ${height} Z`
   370→   370→  const last = points[points.length - 1]
   371→   371→  const gradId = `spark-fill-${color.replace('#', '')}`
   372→   372→  const clipId = `spark-clip-${color.replace('#', '')}`
   373→   373→
   374→   374→  return (
   375→   375→    <svg
   376→   376→      viewBox={`0 0 ${width} ${height}`}
   377→   377→      width="100%"
   378→   378→      height={height}
   379→   379→      preserveAspectRatio="none"
   380→   380→      className="block overflow-visible"
   381→   381→      aria-hidden
   382→   382→    >
   383→   383→      <defs>
   384→   384→        <linearGradient id={gradId} x1="0" y1="0" x2="0" y2="1">
   385→   385→          <stop offset="0%" stopColor={color} stopOpacity={0.45} />
   386→   386→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   387→   387→        </linearGradient>
   388→   388→        <clipPath id={clipId}>
   389→   389→          <rect x="0" y="0" width={width} height={height} />
   390→   390→        </clipPath>
   391→   391→      </defs>
   392→   392→      <g clipPath={`url(#${clipId})`}>
   393→   393→        <motion.path
   394→   394→          d={fillPath}
   395→   395→          fill={`url(#${gradId})`}
   396→   396→          initial={{ opacity: 0 }}
   397→   397→          animate={{ opacity: 1 }}
   398→   398→          transition={{ duration: 0.6, ease: 'easeOut' }}
   399→   399→        />
   400→   400→        <motion.path
   401→   401→          d={linePath}
   402→   402→          fill="none"
   403→   403→          stroke={color}
   404→   404→          strokeWidth={1.5}
   405→   405→          strokeLinecap="round"
   406→   406→          strokeLinejoin="round"
   407→   407→          vectorEffect="non-scaling-stroke"
   408→   408→          initial={reduce ? false : { pathLength: 0 }}
   409→   409→          animate={{ pathLength: 1 }}
   410→   410→          transition={{ duration: 0.8, ease: 'easeOut' }}
   411→   411→        />
   412→   412→        {/* Last-point pulse dot */}
   413→   413→        <motion.circle
   414→   414→          cx={last.x}
   415→   415→          cy={last.y}
   416→   416→          r={2}
   417→   417→          fill={color}
   418→   418→          initial={reduce ? false : { scale: 0, opacity: 0 }}
   419→   419→          animate={{ scale: 1, opacity: 1 }}
   420→   420→          transition={{ duration: 0.3, delay: reduce ? 0 : 0.55 }}
   421→   421→          style={{ transformOrigin: `${last.x}px ${last.y}px` }}
   422→   422→        />
   423→   423→      </g>
   424→   424→    </svg>
   425→   425→  )
   426→   426→}

   477→   477→function ChangePill({
   478→   478→  direction,

   492→   492→      ? 'text-emerald-700 dark:text-emerald-400 bg-emerald-500/10'
   493→   493→      : 'text-red-700 dark:text-red-400 bg-red-500/10'
   494→   494→  const Icon = direction === 'up' ? ArrowUpRight : direction === 'down' ? ArrowDownRight : Minus
   495→   495→  return (
   496→   496→    <span
   497→   497→      className={`inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[11px] font-medium transition-colors ${colorClass}`}
   498→   498→    >
   499→   499→      <Icon className="h-3 w-3" />
   500→   500→      {label}

   563→   563→    }
   564→   564→    run()

   656→   656→      if (curr == null || p == null || curr === p) return 'neutral'

   666→   666→    // the measurement meaningful (at least 60 = 1 hour of 1-min intervals)
   667→   667→    const uptimeIsConcern = uptimePercent < 95 && heartbeatCount > 60

   669→   669→      loadScore >= 85

   673→   673→          : 'healthy'

   677→   677→    const sparkReq = hb.slice().reverse().map((h) => h.reqPerSec)

   699→   699→      totalRows,
   700→   700→      recentErrorsTotal,
   701→   701→      errorRate,
   702→   702→      uptimePercent,
   703→   703→      heartbeatCount,
   704→   704→      uptimeDuration,
   705→   705→      queuedCount,
   706→   706→      processingCount,
   707→   707→      queueTotal,
   708→   708→      connectionsDir,
   709→   709→      reqDir,
   710→   710→      loadDir,
   711→   711→      cpuDir,
   712→   712→      health,
   713→   713→      sparkConnections,
   714→   714→      sparkReq,
   715→   715→      sparkLoad,
   716→   716→      sparkCpu,
   717→   717→      sparkRam,
   718→   718→    }
   719→   719→  }, [data])
   720→   720→
   721→   721→  // chart data — reverse so oldest is first

   735→   735→        connections: h.activeConnections,
   736→   736→      }))
   737→   737→  }, [data.heartbeats, data.load])
   738→   738→
   739→   739→  // framer-motion variants
   740→   740→  const containerVariants: Variants = {
   741→   741→    hidden: { opacity: 0 },
   742→   742→    visible: {
   743→   743→      opacity: 1,
