   148→/**
   149→ * Render a type-aware editor cell for inline editing.
   150→ * - BOOLEAN  -> Switch
   151→ * - INTEGER/DECIMAL -> number Input
   152→ * - JSON/TIMESTAMP -> wider Input
   153→ * - TEXT/default  -> Input
   154→ */
   155→function renderCellInput(
   156→  col: ColumnItem,
   157→  value: unknown,
   158→  onChange: (v: unknown) => void,
   159→  disabled?: boolean,
   160→) {
   161→  const typeUpper = col.type.toUpperCase()
   162→  if (typeUpper === 'BOOLEAN') {
   163→    return (
   164→      <Switch
   165→        checked={Boolean(value)}
   166→        onCheckedChange={(v) => onChange(v)}
   167→        disabled={disabled}
   168→        aria-label={col.name}
   169→      />
   170→    )
   171→  }
   172→  if (typeUpper === 'INTEGER' || typeUpper === 'DECIMAL') {
   173→    return (
   174→      <Input
   175→        type="number"
   176→        value={value === null || value === undefined ? '' : String(value)}
   177→        onChange={(e) => {
   178→          const raw = e.target.value
   179→          if (raw === '') {
   180→            onChange(null)
   181→          } else if (typeUpper === 'INTEGER') {
   182→            const n = parseInt(raw, 10)
   183→            onChange(Number.isNaN(n) ? null : n)
   184→          } else {
   185→            const n = parseFloat(raw)
   186→            onChange(Number.isNaN(n) ? null : n)
   187→          }
   188→        }}
   189→        disabled={disabled}
   190→        className="h-8 font-mono text-xs min-w-[80px]"
   191→        aria-label={col.name}
   192→      />
   193→    )
   194→  }
   195→  if (typeUpper === 'JSON' || typeUpper === 'TIMESTAMP') {
   196→    return (
   197→      <Input
   198→        value={value === null || value === undefined ? '' : String(value)}
   199→        onChange={(e) => onChange(e.target.value)}
   200→        disabled={disabled}
   201→        className="h-8 font-mono text-xs min-w-[140px]"
   202→        placeholder={typeUpper === 'JSON' ? '{...}' : '2024-01-01T00:00:00Z'}
   203→        aria-label={col.name}
   204→      />
   205→    )
   206→  }
   207→  return (
   208→    <Input
   209→      value={value === null || value === undefined ? '' : String(value)}
   210→      onChange={(e) => onChange(e.target.value)}
   211→      disabled={disabled}
   212→      className="h-8 font-mono text-xs min-w-[100px]"
   213→      aria-label={col.name}
   214→    />
   215→  )
   216→}

   542→        ids.map(async (id) => {

   546→          } catch {
   547→            failCount += 1

   552→      setSelectedRowIds(new Set())
   553→      await refreshRows()
   554→      if (failCount === 0) {
   555→        toast({
   556→          title: 'Rows deleted',
   557→          description: `${okCount} row${okCount !== 1 ? 's' : ''} removed`,
   558→        })
   559→      } else {

   561→          title: 'Bulk delete partial',
   562→          description: `${okCount} deleted, ${failCount} failed`,
   563→          variant: 'destructive',
   564→        })
   565→      }
   566→    } finally {
   567→      setBulkDeleting(false)
   568→      setBulkDeleteOpen(false)
