   108→// Normalizes API responses that may be either a raw array or { data: [...] }
   109→function normalize(data: any): any[] {
   110→  if (Array.isArray(data)) return data
   111→  if (Array.isArray(data?.data)) return data.data
   112→  return []

   143→
   144→function FooterStatusBar() {
   145→  const [stats, setStats] = useState<FooterStats | null>(null)
   146→
   147→  useEffect(() => {
   148→    let active = true
   149→    ;(async () => {
   150→      try {
   151→        const [uptimeRes, tablesRes, pipelinesRes, hbRes] = await Promise.all([
   152→          fetch('/api/monitoring/uptime').then((r) => r.json()).catch(() => null),
   153→          fetch('/api/tables').then((r) => r.json()).catch(() => null),
   154→          fetch('/api/pipelines?isActive=true').then((r) => r.json()).catch(() => null),
   155→          fetch('/api/monitoring/heartbeat?limit=1').then((r) => r.json()).catch(() => null),
   156→        ])
   157→        if (!active) return
   158→
   159→        const uptime = uptimeRes?.data ?? uptimeRes
   160→        const tablesArr = Array.isArray(tablesRes) ? tablesRes : tablesRes?.data ?? []
   161→        const pipelinesArr = Array.isArray(pipelinesRes) ? pipelinesRes : pipelinesRes?.data ?? []
   162→        const hbArr = Array.isArray(hbRes) ? hbRes : hbRes?.data ?? []
   163→
   164→        const lastHb = Array.isArray(hbArr) && hbArr.length > 0 ? hbArr[0] : null
   165→        let lastHbAgo = '—'
   166→        if (lastHb?.recordedAt) {
   167→          const diff = Date.now() - new Date(lastHb.recordedAt).getTime()
   168→          const sec = Math.floor(diff / 1000)
   169→          lastHbAgo = sec < 60 ? `${sec}s ago` : `${Math.floor(sec / 60)}m ago`
   170→        }
   171→
   172→        setStats({
   173→          uptimePercent: uptime?.uptimePercent ?? 100,
   174→          tables: tablesArr.length ?? 0,
   175→          pipelines: pipelinesArr.length ?? 0,
   176→          lastHeartbeatAgo: lastHbAgo,
   177→        })
   178→      } catch {
   179→        // silently ignore
   180→      }
   181→    })()
   182→    return () => { active = false }
   183→  }, [])
   184→
   185→  if (!stats) {
   186→    return <span className="hidden sm:inline text-muted-foreground/50">Loading...</span>
   187→  }
   188→
   189→  return (
   190→    <span className="hidden sm:flex items-center gap-3 text-[11px]">
   191→      <span>Uptime: <span className="text-emerald-600 dark:text-emerald-400 font-medium">{stats.uptimePercent.toFixed(1)}%</span></span>
   192→      <span className="text-border">|</span>
   193→      <span>Tables: <span className="font-medium">{stats.tables}</span></span>
   194→      <span className="text-border">|</span>
   195→      <span>Pipelines: <span className="font-medium">{stats.pipelines}</span></span>
   196→      <span className="text-border">|</span>
   197→      <span>Last HB: <span className="font-medium">{stats.lastHeartbeatAgo}</span></span>
   198→    </span>
   199→  )
   200→}

   586→// =====================================================================
   587→// ROOT: Auth Wrapper
   588→// =====================================================================
   589→
   590→export default function SelfBaseApp() {
   591→  const [authState, setAuthState] = useState<AuthState>('checking')
   592→  const [currentUser, setCurrentUser] = useState<UserPayload | null>(null)
   593→  const [authToken, setAuthToken] = useState<string | null>(null)
   594→
   595→  // Check existing session on mount
   596→  useEffect(() => {
   597→    let cancelled = false
   598→    ;(async () => {
   599→      try {
   600→        const token = localStorage.getItem('sb_auth_token')
   601→        if (!token) {
   602→          if (!cancelled) setAuthState('unauthenticated')
   603→          return
   604→        }
   605→
   606→        const res = await fetch('/api/auth/me', {
   607→          headers: { Authorization: `Bearer ${token}` },
   608→        })
   609→
   610→        if (!res.ok) {
   611→          localStorage.removeItem('sb_auth_token')
   612→          if (!cancelled) setAuthState('unauthenticated')
   613→          return
   614→        }
   615→
   616→        const data = await res.json()
   617→        const user: UserPayload = data.user
   618→
   619→        setAuthToken(token)
   620→        setCurrentUser(user)
   621→
   622→        if (user.mustChangePassword) {
   623→          if (!cancelled) setAuthState('must-change-password')
   624→        } else {
   625→          if (!cancelled) setAuthState('authenticated')
   626→        }
   627→      } catch {
   628→        localStorage.removeItem('sb_auth_token')
   629→        if (!cancelled) setAuthState('unauthenticated')
   630→      }
   631→    })()
   632→    return () => { cancelled = true }
   633→  }, [])
   634→
   635→  const handleLogin = useCallback((token: string, user: UserPayload) => {
   636→    setAuthToken(token)
   637→    setCurrentUser(user)
   638→    if (user.mustChangePassword) {
   639→      setAuthState('must-change-password')
   640→    } else {
   641→      setAuthState('authenticated')
   642→    }
   643→  }, [])
   644→
   645→  const handlePasswordChanged = useCallback(() => {
   646→    if (currentUser) {
   647→      const updated = { ...currentUser, mustChangePassword: false }
   648→      setCurrentUser(updated)
   649→    }
   650→    setAuthState('authenticated')
   651→  }, [currentUser])
   652→
   653→  const handleLogout = useCallback(async () => {
   654→    try {
   655→      if (authToken) {
   656→        await fetch('/api/auth/logout', {
   657→          method: 'POST',
   658→          headers: { Authorization: `Bearer ${authToken}` },
   659→        })
   660→      }
   661→    } catch {
   662→      // ignore
   663→    }
   664→    localStorage.removeItem('sb_auth_token')
   665→    setAuthToken(null)
   666→    setCurrentUser(null)
   667→    setAuthState('unauthenticated')
   668→  }, [authToken])
   669→
   670→  // Loading state
   671→  if (authState === 'checking') {
   672→    return (
   673→      <div className="flex min-h-svh items-center justify-center bg-gradient-to-br from-emerald-50 via-white to-teal-50 dark:from-gray-950 dark:via-gray-900 dark:to-gray-950">
   674→        <div className="flex flex-col items-center gap-4">
   675→          <div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-gradient-to-br from-emerald-500 to-teal-600 animate-pulse">
   676→            <Shield className="h-6 w-6 text-white" />
   677→          </div>
   678→          <div className="flex items-center gap-2">
   679→            <Loader2 className="h-4 w-4 animate-spin text-emerald-600" />
   680→            <span className="text-sm text-muted-foreground">Loading SelfBase...</span>
   681→          </div>
   682→        </div>
   683→      </div>
   684→    )
   685→  }
   686→
   687→  // Not authenticated → show login
   688→  if (authState === 'unauthenticated') {
   689→    return <LoginPage onLogin={handleLogin} />
   690→  }
   691→
   692→  // Must change password → force change
   693→  if (authState === 'must-change-password' && currentUser && authToken) {
   694→    return (
   695→      <ForceChangePassword
   696→        user={currentUser}
   697→        token={authToken}
   698→        onPasswordChanged={handlePasswordChanged}
   699→      />
   700→    )
   701→  }
   702→
   703→  // Authenticated → show main app
   704→  if (authState === 'authenticated' && currentUser && authToken) {
   705→    return <AdminStudio user={currentUser} token={authToken} onLogout={handleLogout} />
   706→  }
   707→
   708→  // Fallback
   709→  return <LoginPage onLogin={handleLogin} />
   710→}
