    73→interface FunctionRunItem {
    74→  id: string
    75→  status: 'pending' | 'running' | 'success' | 'failed' | 'timeout'
    76→  triggeredBy: string | null
    77→  durationMs: number | null
    78→  memoryUsedMb: number | null
    79→  startedAt: string
    80→  completedAt: string | null
    81→  errorPayload?: string | null
    82→  output?: string | null
    83→}

    85→interface SbFunctionItem {
    86→  id: string
    87→  name: string
    88→  description: string | null
    89→  code: string
    90→  runtime: string
    91→  triggerType: 'http' | 'schedule' | 'event'
    92→  triggerConfig: string | null
    93→  envVars?: string | null
    94→  timeoutMs: number
    95→  memoryMb: number
    96→  isActive: boolean
    97→  createdAt: string
    98→  functionRuns?: FunctionRunItem[]
    99→}

   115→  running: Play,
   116→  success: CheckCircle2,

   122→  pending: 'text-amber-500',
   123→  running: 'text-blue-500',
   124→  success: 'text-emerald-500',
   125→  failed: 'text-red-500',
   126→  timeout: 'text-amber-500',
   127→}

   130→  pending: 'bg-amber-500/10 text-amber-700 border-amber-200',
   131→  running: 'bg-blue-500/10 text-blue-700 border-blue-200',
   132→  success: 'bg-emerald-500/10 text-emerald-700 border-emerald-200',
   133→  failed: 'bg-red-500/10 text-red-700 border-red-200',
   134→  timeout: 'bg-amber-500/10 text-amber-700 border-amber-200',
   135→}

   137→const defaultCode = `export default async function handler(ctx) {
   138→  const { input, env } = ctx;
   139→  // Your code here
   140→  console.log('Input:', input);
   141→  return { ok: true, received: Object.keys(input || {}).length };
   142→}

   144→
   145→export function FunctionsView() {
   146→  const { toast } = useToast()
   147→  const [functions, setFunctions] = useState<SbFunctionItem[]>([])
   148→  const [loading, setLoading] = useState(true)
   149→  const [search, setSearch] = useState('')
   150→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   151→  const [selectedFunction, setSelectedFunction] = useState<SbFunctionItem | null>(null)
   152→  const [runningId, setRunningId] = useState<string | null>(null)
   153→  const [runResult, setRunResult] = useState<{ fn: SbFunctionItem; result: { runId: string; status: string; output?: unknown; durationMs?: number; error?: string } } | null>(null)
   154→
   155→  // Create form
   156→  const [newName, setNewName] = useState('')
   157→  const [newDesc, setNewDesc] = useState('')
   158→  const [newRuntime, setNewRuntime] = useState('javascript')
   159→  const [newTrigger, setNewTrigger] = useState<'http' | 'schedule' | 'event'>('http')
   160→  const [newTimeout, setNewTimeout] = useState('30000')
   161→  const [newMemory, setNewMemory] = useState('128')
   162→  const [newCron, setNewCron] = useState('')
   163→  const [newEventPattern, setNewEventPattern] = useState('')
   164→  const [newEnvVars, setNewEnvVars] = useState('{}')
   165→  const [codeText, setCodeText] = useState(defaultCode)
   166→
   167→  const loadAll = useCallback(async () => {
   168→    setLoading(true)
   169→    try {
   170→      const data = await apiGet<SbFunctionItem[]>('/api/functions')
   171→      setFunctions(Array.isArray(data) ? data : [])
   172→    } catch (err) {
   173→      toast({
   174→        title: 'Failed to load functions',
   175→        description: err instanceof Error ? err.message : 'Unknown error',
   176→        variant: 'destructive',
   177→      })
   178→    } finally {
   179→      setLoading(false)
   180→    }
   181→  }, [toast])
   182→
   183→  useEffect(() => {
   184→    void loadAll()
   185→  }, [loadAll])
   186→
   187→  const handleCreate = async () => {
   188→    if (!newName.trim()) {
   189→      toast({ title: 'Name required', variant: 'destructive' })
   190→      return
   191→    }
   192→    const triggerConfig: Record<string, unknown> = {}
   193→    if (newTrigger === 'schedule' && newCron) triggerConfig.cron = newCron
   194→    if (newTrigger === 'event' && newEventPattern) triggerConfig.pattern = newEventPattern
   195→    try {
   196→      const created = await apiPost<SbFunctionItem>('/api/functions', {
   197→        name: newName.trim(),
   198→        description: newDesc || null,
   199→        code: codeText,
   200→        runtime: newRuntime,
   201→        triggerType: newTrigger,
   202→        triggerConfig: Object.keys(triggerConfig).length > 0 ? JSON.stringify(triggerConfig) : null,
   203→        envVars: newEnvVars || null,
   204→        timeoutMs: Number(newTimeout) || 30000,
   205→        memoryMb: Number(newMemory) || 128,
   206→        isActive: true,
   207→      })
   208→      setFunctions((prev) => [{ ...created, functionRuns: [] }, ...prev])
   209→      setShowCreateDialog(false)
   210→      setNewName('')
   211→      setNewDesc('')
   212→      setNewRuntime('javascript')
   213→      setNewTrigger('http')
   214→      setNewTimeout('30000')
   215→      setNewMemory('128')
   216→      setNewCron('')
   217→      setNewEventPattern('')
   218→      setNewEnvVars('{}')
   219→      setCodeText(defaultCode)
   220→      toast({ title: 'Function created', description: `"${created.name}" has been deployed.` })
   221→    } catch (err) {
   222→      toast({
   223→        title: 'Failed to create function',
   224→        description: err instanceof Error ? err.message : 'Unknown error',
   225→        variant: 'destructive',
   226→      })
   227→    }
   228→  }
   229→
   230→  const handleToggleActive = async (fn: SbFunctionItem) => {
   231→    try {
   232→      await apiPut(`/api/functions/${fn.id}`, { isActive: !fn.isActive })
   233→      setFunctions((prev) =>
   234→        prev.map((f) => (f.id === fn.id ? { ...f, isActive: !f.isActive } : f)),
   235→      )
   236→      if (selectedFunction?.id === fn.id) {
   237→        setSelectedFunction((prev) => (prev ? { ...prev, isActive: !prev.isActive } : prev))
   238→      }
   239→      toast({ title: `Function ${fn.isActive ? 'deactivated' : 'activated'}` })
   240→    } catch (err) {
   241→      toast({
   242→        title: 'Failed to toggle function',
   243→        description: err instanceof Error ? err.message : 'Unknown error',
   244→        variant: 'destructive',
   245→      })
   246→    }
   247→  }
   248→
   249→  const handleRun = async (fn: SbFunctionItem) => {
   250→    setRunningId(fn.id)
   251→    try {
   252→      const result = await apiPost<{ runId: string; status: string; output?: unknown; durationMs?: number; error?: string }>(
   253→        `/api/functions/${fn.id}/run`,
   254→        {},
   255→      )
   256→      setRunResult({ fn, result })
   257→      toast({
   258→        title: result.status === 'success' ? 'Function executed' : 'Function run failed',
   259→        description:
   260→          result.status === 'success'
   261→            ? `Completed in ${result.durationMs ?? 0}ms.`
   262→            : result.error ?? `Status: ${result.status}`,
   263→        variant: result.status === 'success' ? 'default' : 'destructive',
   264→      })
   265→      await loadAll()
   266→      if (selectedFunction?.id === fn.id) {
   267→        const fresh = (await apiGet<SbFunctionItem[]>('/api/functions')).find((f) => f.id === fn.id)
   268→        if (fresh) setSelectedFunction(fresh)
   269→      }
   270→    } catch (err) {
   271→      toast({
   272→        title: 'Function run failed',
   273→        description: err instanceof Error ? err.message : 'Unknown error',
   274→        variant: 'destructive',
   275→      })
   276→    } finally {
   277→      setRunningId(null)
   278→    }
   279→  }
   280→
   281→  const handleDelete = async (fn: SbFunctionItem) => {
   282→    try {
   283→      await apiDelete(`/api/functions/${fn.id}`)
   284→      setFunctions((prev) => prev.filter((f) => f.id !== fn.id))
   285→      if (selectedFunction?.id === fn.id) setSelectedFunction(null)
   286→      toast({ title: 'Function deleted', variant: 'destructive' })
   287→    } catch (err) {
   288→      toast({
   289→        title: 'Failed to delete function',
   290→        description: err instanceof Error ? err.message : 'Unknown error',
   291→        variant: 'destructive',
   292→      })
   293→    }
   294→  }
   295→
   296→  const filteredFunctions = functions.filter(
   297→    (f) =>
   298→      f.name.toLowerCase().includes(search.toLowerCase()) ||
   299→      (f.description ?? '').toLowerCase().includes(search.toLowerCase()),
   300→  )
   301→
   302→  if (loading) {
   303→    return (
   304→      <div className="space-y-6">
   305→        <Skeleton className="h-8 w-48" />
   306→        <div className="grid gap-4 md:grid-cols-2">
   307→          {Array.from({ length: 4 }).map((_, i) => (
   308→            <Card key={i}>
   309→              <CardContent className="p-4">
   310→                <Skeleton className="h-40 w-full" />
   311→              </CardContent>
   312→            </Card>
   313→          ))}
   314→        </div>
   315→      </div>
   316→    )
   317→  }
   318→
   319→  if (selectedFunction) {
   320→    const TriggerIcon = triggerIcons[selectedFunction.triggerType]
   321→    const runs = selectedFunction.functionRuns ?? []
   322→    return (
   323→      <motion.div
   324→        initial={{ opacity: 0, y: 8 }}
   325→        animate={{ opacity: 1, y: 0 }}
   326→        transition={{ duration: 0.2 }}
   327→        className="space-y-6"
   328→      >
   329→        <div className="flex items-center gap-2 flex-wrap">
   330→          <Button variant="ghost" size="sm" onClick={() => setSelectedFunction(null)}>
   331→            <ArrowRight className="h-4 w-4 rotate-180" /> Back
   332→          </Button>
   333→          <div className="h-6 w-px bg-border" />
   334→          <div className="rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 p-2 text-white">
   335→            <Code2 className="h-5 w-5" />
   336→          </div>
   337→          <h1 className="text-2xl font-bold tracking-tight">{selectedFunction.name}</h1>
   338→          <Badge variant="outline" className={`capitalize ${triggerBadgeColors[selectedFunction.triggerType]}`}>
   339→            <TriggerIcon className="h-3 w-3 mr-1" />
   340→            {selectedFunction.triggerType}
   341→          </Badge>
   342→          <Badge variant={selectedFunction.isActive ? 'default' : 'secondary'}>
   343→            {selectedFunction.isActive ? 'Active' : 'Inactive'}
   344→          </Badge>
   345→          <div className="ml-auto flex gap-2">
   346→            <Button
   347→              size="sm"
   348→              onClick={() => void handleRun(selectedFunction)}
   349→              disabled={runningId === selectedFunction.id || !selectedFunction.isActive}
   350→            >
   351→              {runningId === selectedFunction.id ? (
   352→                <Loader2 className="mr-1 h-3.5 w-3.5 animate-spin" />
   353→              ) : (
   354→                <Play className="mr-1 h-3.5 w-3.5" />
   355→              )}
   356→              Run
   357→            </Button>
   358→          </div>
   359→        </div>
   360→
   361→        <div className="grid gap-4 md:grid-cols-4">
   362→          <Card>
   363→            <CardContent className="pt-4">
   364→              <div className="text-sm text-muted-foreground">Trigger</div>
   365→              <div className="flex items-center gap-1 text-lg font-bold capitalize">
   366→                <TriggerIcon className="h-4 w-4" />
   367→                {selectedFunction.triggerType}
   368→              </div>
   369→            </CardContent>
   370→          </Card>
   371→          <Card>
   372→            <CardContent className="pt-4">
   373→              <div className="text-sm text-muted-foreground">Timeout</div>
   374→              <div className="text-lg font-bold">{selectedFunction.timeoutMs / 1000}s</div>
   375→            </CardContent>
   376→          </Card>
   377→          <Card>
   378→            <CardContent className="pt-4">
   379→              <div className="text-sm text-muted-foreground">Memory</div>
   380→              <div className="text-lg font-bold">{selectedFunction.memoryMb} MB</div>
   381→            </CardContent>
   382→          </Card>
   383→          <Card>
   384→            <CardContent className="pt-4">
   385→              <div className="text-sm text-muted-foreground">Total Runs</div>
   386→              <div className="text-lg font-bold">{runs.length}</div>
   387→            </CardContent>
   388→          </Card>
   389→        </div>
   390→
   391→        <Card>
   392→          <CardHeader className="flex flex-row items-center justify-between">
   393→            <div>
   394→              <CardTitle className="text-base">Source Code</CardTitle>
   395→              <CardDescription>
   396→                {selectedFunction.runtime} · {selectedFunction.code.split('\n').length} lines
   397→              </CardDescription>
   398→            </div>
   399→            <div className="flex gap-2">
   400→              <Tooltip>
   401→                <TooltipTrigger asChild>
   402→                  <Button
   403→                    variant="outline"
   404→                    size="sm"
   405→                    onClick={() => {
   406→                      navigator.clipboard.writeText(selectedFunction.code)
   407→                      toast({ title: 'Code copied' })
   408→                    }}
   409→                  >
   410→                    <Copy className="h-3.5 w-3.5 mr-1" /> Copy
   411→                  </Button>
   412→                </TooltipTrigger>
   413→                <TooltipContent>Copy code to clipboard</TooltipContent>
   414→              </Tooltip>
   415→              <Button size="sm" onClick={() => void handleRun(selectedFunction)} disabled={runningId === selectedFunction.id || !selectedFunction.isActive}>
   416→                <Play className="h-3.5 w-3.5 mr-1" /> Run
   417→              </Button>
   418→            </div>
   419→          </CardHeader>
   420→          <CardContent>
   421→            <div className="rounded-md bg-slate-950 text-emerald-400 p-4 overflow-auto max-h-[400px]">
   422→              <pre className="text-sm font-mono whitespace-pre-wrap">{selectedFunction.code}</pre>
   423→            </div>
   424→          </CardContent>
   425→        </Card>
   426→
   427→        {selectedFunction.triggerConfig && (
   428→          <Card>
   429→            <CardHeader>
   430→              <CardTitle className="text-base">Trigger Configuration</CardTitle>
   431→              <CardDescription>How this function is invoked</CardDescription>
   432→            </CardHeader>
   433→            <CardContent>
   434→              <div className="rounded-md bg-muted/50 p-3">
   435→                <pre className="text-sm font-mono">
   436→                  {JSON.stringify(parseJsonField(selectedFunction.triggerConfig, {}), null, 2)}
   437→                </pre>
   438→              </div>
   439→            </CardContent>
   440→          </Card>
   441→        )}
   442→
   443→        <Card>
   444→          <CardHeader>
   445→            <CardTitle className="text-base">Run History</CardTitle>
   446→            <CardDescription>Recent function executions</CardDescription>
   447→          </CardHeader>
   448→          <CardContent>
   449→            {runs.length === 0 ? (
   450→              <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   451→                <Terminal className="h-10 w-10 mb-2 opacity-40" />
   452→                <p className="text-sm">No runs yet</p>
   453→                <p className="text-xs mt-1">Click "Run" to execute this function.</p>
   454→              </div>
   455→            ) : (
   456→              <div className="rounded-md border max-h-96 overflow-y-auto">
   457→                <Table>
   458→                  <TableHeader className="sticky top-0 bg-background">
   459→                    <TableRow>
   460→                      <TableHead>Status</TableHead>
   461→                      <TableHead>Trigger</TableHead>
   462→                      <TableHead>Duration</TableHead>
   463→                      <TableHead>Memory</TableHead>
   464→                      <TableHead>Started</TableHead>
   465→                    </TableRow>
   466→                  </TableHeader>
   467→                  <TableBody>
   468→                    {runs.map((run) => {
   469→                      const RunIcon = statusIcons[run.status]
   470→                      return (
   471→                        <TableRow key={run.id} className="hover:bg-muted/40">
   472→                          <TableCell>
   473→                            <div className="flex items-center gap-1.5">
   474→                              <RunIcon className={`h-3.5 w-3.5 ${statusColors[run.status]}`} />
   475→                              <span className={`text-sm capitalize font-medium ${statusColors[run.status]}`}>
   476→                                {run.status}
   477→                              </span>
   478→                            </div>
   479→                          </TableCell>
   480→                          <TableCell>
   481→                            <Badge variant="outline" className="text-xs capitalize">
   482→                              {run.triggeredBy ?? '—'}
   483→                            </Badge>
   484→                          </TableCell>
   485→                          <TableCell className="font-mono text-sm">
   486→                            {run.durationMs ? `${run.durationMs}ms` : '—'}
   487→                          </TableCell>
   488→                          <TableCell className="text-sm">
   489→                            {run.memoryUsedMb ? `${Math.round(run.memoryUsedMb)} MB` : '—'}
   490→                          </TableCell>
   491→                          <TableCell className="text-sm text-muted-foreground">
   492→                            {new Date(run.startedAt).toLocaleString()}
   493→                          </TableCell>
   494→                        </TableRow>
   495→                      )
   496→                    })}
   497→                  </TableBody>
   498→                </Table>
   499→              </div>
   500→            )}
   501→          </CardContent>
   502→        </Card>
   503→
   504→        {/* Run Result Dialog */}
   505→        <Dialog open={!!runResult} onOpenChange={(open) => !open && setRunResult(null)}>
   506→          <DialogContent className="max-w-2xl">
   507→            <DialogHeader>
   508→              <DialogTitle className="flex items-center gap-2">
   509→                {runResult?.result.status === 'success' ? (
   510→                  <CheckCircle2 className="h-4 w-4 text-emerald-600" />
   511→                ) : (
   512→                  <XCircle className="h-4 w-4 text-red-500" />
   513→                )}
   514→                Function Result: {runResult?.fn.name}
   515→              </DialogTitle>
   516→              <DialogDescription>
   517→                {runResult?.result.status === 'success' ? 'Executed successfully' : 'Execution failed'}
   518→                {runResult?.result.durationMs ? ` in ${runResult.result.durationMs}ms` : ''}
   519→              </DialogDescription>
   520→            </DialogHeader>
   521→            <div className="space-y-3">
   522→              <div className="flex gap-3 text-xs">
   523→                <Badge variant={runResult?.result.status === 'success' ? 'default' : 'destructive'} className="capitalize">
   524→                  {runResult?.result.status}
   525→                </Badge>
   526→                {runResult?.result.durationMs && (
   527→                  <Badge variant="outline">Duration: {runResult.result.durationMs}ms</Badge>
   528→                )}
   529→                <Badge variant="outline">Run ID: {runResult?.result.runId?.slice(0, 8)}...</Badge>
   530→              </div>
   531→              {runResult?.result.error && (
   532→                <div className="rounded-md bg-red-500/10 border border-red-200 p-3 text-sm text-red-700">
   533→                  {runResult.result.error}
   534→                </div>
   535→              )}
   536→              {runResult?.result.output !== undefined && (
   537→                <div>
   538→                  <div className="text-sm font-medium mb-1">Output</div>
   539→                  <div className="rounded-md bg-slate-950 text-emerald-400 p-3 overflow-auto max-h-[40vh]">
   540→                    <pre className="text-sm font-mono whitespace-pre-wrap">
   541→                      {typeof runResult.result.output === 'string'
   542→                        ? runResult.result.output
   543→                        : JSON.stringify(runResult.result.output, null, 2)}
   544→                    </pre>
   545→                  </div>
   546→                </div>
   547→              )}
   548→            </div>
   549→            <DialogFooter>
   550→              <Button variant="outline" onClick={() => setRunResult(null)}>Close</Button>
   551→              <Button
   552→                onClick={() => {
   553→                  if (runResult?.result.output) {
   554→                    navigator.clipboard.writeText(
   555→                      typeof runResult.result.output === 'string'
   556→                        ? runResult.result.output
   557→                        : JSON.stringify(runResult.result.output, null, 2),
   558→                    )
   559→                    toast({ title: 'Output copied' })
   560→                  }
   561→                }}
   562→              >
   563→                <Copy className="mr-1 h-3.5 w-3.5" /> Copy Output
   564→              </Button>
   565→            </DialogFooter>
   566→          </DialogContent>
   567→        </Dialog>
   568→      </motion.div>
   569→    )
   570→  }
