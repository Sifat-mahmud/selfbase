   134→const emptyTrend = (rowCount: number) => {
   135→  // Demo trend derived from row count for visual interest.
   136→  // Real trend would require an API endpoint.
   137→  if (rowCount === 0) return { delta: 0, direction: 'none' as const }
   138→  const delta = Math.round((rowCount % 13) - 5)
   139→  return { delta: Math.abs(delta), direction: delta >= 0 ? ('up' as const) : ('down' as const) }
   140→}

   142→export function TablesView() {
   143→  const { toast } = useToast()
   144→  const [tables, setTables] = useState<SbTableItem[]>([])
   145→  const [loading, setLoading] = useState(true)
   146→  const [search, setSearch] = useState('')
   147→  const [selectedTable, setSelectedTable] = useState<SbTableItem | null>(null)
   148→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   149→  const [showColumnDialog, setShowColumnDialog] = useState(false)
   150→  const [deleteTarget, setDeleteTarget] = useState<SbTableItem | null>(null)
   151→  const [showDataDialog, setShowDataDialog] = useState(false)
   152→  const [dataRows, setDataRows] = useState<SbRowItem[]>([])
   153→  const [dataLoading, setDataLoading] = useState(false)
   154→
   155→  // New table form
   156→  const [newTableName, setNewTableName] = useState('')
   157→  const [newTableDesc, setNewTableDesc] = useState('')
   158→  const [newTablePriority, setNewTablePriority] = useState(2)
   159→  const [newColumns, setNewColumns] = useState<ColumnItem[]>([
   160→    { id: 'c-init', name: 'id', type: 'TEXT', nullable: false, isPrimaryKey: true, isUnique: true, isIndexed: true, defaultValue: null, order: 0 },
   161→  ])
   162→
   163→  // New column form
   164→  const [newColumnName, setNewColumnName] = useState('')
   165→  const [newColumnType, setNewColumnType] = useState('TEXT')
   166→
   167→  const [sortField, setSortField] = useState<'name' | 'rowCount' | 'priority'>('name')
   168→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
   169→
   170→  const loadTables = useCallback(async () => {
   171→    setLoading(true)
   172→    try {
   173→      const data = await apiGet<SbTableItem[]>('/api/tables')
   174→      setTables(Array.isArray(data) ? data : [])
   175→    } catch (err) {
   176→      toast({
   177→        title: 'Failed to load tables',
   178→        description: err instanceof Error ? err.message : 'Unknown error',
   179→        variant: 'destructive',
   180→      })
   181→    } finally {
   182→      setLoading(false)
   183→    }
   184→  }, [toast])
   185→
   186→  useEffect(() => {
   187→    loadTables()
   188→  }, [loadTables])
   189→
   190→  const filteredTables = tables
   191→    .filter(
   192→      (t) =>
   193→        t.name.toLowerCase().includes(search.toLowerCase()) ||
   194→        (t.displayName ?? '').toLowerCase().includes(search.toLowerCase()),
   195→    )
   196→    .sort((a, b) => {
   197→      const dir = sortDir === 'asc' ? 1 : -1
   198→      if (sortField === 'name') return a.name.localeCompare(b.name) * dir
   199→      if (sortField === 'rowCount') return (a.rowCount - b.rowCount) * dir
   200→      return (a.priority - b.priority) * dir
   201→    })
   202→
   203→  const handleSort = (field: 'name' | 'rowCount' | 'priority') => {
   204→    if (sortField === field) setSortDir((d) => (d === 'asc' ? 'desc' : 'asc'))
   205→    else {
   206→      setSortField(field)
   207→      setSortDir('asc')
   208→    }
   209→  }
   210→
   211→  const handleCreateTable = async () => {
   212→    if (!newTableName.trim()) {
   213→      toast({ title: 'Name required', variant: 'destructive' })
   214→      return
   215→    }
   216→    try {
   217→      const created = await apiPost<SbTableItem>('/api/tables', {
   218→        name: newTableName.trim().toLowerCase().replace(/\s+/g, '_'),
   219→        displayName: newTableName.trim(),
   220→        description: newTableDesc || null,
   221→        priority: newTablePriority,
   222→      })
   223→      setTables((prev) => [created, ...prev])
   224→      setShowCreateDialog(false)
   225→      setNewTableName('')
   226→      setNewTableDesc('')
   227→      setNewTablePriority(2)
   228→      setNewColumns([
   229→        { id: 'c-init', name: 'id', type: 'TEXT', nullable: false, isPrimaryKey: true, isUnique: true, isIndexed: true, defaultValue: null, order: 0 },
   230→      ])
   231→      toast({
   232→        title: 'Table created',
   233→        description: `"${created.name}" has been created successfully.`,
   234→      })
   235→    } catch (err) {
   236→      toast({
   237→        title: 'Failed to create table',
   238→        description: err instanceof Error ? err.message : 'Unknown error',
   239→        variant: 'destructive',
   240→      })
   241→    }
   242→  }
   243→
   244→  const handleDeleteTable = async (table: SbTableItem) => {
   245→    try {
   246→      await apiDelete(`/api/tables/${table.id}`)
   247→      setTables((prev) => prev.filter((t) => t.id !== table.id))
   248→      if (selectedTable?.id === table.id) setSelectedTable(null)
   249→      toast({
   250→        title: 'Table deleted',
   251→        description: `"${table.name}" has been removed.`,
   252→        variant: 'destructive',
   253→      })
   254→    } catch (err) {
   255→      toast({
   256→        title: 'Failed to delete table',
   257→        description: err instanceof Error ? err.message : 'Unknown error',
   258→        variant: 'destructive',
   259→      })
   260→    } finally {
   261→      setDeleteTarget(null)
   262→    }
   263→  }
   264→
   265→  const handleAddColumn = () => {
   266→    if (!selectedTable || !newColumnName.trim()) return
   267→    const newCol: ColumnItem = {
   268→      id: `c${Date.now()}`,
   269→      name: newColumnName.trim().toLowerCase().replace(/\s+/g, '_'),
   270→      type: newColumnType,
   271→      nullable: true,
   272→      isPrimaryKey: false,
   273→      isUnique: false,
   274→      isIndexed: false,
   275→      defaultValue: null,
   276→      order: selectedTable.columns.length,
   277→    }
   278→    setTables((prev) =>
   279→      prev.map((t) => (t.id === selectedTable.id ? { ...t, columns: [...t.columns, newCol] } : t)),
   280→    )
   281→    setSelectedTable((prev) => (prev ? { ...prev, columns: [...prev.columns, newCol] } : prev))
   282→    setShowColumnDialog(false)
   283→    setNewColumnName('')
   284→    setNewColumnType('TEXT')
   285→    toast({ title: 'Column added', description: `"${newCol.name}" added to ${selectedTable.name}` })
   286→  }
   287→
   288→  const handleViewData = async (table: SbTableItem) => {
   289→    setShowDataDialog(true)
   290→    setDataLoading(true)
   291→    try {
   292→      const rows = await apiGet<SbRowItem[]>(`/api/tables/${table.id}/rows`)
   293→      setDataRows(Array.isArray(rows) ? rows : [])
   294→    } catch (err) {
   295→      toast({
   296→        title: 'Failed to load rows',
   297→        description: err instanceof Error ? err.message : 'Unknown error',
   298→        variant: 'destructive',
   299→      })
   300→      setDataRows([])
   301→    } finally {
   302→      setDataLoading(false)
   303→    }
   304→  }
   305→
   306→  if (loading) {
   307→    return (
   308→      <div className="space-y-6">
   309→        <div className="flex justify-between">
   310→          <Skeleton className="h-8 w-40" />
   311→          <Skeleton className="h-9 w-32" />
   312→        </div>
   313→        <div className="grid gap-4 md:grid-cols-3">
   314→          {Array.from({ length: 6 }).map((_, i) => (
   315→            <Card key={i}>
   316→              <CardContent className="p-4">
   317→                <Skeleton className="h-20 w-full" />
   318→              </CardContent>
   319→            </Card>
   320→          ))}
   321→        </div>
   322→      </div>
   323→    )
   324→  }
   325→
   326→  // Detail view
   327→  if (selectedTable) {
   328→    return (
   329→      <motion.div
   330→        initial={{ opacity: 0, y: 8 }}
   331→        animate={{ opacity: 1, y: 0 }}
   332→        transition={{ duration: 0.2 }}
   333→        className="space-y-6"
   334→      >
   335→        <div className="flex items-center gap-2 flex-wrap">
   336→          <Button variant="ghost" size="sm" onClick={() => setSelectedTable(null)}>
   337→            <ChevronRight className="h-4 w-4 rotate-180" />
   338→            Back
   339→          </Button>
   340→          <Separator orientation="vertical" className="h-6" />
   341→          <div className="rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 p-2 text-white">
   342→            <Database className="h-5 w-5" />
   343→          </div>
   344→          <h1 className="text-2xl font-bold tracking-tight">
   345→            {selectedTable.displayName || selectedTable.name}
   346→          </h1>
   347→          <Badge variant="outline" className="ml-2 font-mono text-xs">
   348→            {selectedTable.name}
   349→          </Badge>
   350→          {selectedTable.isSystem && (
   351→            <Badge variant="secondary" className="gap-1">
   352→              <Lock className="h-3 w-3" /> System
   353→            </Badge>
   354→          )}
   355→          <Badge variant="outline" className={priorityColors[selectedTable.priority]}>
   356→            {priorityLabels[selectedTable.priority]}
   357→          </Badge>
   358→          <div className="ml-auto flex gap-2">
   359→            <Button variant="outline" size="sm" onClick={() => handleViewData(selectedTable)}>
   360→              <Eye className="mr-1 h-3.5 w-3.5" /> View Data
   361→            </Button>
   362→            <Button
   363→              variant="outline"
   364→              size="sm"
   365→              onClick={() => {
   366→                void loadTables()
   367→                const fresh = tables.find((t) => t.id === selectedTable.id)
   368→                if (fresh) setSelectedTable(fresh)
   369→              }}
   370→            >
   371→              <RefreshCw className="mr-1 h-3.5 w-3.5" /> Refresh
   372→            </Button>
   373→          </div>
   374→        </div>
   375→
   376→        <div className="grid gap-4 md:grid-cols-4">
   377→          <Card>
   378→            <CardContent className="pt-4">
   379→              <div className="text-sm text-muted-foreground">Rows</div>
   380→              <div className="text-2xl font-bold">{selectedTable.rowCount.toLocaleString()}</div>
   381→            </CardContent>
   382→          </Card>
   383→          <Card>
   384→            <CardContent className="pt-4">
   385→              <div className="text-sm text-muted-foreground">Columns</div>
   386→              <div className="text-2xl font-bold">{selectedTable.columns.length}</div>
   387→            </CardContent>
   388→          </Card>
   389→          <Card>
   390→            <CardContent className="pt-4">
   391→              <div className="text-sm text-muted-foreground">Version</div>
   392→              <div className="text-lg font-bold font-mono">{selectedTable.versionHash}</div>
   393→            </CardContent>
   394→          </Card>
   395→          <Card>
   396→            <CardContent className="pt-4">
   397→              <div className="text-sm text-muted-foreground">Priority</div>
   398→              <div className="text-lg font-bold">{priorityLabels[selectedTable.priority]}</div>
   399→            </CardContent>
   400→          </Card>
   401→        </div>
   402→
   403→        <Card>
   404→          <CardHeader>
   405→            <CardTitle className="text-base bg-gradient-to-r from-emerald-600 to-teal-600 bg-clip-text text-transparent">
   406→              Features
   407→            </CardTitle>
   408→            <CardDescription>Table-level feature flags</CardDescription>
   409→          </CardHeader>
   410→          <CardContent>
   411→            <div className="flex flex-wrap gap-6">
   412→              <div className="flex items-center gap-2">
   413→                <Switch checked={selectedTable.enableRealtime} />
   414→                <Label className="flex items-center gap-1">
   415→                  <Radio className="h-3.5 w-3.5 text-emerald-600" /> Realtime
   416→                </Label>
   417→              </div>
   418→              <div className="flex items-center gap-2">
   419→                <Switch checked={selectedTable.enableEmbedding} />
   420→                <Label className="flex items-center gap-1">
   421→                  <Brain className="h-3.5 w-3.5 text-emerald-600" /> Embeddings
   422→                </Label>
   423→              </div>
   424→              <div className="flex items-center gap-2">
   425→                <Switch checked={selectedTable.rlsEnabled} />
   426→                <Label className="flex items-center gap-1">
   427→                  <Shield className="h-3.5 w-3.5 text-emerald-600" /> Row-Level Security
   428→                </Label>
   429→              </div>
   430→            </div>
   431→          </CardContent>
   432→        </Card>
   433→
   434→        <Card>
   435→          <CardHeader className="flex flex-row items-center justify-between">
   436→            <div>
   437→              <CardTitle className="text-base">Columns</CardTitle>
   438→              <CardDescription>Schema definition for this table</CardDescription>
   439→            </div>
   440→            <Dialog open={showColumnDialog} onOpenChange={setShowColumnDialog}>
   441→              <DialogTrigger asChild>
   442→                <Button size="sm">
   443→                  <Plus className="mr-1 h-4 w-4" /> Add Column
   444→                </Button>
   445→              </DialogTrigger>
   446→              <DialogContent>
   447→                <DialogHeader>
   448→                  <DialogTitle>Add Column</DialogTitle>
   449→                  <DialogDescription>Add a new column to {selectedTable.name}</DialogDescription>
   450→                </DialogHeader>
   451→                <div className="space-y-4 py-2">
   452→                  <div className="space-y-2">
   453→                    <Label>Column Name</Label>
   454→                    <Input
   455→                      value={newColumnName}
   456→                      onChange={(e) => setNewColumnName(e.target.value)}
   457→                      placeholder="column_name"
   458→                    />
   459→                  </div>
   460→                  <div className="space-y-2">
   461→                    <Label>Type</Label>
   462→                    <Select value={newColumnType} onValueChange={setNewColumnType}>
   463→                      <SelectTrigger>
   464→                        <SelectValue />
   465→                      </SelectTrigger>
   466→                      <SelectContent>
   467→                        {columnTypes.map((t) => (
   468→                          <SelectItem key={t} value={t}>
   469→                            {t}
   470→                          </SelectItem>
   471→                        ))}
   472→                      </SelectContent>
   473→                    </Select>
   474→                  </div>
   475→                </div>
   476→                <DialogFooter>
   477→                  <Button variant="outline" onClick={() => setShowColumnDialog(false)}>
   478→                    Cancel
   479→                  </Button>
   480→                  <Button onClick={handleAddColumn}>Add Column</Button>
   481→                </DialogFooter>
   482→              </DialogContent>
   483→            </Dialog>
   484→          </CardHeader>
   485→          <CardContent>
   486→            {selectedTable.columns.length === 0 ? (
   487→              <div className="flex flex-col items-center justify-center py-8 text-muted-foreground">
   488→                <Columns3 className="h-10 w-10 mb-2 opacity-40" />
   489→                <p className="text-sm">No columns defined yet</p>
   490→                <Button variant="outline" size="sm" className="mt-2" onClick={() => setShowColumnDialog(true)}>
   491→                  <Plus className="mr-1 h-3.5 w-3.5" /> Add first column
   492→                </Button>
   493→              </div>
   494→            ) : (
   495→              <Table>
   496→                <TableHeader>
   497→                  <TableRow>
   498→                    <TableHead className="w-8">#</TableHead>
   499→                    <TableHead>Name</TableHead>
   500→                    <TableHead>Type</TableHead>
   501→                    <TableHead>Nullable</TableHead>
   502→                    <TableHead>Primary Key</TableHead>
   503→                    <TableHead>Unique</TableHead>
   504→                    <TableHead>Indexed</TableHead>
   505→                    <TableHead>Default</TableHead>
   506→                    <TableHead className="w-8" />
   507→                  </TableRow>
   508→                </TableHeader>
   509→                <TableBody>
   510→                  {selectedTable.columns.map((col, i) => (
   511→                    <TableRow key={col.id} className="hover:bg-muted/40 transition-colors">
   512→                      <TableCell className="text-muted-foreground text-xs">{i + 1}</TableCell>
   513→                      <TableCell className="font-mono font-medium">{col.name}</TableCell>
   514→                      <TableCell>
   515→                        <Badge variant="outline" className="font-mono text-xs">
   516→                          {col.type}
   517→                        </Badge>
   518→                      </TableCell>
   519→                      <TableCell>
   520→                        {col.nullable ? (
   521→                          <span className="text-emerald-500">✓</span>
   522→                        ) : (
   523→                          <span className="text-muted-foreground">—</span>
   524→                        )}
   525→                      </TableCell>
   526→                      <TableCell>{col.isPrimaryKey ? <span>🔑</span> : <span className="text-muted-foreground">—</span>}</TableCell>
   527→                      <TableCell>
   528→                        {col.isUnique ? (
   529→                          <span className="text-emerald-500">✓</span>
   530→                        ) : (
   531→                          <span className="text-muted-foreground">—</span>
   532→                        )}
   533→                      </TableCell>
   534→                      <TableCell>
   535→                        {col.isIndexed ? (
   536→                          <span className="text-emerald-500">✓</span>
   537→                        ) : (
   538→                          <span className="text-muted-foreground">—</span>
   539→                        )}
   540→                      </TableCell>
   541→                      <TableCell className="font-mono text-xs text-muted-foreground">
   542→                        {col.defaultValue ?? '—'}
   543→                      </TableCell>
   544→                      <TableCell>
   545→                        <DropdownMenu>
   546→                          <DropdownMenuTrigger asChild>
   547→                            <Button variant="ghost" size="icon" className="h-7 w-7">
   548→                              <MoreHorizontal className="h-3.5 w-3.5" />
   549→                            </Button>
   550→                          </DropdownMenuTrigger>
   551→                          <DropdownMenuContent align="end">
   552→                            <DropdownMenuItem>
   553→                              <Pencil className="mr-2 h-3.5 w-3.5" /> Edit
   554→                            </DropdownMenuItem>
   555→                            <DropdownMenuItem>
   556→                              <Copy className="mr-2 h-3.5 w-3.5" /> Duplicate
   557→                            </DropdownMenuItem>
   558→                            <DropdownMenuSeparator />
   559→                            <DropdownMenuItem className="text-destructive">
   560→                              <Trash2 className="mr-2 h-3.5 w-3.5" /> Delete
   561→                            </DropdownMenuItem>
   562→                          </DropdownMenuContent>
   563→                        </DropdownMenu>
   564→                      </TableCell>
   565→                    </TableRow>
   566→                  ))}
   567→                </TableBody>
   568→              </Table>
   569→            )}
   570→          </CardContent>
   571→        </Card>
   572→
   573→        {/* View Data Dialog */}
   574→        <Dialog open={showDataDialog} onOpenChange={setShowDataDialog}>
   575→          <DialogContent className="max-w-5xl">
   576→            <DialogHeader>
   577→              <DialogTitle className="flex items-center gap-2">
